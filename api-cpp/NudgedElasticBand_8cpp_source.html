45// Nudged Elastic Band definitions
46// First constructor: Now delegates to the second constructor
47NudgedElasticBand::NudgedElasticBand(std::shared_ptr<Matter> initialPassed,
48 std::shared_ptr<Matter> finalPassed,
49 const Parameters &parametersPassed,
50 std::shared_ptr<Potential> potPassed)
51 : NudgedElasticBand(
52 [&]() {
53 auto &init_opt = parametersPassed.neb_options().initialization;
54 const size_t base_count =
55 parametersPassed.neb_options().image_count;
56 if (parametersPassed.neb_options().match_endpoints) {
57 auto aligned = eonc::IRACompare::alignReactantToProduct(
58 *initialPassed, *finalPassed, 1.0);
59 auto *log = eonc::log::get();
60 if (aligned.error != 0) {
61 QUILL_LOG_WARNING(
62 log,
63 "match_endpoints: IRA align failed (error {}), "
64 "interpolating the input order",
65 aligned.error);
66 } else {
67 QUILL_LOG_INFO(log,
68 "match_endpoints: Hausdorff {:.4f} A after IRA "
69 "permute+rotate of the reactant",
70 aligned.hausdorffDistance);
71 }
72 }
73
74 // Apply oversampling factor if flag exists
75 const size_t relax_count =
76 init_opt.oversampling
77 ? base_count * init_opt.oversampling_factor
78 : base_count;
79
80 std::vector<Matter> path;
81 switch (init_opt.method) {
82 case NEBInit::FILE: {
83 std::vector<fs::path> file_paths =
84 eonc::helpers::neb_paths::readFilePaths(init_opt.input_path);
85 path = eonc::helpers::neb_paths::filePathInit(
86 file_paths, *initialPassed, base_count);
87 // filePathInit reloads every frame, including ends the caller
88 // may already have minimized.
89 path.front() = Matter(*initialPassed);
90 path.back() = Matter(*finalPassed);
91 break;
92 }
93 case NEBInit::IDPP:
94 path = eonc::helpers::neb_paths::idppPath(
95 *initialPassed, *finalPassed, relax_count, parametersPassed);
96 break;
97 case NEBInit::IDPP_COLLECTIVE:
98 path = eonc::helpers::neb_paths::idppCollectivePath(
99 *initialPassed, *finalPassed, relax_count, parametersPassed);
100 break;
101 case NEBInit::SIDPP:
102 case NEBInit::SIDPP_ZBL:
103 path = eonc::helpers::neb_paths::sidppPath(
104 *initialPassed, *finalPassed, relax_count, parametersPassed,
105 (init_opt.method == NEBInit::SIDPP_ZBL));
106 break;
107 case NEBInit::LINEAR:
108 default:
109 path = eonc::helpers::neb_paths::linearPath(
110 *initialPassed, *finalPassed, base_count);
111 break;
112 }
113
114 // Decimate path back to the target count using the cubic spline
115 if (init_opt.oversampling && path.size() > (base_count + 2)) {
116 auto *log = eonc::log::get();
117 QUILL_LOG_INFO(log,
118 "Decimating oversampled path ({} images) to "
119 "{} images via cubic spline.",
120 path.size() - 2, base_count);
121
122 // Perform the Spline Resampling
123 path = eonc::helpers::neb_paths::resamplePath(path, base_count);
124
125 // POST-DECIMATION RE-RELAXATION
126 // The spline might have placed atoms in high-energy positions.
127 QUILL_LOG_INFO(
128 log, "Relaxing decimated path to restore IDPP surface...");
129
130 // Collective IDPP objective for the new reduced path
131 std::shared_ptr<ObjectiveFunction> post_decim_objf =
132 std::make_shared<CollectiveIDPPObjectiveFunction>(
133 path, parametersPassed);
134
135 // Wrap with ZBL if the original method used ZBL options
136 bool use_zbl = (init_opt.method == NEBInit::SIDPP_ZBL);
137 if (use_zbl) {
138 auto zbl_pot = eonc::helpers::neb_paths::createZBLPotential();
139 post_decim_objf = std::make_shared<ZBLRepulsiveIDPPObjective>(
140 post_decim_objf, zbl_pot, path, parametersPassed, 1.0);
141 }
142
143 auto optim = eonc::helpers::create::mkOptim(
144 post_decim_objf, parametersPassed.neb_options().opt_method,
145 parametersPassed);
146
147 // Run a short optimization
148 optim->run(
149 parametersPassed.neb_options().initialization.max_iterations,
150 parametersPassed.neb_options().initialization.max_move);
151 }
152 return path;
153 }(),
154 parametersPassed, potPassed) {}

156// Second constructor: Contains all the common setup code
157NudgedElasticBand::NudgedElasticBand(std::vector<Matter> initPath,
158 const Parameters &parametersPassed,
159 std::shared_ptr<Potential> potPassed)
160 : ci_enabled_{parametersPassed.neb_options().climbing_image.enabled},
161 params{parametersPassed},
162 pot{potPassed},
163 E_ref{0.0} {
164
165 log = eonc::log::get();
166 eonc::helpers::requireKnownConvergenceMetric(
167 params.optimizer_options().convergence_metric, "[Nudged Elastic Band]");
168 this->status = NEBStatus::INIT;
169 numImages = params.neb_options().image_count;
170 if (initPath.empty()) {
171 throw std::invalid_argument("NEB: initPath is empty");
172 }
173 if (initPath.size() != static_cast<size_t>(numImages + 2)) {
174 throw std::invalid_argument("NEB: initPath.size() must be image_count + 2");
175 }
176 atoms = initPath.front().numberOfAtoms();
177 for (size_t i = 1; i < initPath.size(); ++i) {
178 eonc::helpers::neb_paths::requireSameAtomCount(initPath.front(),
179 initPath[i], "path images");
180 }
181
182 // Common initialization logic
183 path.resize(numImages + 2);
184 tangent.resize(numImages + 2);
185 projectedForce.resize(numImages + 2);
186 extremumPosition.resize(2 * (numImages + 1));
187 extremumEnergy.resize(2 * (numImages + 1));
188 extremumCurvature.resize(2 * (numImages + 1));
189 numExtrema = 0;
190
191 // Create per-image potentials if needed for true parallel force evaluation
192 perImagePotentials_ =
193 pot->needsPerImageInstance() && params.main_options().parallel;
194 if (perImagePotentials_) {
195 QUILL_LOG_INFO(log,
196 "NEB: Creating per-image potential instances for "
197 "parallel force evaluation ({} images)",
198 numImages + 2);
199 }
200
201 for (long i = 0; i <= numImages + 1; i++) {
202 path[i] = std::make_shared<Matter>(std::move(initPath[i]));
203
204 // Give each INTERMEDIATE image its own potential for true parallelism.
205 // Endpoints (i=0, i=numImages+1) keep the shared pot -- they are only
206 // evaluated once during initialization and never in parallel.
207 if (perImagePotentials_ && i > 0 && i <= numImages) {
208 auto cloned = pot->clonePotential();
209 path[i]->setPotential(cloned ? cloned
210 : eonc::helpers::makePotential(params));
211 }
212
213 tangent[i] = std::make_shared<AtomMatrix>();
214 tangent[i]->resize(atoms, 3);
215 tangent[i]->setZero();
216 projectedForce[i] = std::make_shared<AtomMatrix>();
217 projectedForce[i]->resize(atoms, 3);
218 projectedForce[i]->setZero();
219 }
220
221 // Common final setup
222 movedAfterForceCall = true;
223 reactantEnergy = path[0]->getPotentialEnergy();
224 path[numImages + 1]->getPotentialEnergy();
225 climbingImage = 0;
226
227 // Setup springs
228 k_u = params.neb_options().spring.weighting.k_max;
229 k_l = params.neb_options().spring.weighting.k_min;
230 if (params.neb_options().spring.weighting.enabled) {
231 ksp = k_l;
232 } else {
233 ksp = params.neb_options().spring.constant;
234 }
235
236 // Cache strategies that are constant across iterations
237 tangentStrat_ = eonc::neb::buildTangentStrategy(params);
238 projectionStrat_ = eonc::neb::buildProjectionStrategy(params);
239
240 // Optional debugging setup
241 if (params.debug_options().estimate_neb_eigenvalues) {
242 eigenmode_solvers.resize(numImages + 2);
243 for (long i = 0; i <= numImages + 1; i++) {
244 eigenmode_solvers[i] =
245 eonc::buildEigenmodeStrategy(path[i], parametersPassed, pot);
246 }
247 }
248}

250NudgedElasticBand::NEBStatus NudgedElasticBand::compute() {
251 long iteration = 0;
252 this->status = NEBStatus::RUNNING;
253
254 QUILL_LOG_DEBUG(log, "Nudged elastic band calculation started.");
255
256 // Higher endpoint: springs stay at k_min until an image exceeds it.
257 E_ref = std::max(path[0]->getPotentialEnergy(),
258 path[numImages + 1]->getPotentialEnergy());
259
260 updateForces();
261
262 auto objf = std::make_shared<NEBObjectiveFunction>(this, params);
263
264 bool switched{false};
265 auto optim = eonc::helpers::create::mkOptim(
266 objf, params.neb_options().opt_method, params);
267 std::unique_ptr<Optimizer> refine_optim{nullptr};
268 if (params.optimizer_options().refine.method != OptType::None) {
269 refine_optim = eonc::helpers::create::mkOptim(
270 objf, params.optimizer_options().refine.method, params);
271 }
272 // OCINEB controller
273 auto ocinebCfg = eonc::neb::OCINEBController::fromParams(params);
274 eonc::neb::OCINEBController ocineb(ocinebCfg);
275 bool zoomDone{false};
276 long zoomStable{0};
277 long zoomPrevCI{-1};
278 long zoomAt{-1};
279
280 while (this->status != NEBStatus::GOOD) {
281 if (params.debug_options().write_movies &&
282 (iteration % params.debug_options().write_movies_interval == 0)) {
283 bool append = (iteration != 0);
284 if (!eonc::io::io_ok(eonc::neb::writePathCon(
285 path, tangent, eigenmode_solvers, numImages,
286 params.debug_options().estimate_neb_eigenvalues,
287 std::format("neb_path_{:03d}.con", iteration), iteration,
288 reactantEnergy))) {
289 QUILL_LOG_ERROR(log, "Failed to write NEB path movie for iteration {}",
290 iteration);
291 }
292
293 AtomMatrix maxTang;
294 if (maxEnergyImage == 0) {
295 maxTang =
296 path[0]->pbc(path[1]->getPositions() - path[0]->getPositions());
297 } else if (maxEnergyImage == static_cast<size_t>(numImages + 1)) {
298 maxTang = path[numImages]->pbc(path[numImages + 1]->getPositions() -
299 path[numImages]->getPositions());
300 } else {
301 maxTang = *tangent[maxEnergyImage];
302 }
303 eonc::safemath::safe_normalize_inplace(maxTang);
304 auto maxImageMetadata = eonc::io::ConFrameMetadata{};
305 maxImageMetadata.frame_index = static_cast<uint64_t>(maxEnergyImage);
306 maxImageMetadata.energy = path[maxEnergyImage]->getPotentialEnergy();
307 maxImageMetadata.neb_bead = static_cast<uint64_t>(maxEnergyImage);
308 maxImageMetadata.neb_band = static_cast<uint64_t>(iteration);
309 maxImageMetadata.scalars.push_back(
310 {"relative_energy",
311 path[maxEnergyImage]->getPotentialEnergy() - reactantEnergy});
312 maxImageMetadata.scalars.push_back(
313 {"parallel_force",
314 matDot(path[maxEnergyImage]->getForces(), maxTang)});
315 maxImageMetadata.strings.push_back({"movie_kind", "neb_maximage"});
316 if (!eonc::io::io_ok(path[maxEnergyImage]->matter2con(
317 "neb_maximage.con", append, &maxImageMetadata))) {
318 EONC_LOG_WARNING("Failed to write neb_maximage.con");
319 }
320 printImageData(true, iteration);
321 }
322
323 VectorXd pos = objf->getPositions();
324 double convForce = convergenceForce();
325
326 ocineb.updateStability(climbingImage);
327
328 if (iteration == 0) {
329 baseline_force = convForce;
330 ocineb.initBaseline(convForce);
331
332 // Log configuration banner
333 auto &ci_opt = params.neb_options().climbing_image;
334 auto &mmf_opt = ci_opt.ocineb;
335 auto fmt_trigger = [](double val) -> std::string {
336 if (val > 1e100)
337 return "INF";
338 return std::format("{:.4f}", val);
339 };
340
341 QUILL_LOG_INFO(
342 log,
343 "===============================================================");
344 QUILL_LOG_INFO(log, " NEB Optimization Configuration");
345 QUILL_LOG_INFO(
346 log,
347 "===============================================================");
348 QUILL_LOG_INFO(log, " {:<25} : {:.4f}", "Baseline Force", baseline_force);
349
350 std::string ci_status = ci_opt.enabled ? "ENABLED" : "DISABLED";
351 QUILL_LOG_INFO(log, " {:<25} : {}", "Climbing Image (CI)", ci_status);
352 if (ci_opt.enabled) {
353 double ci_rel_val = baseline_force * ci_opt.trigger_factor;
354 QUILL_LOG_INFO(log, " - {:<21} : {} (Factor: {:.2f})",
355 "Relative Trigger", fmt_trigger(ci_rel_val),
356 ci_opt.trigger_factor);
357 QUILL_LOG_INFO(log, " - {:<21} : {}", "Absolute Trigger",
358 fmt_trigger(ci_opt.trigger_force));
359 QUILL_LOG_INFO(log, " - {:<21} : {}", "Converged Only",
360 ci_opt.converged_only);
361 }
362
363 std::string mmf_status =
364 (ci_opt.enabled && mmf_opt.use_mmf) ? "ENABLED" : "DISABLED";
365 QUILL_LOG_INFO(log, " {:<25} : {}", "Hybrid MMF (OCINEB)", mmf_status);
366 if (ci_opt.enabled && mmf_opt.use_mmf) {
367 QUILL_LOG_INFO(log, " - {:<21} : {:.4f} (Factor: {:.2f})",
368 "Initial Threshold", ocineb.threshold(),
369 mmf_opt.trigger_factor);
370 QUILL_LOG_INFO(log, " - {:<21} : {:.4f}", "Absolute Floor",
371 mmf_opt.trigger_force);
372 QUILL_LOG_INFO(log, " - {:<21} : {:.4f}", "Angle Tolerance",
373 mmf_opt.angle_tol);
374 }
375 QUILL_LOG_INFO(
376 log,
377 "---------------------------------------------------------------");
378
379 EONC_LOG_DEBUG("{:>10s} {:>12s} {:>14s} {:>11s} {:>12s}", "iteration",
380 "step size",
381 params.optimizer_options().convergence_metric_label,
382 "max image", "max energy");
383 QUILL_LOG_DEBUG(
384 eonc::log::get(),
385 "---------------------------------------------------------------\n");
386 }
387
388 // CI active when force drops below relative threshold
389 bool ci_active =
390 params.neb_options().climbing_image.enabled &&
391 (convForce < baseline_force *
392 params.neb_options().climbing_image.trigger_factor ||
393 convForce < params.neb_options().climbing_image.trigger_force);
394
395 bool zoomedThisStep = false;
396 if (iteration && !zoomDone && params.neb_options().zoom.enabled &&
397 climbingImage > 0 && climbingImage + 1 < path.size()) {
398 if (static_cast<long>(climbingImage) == zoomPrevCI) {
399 ++zoomStable;
400 } else {
401 zoomPrevCI = static_cast<long>(climbingImage);
402 zoomStable = 1;
403 }
404 const auto &zoom = params.neb_options().zoom;
405 const double zoomForce =
406 zoom.activation_threshold > 0.0
407 ? zoom.activation_threshold
408 : 10.0 * params.neb_options().force_tolerance;
409 if (zoomStable >= zoom.stability_count && convForce < zoomForce) {
410 std::vector<double> energy;
411 energy.reserve(path.size());
412 for (const auto &image : path) {
413 energy.push_back(image->getPotentialEnergy());
414 }
415 const auto window =
416 eonc::neb::zoom::selectWindow(energy, climbingImage, zoom);
417 if (eonc::neb::zoom::redistributePath(path, window,
418 zoom.interpolation)) {
419 movedAfterForceCall = true;
420 optim = eonc::helpers::create::mkOptim(
421 objf, params.neb_options().opt_method, params);
422 zoomDone = true;
423 zoomedThisStep = true;
424 zoomAt = iteration;
425 QUILL_LOG_INFO(log, "Zoom-NEB: packed the band onto images [{}, {}]",
426 window.lo, window.hi);
427 }
428 }
429 }
430
431 if (iteration) {
432 // MMF triggering via controller. Skipped on the zoom step so the
433 // dimer sees the redistributed band, not the pre-zoom geometry.
434 if (!zoomedThisStep &&
435 ocineb.shouldTrigger(convForce, ci_active, climbingImage, numImages,
436 ocineb.stabilityCount())) {
437 auto result = ocineb.run(*this, convForce);
438
439 if (result.convergedAfterMMF) {
440 status = NEBStatus::GOOD;
441 break;
442 }
443 // Post-MMF arc-length reparameterization: pass full path
444 // (endpoints are fixed by resamplePathInPlace, only interior
445 // images are redistributed). Zero force-call cost; next NEB
446 // iteration recomputes all forces anyway.
447 bool didResample = false;
448 if (!result.convergedAfterMMF && result.newForce < convForce &&
449 path[climbingImage]->getPeriodic() &&
450 path[0]->numberOfAtoms() > 6) {
451 eonc::helpers::neb_paths::resamplePathInPlace(
452 std::span{path.data(), path.size()});
453 movedAfterForceCall = true;
454 didResample = true;
455 }
456
457 // Reset optimizer AFTER reparameterization so fresh L-BFGS
458 // starts from the redistributed positions.
459 if (result.shouldResetOptimizer || didResample) {
460 optim = eonc::helpers::create::mkOptim(
461 objf, params.neb_options().opt_method, params);
462 }
463 }
464
465 long iterLimit = params.neb_options().max_iterations;
466 if (zoomDone && params.neb_options().zoom.max_iterations > 0 &&
467 zoomAt >= 0) {
468 iterLimit = zoomAt + params.neb_options().zoom.max_iterations;
469 }
470 if (iteration >= iterLimit) {
471 status = NEBStatus::BAD_MAX_ITERATIONS;
472 break;
473 }
474
475 if (zoomedThisStep) {
476 iteration++;
477 continue;
478 }
479
480 // Set CI state so updateForces() inside the optimizer step
481 // applies the correct force projection.
482 setCIEnabled(ci_active);
483
484 auto &activeOptim =
485 (refine_optim &&
486 convForce <= params.optimizer_options().refine.threshold)
487 ? refine_optim
488 : optim;
489 if (refine_optim &&
490 convForce <= params.optimizer_options().refine.threshold &&
491 !switched) {
492 switched = true;
493 EONC_LOG_DEBUG("Switched to {}",
494 magic_enum::enum_name<OptType>(
495 params.optimizer_options().refine.method));
496 }
497 activeOptim->step(params.optimizer_options().max_move);
498
499 setCIEnabled(params.neb_options().climbing_image.enabled);
500 }
501
502 iteration++;
503
504 double dE = path[maxEnergyImage]->getPotentialEnergy() - reactantEnergy;
505 double stepSize = eonc::geometry::maxAtomMotionV(
506 path[0]->pbcV(objf->getPositions() - pos));
507 QUILL_LOG_DEBUG(log, "{:>10} {:>12.4e} {:>14.4e} {:>11} {:>12.4}",
508 iteration, stepSize, convergenceForce(), maxEnergyImage,
509 dE);
510
511 if (pot->getType() == PotType::CatLearn) {
512 if (objf->isUncertain()) {
513 QUILL_LOG_DEBUG(log, "NEB failed due to high uncertainty");
514 status = NEBStatus::MAX_UNCERTAINTY;
515 break;
516 } else if (objf->isConverged()) {
517 QUILL_LOG_DEBUG(log, "NEB converged\n");
518 status = NEBStatus::GOOD;
519 break;
520 }
521 } else {
522 if (objf->isConverged()) {
523 QUILL_LOG_DEBUG(log, "NEB converged\n");
524 status = NEBStatus::GOOD;
525 break;
526 }
527 }
528 }
529 return status;
530}

532// generate the force value that is compared to the convergence criterion
533double NudgedElasticBand::convergenceForce() {
534 if (movedAfterForceCall)
535 updateForces();
536
537 auto imageForce = [&](long i) -> double {
538 if (params.optimizer_options().convergence_metric == "norm") {
539 return projectedForce[i]->norm();
540 }
541 if (params.optimizer_options().convergence_metric == "max_atom") {
542 // Every image shares the reactant constraint mask.
543 return path[0]->maxFreeAtomForce(*projectedForce[i]);
544 }
545 if (params.optimizer_options().convergence_metric == "max_component") {
546 return projectedForce[i]->cwiseAbs().maxCoeff();
547 }
548 log = eonc::log::traceback();
549 QUILL_LOG_CRITICAL(
550 log, "[Nudged Elastic Band] unknown opt_convergence_metric: {}",
551 params.optimizer_options().convergence_metric);
552 throw std::invalid_argument(
553 std::format("[Nudged Elastic Band] unknown convergence_metric: {}",
554 params.optimizer_options().convergence_metric));
555 };
556
557 double bandMax = 0;
558 for (long i = 1; i <= numImages; ++i) {
559 bandMax = std::max(bandMax, imageForce(i));
560 }
561
562 const bool ciOnly = params.neb_options().climbing_image.converged_only &&
563 ci_enabled_ && climbingImage != 0;
564 if (!ciOnly) {
565 return bandMax;
566 }
567
568 const double ciForce = imageForce(climbingImage);
569 const double slack = params.neb_options().climbing_image.band_slack;
570 const double tol = params.neb_options().force_tolerance;
571 if (slack > 0.0 && bandMax > slack * tol) {
572 return bandMax;
573 }
574 return ciForce;
575}

577// Update the forces, do the projections, and add spring forces
578void NudgedElasticBand::updateForces(bool ci_active) {
579 // Update forces for all intermediate images. Prefer batched evaluation
580 // (single model.forward() over all dirty images, e.g. MetatomicPotential
581 // on GPU). Else fall back to per-image evaluation, which is itself
582 // thread-parallel when (a) the potential is thread-safe on the same
583 // instance, or (b) per-image instances were created (separate models).
584 if (pot->supportsBatchEvaluation() && numImages > 1) {
585 // Collect only images that need recomputation (positions changed).
586 // Materialize atomic numbers and cells first, then build the raw-pointer
587 // arrays after storage is stable. Otherwise vector growth can invalidate
588 // earlier .data() pointers and hand garbage cells/types to forceBatch().
589 std::vector<long> dirty; // indices into path[] (1-based)
590 dirty.reserve(numImages);
591 for (long i = 1; i <= numImages; i++) {
592 if (path[i]->needsForceUpdate()) {
593 dirty.push_back(i);
594 }
595 }
596
597 if (!dirty.empty()) {
598 auto nDirty = static_cast<long>(dirty.size());
599 std::vector<VectorXi> nrsStore;
600 std::vector<Matrix3d> boxStore;
601 std::vector<const double *> posVec, boxVec;
602 std::vector<const int *> nrsVec;
603 std::vector<double *> frcVec;
604 nrsStore.reserve(static_cast<size_t>(nDirty));
605 boxStore.reserve(static_cast<size_t>(nDirty));
606 posVec.reserve(static_cast<size_t>(nDirty));
607 boxVec.reserve(static_cast<size_t>(nDirty));
608 nrsVec.reserve(static_cast<size_t>(nDirty));
609 frcVec.reserve(static_cast<size_t>(nDirty));
610
611 for (long idx : dirty) {
612 nrsStore.push_back(path[idx]->getAtomicNrs());
613 // Isolated molecules still store a box for I/O. Pots that infer
614 // PBC from a non-zero cell must see a zero box, as
615 // Matter::computePotential does on the endpoints.
616 boxStore.push_back(path[idx]->getPeriodic() ? path[idx]->getCell()
617 : Matrix3d::Zero());
618 }
619 for (long j = 0; j < nDirty; j++) {
620 auto idx = dirty[static_cast<size_t>(j)];
621 posVec.push_back(path[idx]->getPositions().data());
622 nrsVec.push_back(nrsStore[static_cast<size_t>(j)].data());
623 frcVec.push_back(path[idx]->forcesData());
624 boxVec.push_back(boxStore[static_cast<size_t>(j)].data());
625 }
626
627 std::vector<double> energies(nDirty), variances(nDirty);
628 pot->forceBatch(nDirty, atoms, posVec.data(), nrsVec.data(),
629 frcVec.data(), energies.data(), variances.data(),
630 boxVec.data());
631 for (long j = 0; j < nDirty; j++) {
632 path[dirty[j]]->setComputedPotential(energies[j], variances[j]);
633 }
634 }
635 } else {
636 // Per-image evaluation (sequential or parallel threads)
637 bool canParallel =
638 eonc::potAllowsSharedInstance(*pot) || perImagePotentials_;
639 if (numImages > 1 && params.main_options().parallel && canParallel) {
640#ifdef EON_PARALLEL_NEB
641 // TBB-backed std::execution::par (meson -Dwith_parallel_neb=true).
642 // One thread per image oversubscribes a 20-bead band on 8 cores.
643 std::vector<long> beads(static_cast<size_t>(numImages));
644 std::iota(beads.begin(), beads.end(), 1);
645 std::for_each(std::execution::par, beads.begin(), beads.end(),
646 [this](long i) { path[i]->getForcesRaw(); });
647#else
648 // std::thread rather than std::jthread -- Apple Clang libc++ lacks the
649 // latter. Wrap launch + join so a throw from any lambda still joins the
650 // remaining threads before we rethrow; otherwise the unjoined std::thread
651 // destructors call std::terminate().
652 std::vector<std::thread> threads;
653 threads.reserve(static_cast<size_t>(numImages));
654 try {
655 for (long i = 1; i <= numImages; i++) {
656 threads.emplace_back([this, i] { path[i]->getForcesRaw(); });
657 }
658 for (auto &t : threads)
659 t.join();
660 } catch (...) {
661 for (auto &t : threads)
662 if (t.joinable())
663 t.join();
664 throw;
665 }
666#endif
667 } else {
668 for (long i = 1; i <= numImages; i++) {
669 path[i]->getForcesRaw();
670 }
671 }
672 }
673
674 // Find the highest energy non-endpoint image
675 auto first = path.begin() + 1;
676 auto last = path.begin() + numImages + 1;
677 auto it = std::max_element(
678 first, last,
679 [](const std::shared_ptr<Matter> &a, const std::shared_ptr<Matter> &b) {
680 return a->getPotentialEnergy() < b->getPotentialEnergy();
681 });
682 maxEnergyImage = std::distance(path.begin(), it);
683 double maxEnergy = (*it)->getPotentialEnergy();
684
685 // Update E_ref for energy weighting. The higher endpoint keeps the
686 // soft spring on the side below that minimum.
687 if (params.neb_options().spring.weighting.enabled) {
688 E_ref = std::max(path[0]->getPotentialEnergy(),
689 path[numImages + 1]->getPotentialEnergy());
690 }
691
692 // Climbing requires an interior peak above both fixed endpoints.
693 // A monotonic path retains the spring force on every interior image.
694 const double endpointEnergy = std::max(path.front()->getPotentialEnergy(),
695 path.back()->getPotentialEnergy());
696 const bool climb = ci_active && maxEnergy > endpointEnergy;
697 climbingImage = 0;
698
699 // Spring strategy must be rebuilt each iteration (depends on maxEnergy,
700 // E_ref). Tangent and projection strategies are cached as members.
701 auto spring = eonc::neb::buildSpringStrategy(params, path, numImages, atoms,
702 maxEnergy, E_ref);
703
704 // Pre-allocate temporaries outside the loop to avoid repeated heap
705 // allocation of Nx3 matrices (each ~8KB for 337 atoms).
706 AtomMatrix posDiffNext(atoms, 3), posDiffPrev(atoms, 3);
707
708 for (long i = 1; i <= numImages; i++) {
709 const AtomMatrix &force = path[i]->getForces();
710 const AtomMatrix &pos = path[i]->getPositions();
711 const AtomMatrix &posPrev = path[i - 1]->getPositions();
712 const AtomMatrix &posNext = path[i + 1]->getPositions();
713 double energy = path[i]->getPotentialEnergy();
714 double energyPrev = path[i - 1]->getPotentialEnergy();
715 double energyNext = path[i + 1]->getPotentialEnergy();
716 posDiffNext.noalias() = posNext - pos;
717 posDiffNext = path[i]->pbc(posDiffNext);
718 posDiffPrev.noalias() = pos - posPrev;
719 posDiffPrev = path[i]->pbc(posDiffPrev);
720 double distNext = posDiffNext.norm();
721 double distPrev = posDiffPrev.norm();
722
723 // Tangent via strategy dispatch
724 *tangent[i] = std::visit(
725 [&](auto &t) {
726 return t.compute(posDiffNext, posDiffPrev, energy, energyPrev,
727 energyNext);
728 },
729 tangentStrat_);
730
731 // Spring forces via strategy dispatch
732 eonc::neb::SpringResult springResult = std::visit(
733 [&](auto &s) -> eonc::neb::SpringResult {
734 using T = std::decay_t<decltype(s)>;
735 if constexpr (std::is_same_v<T, eonc::neb::UniformSpring>) {
736 this->ksp = s.ksp;
737 return s.compute(i, *tangent[i], distNext, distPrev, posDiffNext,
738 posDiffPrev, path[i]);
739 } else if constexpr (std::is_same_v<T, eonc::neb::WeightedSpring>) {
740 return s.compute(i, *tangent[i], distNext, distPrev);
741 } else {
742 return s.compute(i, *tangent[i], posNext, posPrev, pos, path[i]);
743 }
744 },
745 spring);
746
747 // Climbing image or projected force
748 if (climb && i == static_cast<long>(maxEnergyImage)) {
749 climbingImage = maxEnergyImage;
750 // CI force: F - 2*(F.t)*t, plus DNEB correction if active
751 AtomMatrix forceDNEB = AtomMatrix::Zero(atoms, 3);
752 if (const auto *dnebProj =
753 std::get_if<eonc::neb::DNEB_Projection>(&projectionStrat_)) {
754 AtomMatrix fPerp = eonc::neb::forcePerp(force, *tangent[i]);
755 forceDNEB = eonc::neb::computeDNEBComponent(springResult.forceSpring,
756 *tangent[i], fPerp,
757 dnebProj->use_switching);
758 }
759 *projectedForce[i] =
760 eonc::neb::climbingImageForce(force, *tangent[i], forceDNEB);
761 } else {
762 eonc::neb::ImageForceData data{force, *tangent[i], springResult,
763 path[i]->numberOfFreeAtoms(),
764 path[i]->numberOfAtoms()};
765 *projectedForce[i] = std::visit([&](auto &p) { return p.project(data); },
766 projectionStrat_);
767 }
768
769 // The spring force acts along the tangent, which has a component on a
770 // fixed atom whenever the endpoints place it differently; a fixed atom
771 // carries no force on the band.
772 if (path[i]->numberOfFreeAtoms() < path[i]->numberOfAtoms()) {
773 for (long j = 0; j < atoms; j++) {
774 if (path[i]->getFixed(j)) {
775 projectedForce[i]->row(j).setZero();
776 }
777 }
778 }
779
780 eonc::neb::zeroTranslation(*projectedForce[i], path[i]->numberOfFreeAtoms(),
781 path[i]->numberOfAtoms());
782 }
783
784 movedAfterForceCall = false;
785}
