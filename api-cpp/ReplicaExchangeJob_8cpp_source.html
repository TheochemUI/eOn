31std::vector<std::string> ReplicaExchangeJob::run() {
32 std::string posFilename =
33 eonc::helpers::getRelevantFile(params.main_options().conFilename);
34 pos = std::make_shared<Matter>(pot, params);
35 if (!eonc::io::io_ok(pos->con2matter(posFilename))) {
36 QUILL_LOG_CRITICAL(log, "Failed to load {}", posFilename);
37 throw std::runtime_error("failed to load " + posFilename);
38 }
39 (void)runFromMatter(pos);
40 return returnFiles;
41}

43std::shared_ptr<Matter>
44ReplicaExchangeJob::runFromMatter(std::shared_ptr<Matter> initial) {
45 if (!initial) {
46 throw std::runtime_error("ReplicaExchangeJob::runFromMatter: null Matter");
47 }
48 pos = initial;
49 pos->setPotential(pot);
50
51 auto &rex = ParametersLoadAccess::replica_exchange_options(params);
52 if (rex.replicas < 1) {
53 throw std::invalid_argument(
54 "ReplicaExchangeJob: replica_exchange.replicas must be >= 1");
55 }
56 if (rex.temperature_low <= 0.0) {
57 rex.temperature_low = params.main_options().temperature > 0.0
58 ? params.main_options().temperature
59 : 300.0;
60 }
61 if (rex.temperature_high <= rex.temperature_low) {
62 rex.temperature_high = rex.temperature_low * 1.5;
63 }
64
65 long samplingSteps =
66 static_cast<long>(params.replica_exchange_options().sampling_time /
67 params.dynamics_options().time_step +
68 0.5);
69 long exchangePeriodSteps =
70 static_cast<long>(params.replica_exchange_options().exchange_period /
71 params.dynamics_options().time_step +
72 0.5);
73 const double kB = params.constants().kB;
74 if (samplingSteps <= 0)
75 samplingSteps = 1;
76 if (exchangePeriodSteps <= 0)
77 exchangePeriodSteps = 1;
78
79 QUILL_LOG_DEBUG(log, "Running Replica Exchange");
80
81 long refForceCalls = PotRegistry::get().total_force_calls();
82
83 const long nReplicas = params.replica_exchange_options().replicas;
84 std::vector<std::shared_ptr<Matter>> replica(nReplicas);
85 std::vector<std::unique_ptr<Dynamics>> replicaDynamics(nReplicas);
86
87 // Per-image potentials: each replica gets its own potential instance when
88 // the potential requires it (e.g. ML potentials with internal state)
89 const bool perImage = pot->needsPerImageInstance();
90 for (long i = 0; i < nReplicas; i++) {
91 auto replicaPot = perImage ? eonc::helpers::sharePotential(
92 eonc::helpers::makePotential(params))
93 : pot;
94 replica[i] = std::make_shared<Matter>(replicaPot, params);
95 *replica[i] = *pos;
96 replica[i]->setPotential(replicaPot);
97 replicaDynamics[i] = std::make_unique<Dynamics>(replica[i].get(), params);
98 }
99
100 std::vector<double> replicaTemperature(nReplicas);
101
102 QUILL_LOG_DEBUG(log, "Temperature distribution:");
103 if (nReplicas < 2) {
104 replicaTemperature[0] = params.replica_exchange_options().temperature_low;
105 replicaDynamics[0]->setTemperature(replicaTemperature[0]);
106 replicaDynamics[0]->setThermalVelocity();
107 } else if (params.replica_exchange_options().temperature_distribution ==
108 "linear") {
109 for (long i = 0; i < nReplicas; i++) {
110 replicaTemperature[i] =
111 params.replica_exchange_options().temperature_low +
112 static_cast<double>(i) / static_cast<double>(nReplicas - 1) *
113 (params.replica_exchange_options().temperature_high -
114 params.replica_exchange_options().temperature_low);
115 replicaDynamics[i]->setTemperature(replicaTemperature[i]);
116 replicaDynamics[i]->setThermalVelocity();
117 }
118 } else if (params.replica_exchange_options().temperature_distribution ==
119 "exponential") {
120 double kTemp = std::log(params.replica_exchange_options().temperature_high /
121 params.replica_exchange_options().temperature_low) /
122 static_cast<double>(nReplicas - 1);
123 for (long i = 0; i < nReplicas; i++) {
124 replicaTemperature[i] =
125 params.replica_exchange_options().temperature_low *
126 std::exp(kTemp * static_cast<double>(i));
127 replicaDynamics[i]->setTemperature(replicaTemperature[i]);
128 replicaDynamics[i]->setThermalVelocity();
129 QUILL_LOG_DEBUG(log, "replica: {} temperature {:.0f}", i + 1,
130 replicaTemperature[i]);
131 }
132 } else {
133 throw std::invalid_argument(
134 "replica_exchange.temperature_distribution must be linear or "
135 "exponential");
136 }
137
138 QUILL_LOG_DEBUG(
139 log, "Replica Exchange sampling for {:.0f} fs; {} steps; {} replicas.",
140 params.replica_exchange_options().sampling_time * 10.18, samplingSteps,
141 params.replica_exchange_options().replicas);
142
143 // Parallel replica dynamics when enabled and potential supports it
144 const bool canParallel = params.main_options().parallel &&
145 (eonc::potAllowsSharedInstance(*pot) || perImage);
146
147 for (long step = 1; step <= samplingSteps; step++) {
148 if (canParallel && nReplicas > 1) {
149 // Parallel: each replica runs its MD step in a separate thread
150 std::vector<std::thread> threads;
151 threads.reserve(nReplicas);
152 for (long i = 0; i < nReplicas; i++) {
153 threads.emplace_back([&, i] {
154 // New std::thread, new TLS ran2. Split the stream by replica and
155 // step so Langevin/Andersen noise is not identical across replicas.
156 const long userSeed = params.main_options().randomSeed;
157 const long base = (userSeed > 0) ? userSeed : 1;
158 eonc::rng::random(base + (step + 1) * 10007 + (i + 1));
159 replicaDynamics[i]->oneStep();
160 });
161 }
162 for (auto &t : threads) {
163 t.join();
164 }
165 } else {
166 for (long i = 0; i < nReplicas; i++) {
167 replicaDynamics[i]->oneStep();
168 }
169 }
170
171 // Metropolis replica exchange
172 if (nReplicas >= 2 && (step % exchangePeriodSteps) == 0) {
173 for (long trial = 0;
174 trial < params.replica_exchange_options().exchange_trials; trial++) {
175 long i = eonc::rng::randomInt(0, nReplicas - 2);
176 double energyLow = replica[i]->getPotentialEnergy();
177 double energyHigh = replica[i + 1]->getPotentialEnergy();
178 double kbTLow = kB * replicaTemperature[i];
179 double kbTHigh = kB * replicaTemperature[i + 1];
180 double pAcc =
181 std::min(1.0, std::exp((energyHigh - energyLow) *
182 (eonc::safemath::safe_recip(kbTHigh, 0.0) -
183 eonc::safemath::safe_recip(kbTLow, 0.0))));
184 double rnd = eonc::rng::randomDouble();
185 QUILL_LOG_INFO(log,
186 "step: {} trial swap, i {}, elow: {:.5f}, ehigh: "
187 "{:.5f}, pAcc: {:.5f}, rand: {}",
188 step, i, energyLow, energyHigh, pAcc, rnd);
189 if (rnd < pAcc) {
190 QUILL_LOG_INFO(log, "swap");
191 // Swap configurations, not pointers. Dynamics stays on its T slot.
192 auto potLow = replica[i]->getPotential();
193 auto potHigh = replica[i + 1]->getPotential();
194 Matter tmp = *replica[i];
195 *replica[i] = *replica[i + 1];
196 *replica[i + 1] = tmp;
197 replica[i]->setPotential(potLow);
198 replica[i + 1]->setPotential(potHigh);
199 replicaDynamics[i]->setThermalVelocity();
200 replicaDynamics[i + 1]->setThermalVelocity();
201 } else {
202 QUILL_LOG_INFO(log, "no swap");
203 }
204 }
205 }
206 }
207
208 forceCalls = PotRegistry::get().total_force_calls() - refForceCalls;
209 if (!replica.empty()) {
210 *pos = *replica[0];
211 }
212 saveData();
213 return pos;
214}

216void ReplicaExchangeJob::saveData() {
217 std::string resultsFilename("results.dat");
218 returnFiles.push_back(resultsFilename);
219
220 std::ofstream out(resultsFilename, std::ios::binary);
221 if (out) {
222 out << std::format("{} termination_reason\n", 0);
223 out << "GOOD termination_reason_text\n";
224 out << "replica_exchange job_type\n";
225 out << std::format("{} random_seed\n", params.main_options().randomSeed);
226 out << std::format(
227 "{} potential_type\n",
228 magic_enum::enum_name<PotType>(params.potential_options().potential));
229 out << std::format("{} force_calls_sampling\n", forceCalls);
230 }
231
232 std::string posFilename("pos_out.con");
233 returnFiles.push_back(posFilename);
234 if (pos) {
235 if (!eonc::io::io_ok(pos->matter2con(posFilename))) {
236 QUILL_LOG_ERROR(log, "Failed to write {}", posFilename);
237 }
238 }
239}
