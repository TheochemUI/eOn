162template <int N, class R, class F>
163void forcefields::PotentialBase::switching(ChargeGroup<N, R, F> &g1,
164 ChargeGroup<N, R, F> &g2,
165 double &energy, double cutoff,
166 double switchingWidth) {
167 double z[3], z1;
168 distance(g1.centre_, g2.centre_, z, z1);
169 assert(z1 <= cutoff);
170 assert(z1 >= cutoff - switchingWidth);
171 double x = (z1 - cutoff + switchingWidth) / switchingWidth;
172 double S = (2 * x - 3) * x * x + 1;
173 double dS = 6 * x * (x - 1);
174 double const u = energy;
175 for (int i = 0; i < N; ++i) {
176 for (int k = 0; k < 3; ++k) {
177 g1.force_[i][k] =
178 g1.force_[i][k] * S - u * dS / switchingWidth * z[k] / z1 / N;
179 g2.force_[i][k] =
180 g2.force_[i][k] * S + u * dS / switchingWidth * z[k] / z1 / N;
181 }
182 };
183 energy *= S;
184}

192double *forcefields::PotentialBase::crossProduct(const double v[],
193 const double w[], double e[]) {
194 e[0] = v[1] * w[2] - v[2] * w[1];
195 e[1] = v[2] * w[0] - v[0] * w[2];
196 e[2] = v[0] * w[1] - v[1] * w[0];
197 return e;
198}
