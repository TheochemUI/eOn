51uint64_t fnv1a64(const char *s) {
52 uint64_t h = 14695981039346656037ULL;
53 if (!s) {
54 return h;
55 }
56 for (const unsigned char *p = reinterpret_cast<const unsigned char *>(s); *p;
57 ++p) {
58 h ^= static_cast<uint64_t>(*p);
59 h *= 1099511628211ULL;
60 }
61 return h;
62}

75void fill_matter(Matter &m, const eon_relax_band_t *band, long image,
76 uint64_t epoch) {
77 m.resize(band->n_atoms);
78 Eigen::VectorXi z(band->n_atoms);
79 for (long a = 0; a < band->n_atoms; ++a) {
80 z(a) = band->atomic_nrs[a];
81 }
82 m.setAtomicNrs(z);
83 Eigen::VectorXd masses = Eigen::VectorXd::Ones(band->n_atoms);
84 if (band->version.minor >= 1 && band->masses) {
85 for (long a = 0; a < band->n_atoms; ++a) {
86 masses(a) = band->masses[a];
87 }
88 }
89 m.setMasses(masses);
90 AtomMatrix pos(band->n_atoms, 3);
91 const double *src = band->positions + image * 3 * band->n_atoms;
92 // Match Potential::force / Matter storage (AtomMatrix row-major xyz).
93 Eigen::Map<const AtomMatrix> mapped(src, band->n_atoms, 3);
94 pos = mapped;
95 const double *b = band->boxes + image * 9;
96 const Matrix3d cell = Eigen::Map<const Matrix3d>(b);
97 // Cell first. PBC stays off so dest optimizer writes
98 // (setPositionsFree) do not wrap a host-frame x=-0.667 to 9.333
99 // and hop the climbing image onto a wrapped shoulder.
100 m.setPeriodic(false);
101 m.setCell(cell);
102 m.setPositions(pos);
103 if (band->is_fixed) {
104 for (long a = 0; a < band->n_atoms; ++a) {
105 m.setFixed(a, band->is_fixed[a] ? 1 : 0);
106 }
107 } else {
108 for (long a = 0; a < band->n_atoms; ++a) {
109 m.setFixed(a, 0);
110 }
111 }
112 m.setSurfaceEpoch(epoch);
113}
114
115void write_image(double *dst, const Matter &m, long n_atoms) {
116 const AtomMatrix &pos = m.getPositions();
117 Eigen::Map<AtomMatrix> mapped(dst, n_atoms, 3);
118 mapped = pos;
119}

121int masses_ok(const eon_relax_band_t *band) {
122 if (!band || band->version.minor < 1 || !band->masses) {
123 return EON_RELAX_OK;
124 }
125 for (int64_t a = 0; a < band->n_atoms; ++a) {
126 const double m = band->masses[a];
127 if (!(m > 0.0) || !std::isfinite(m)) {
128 return EON_RELAX_INVALID_PARAMETER;
129 }
130 }
131 return EON_RELAX_OK;
132}

134void stamp_outcome(eon_relax_outcome_t *out, eon_relax_kind_t kind,
135 uint64_t epoch) {
136 std::memset(out, 0, sizeof(*out));
137 out->version.major = EON_RELAX_ABI_MAJOR;
138 out->version.minor = EON_RELAX_ABI_MINOR;
139 out->kind = static_cast<int32_t>(kind);
140 out->version_hash = eon_relax_version_hash();
141 out->surface_epoch = epoch;
142}

144void dirty_keep_frame(Matter &m) {
145 const bool pbc = m.getPeriodic();
146 m.setPeriodic(false);
147 m.setPositions(m.getPositions());
148 m.setPeriodic(pbc);
149}

151void dirty_endpoints(NudgedElasticBand &neb) {
152 dirty_keep_frame(*neb.path[0]);
153 dirty_keep_frame(*neb.path[static_cast<size_t>(neb.numImages + 1)]);

156int apply_relax_params(Parameters &params, eon_relax_kind_t *kind,
157 uint64_t *epoch, const void *config, size_t config_len,
158 char *errbuf, size_t errlen) {
159 if (config_len % sizeof(::capnp::word) != 0) {
160 set_err(errbuf, errlen, "-13 RelaxEngineParams capnp root");
161 return EON_RELAX_CAPNP_ROOT;
162 }
163 try {
164 const auto nwords = config_len / sizeof(::capnp::word);
165 std::vector<::capnp::word> words(nwords);
166 std::memcpy(words.data(), config, config_len);
167 ::kj::ArrayPtr<const ::capnp::word> view(words.data(), nwords);
168 ::capnp::FlatArrayMessageReader reader(view);
169 auto root = reader.getRoot<eonc::params_ssot::RelaxEngineParams>();
170 const auto kind_txt = root.getKind();
171 if (kind_txt == "neb") {
172 *kind = EON_RELAX_KIND_NEB;
173 } else if (kind_txt == "saddle") {
174 *kind = EON_RELAX_KIND_SADDLE;
175 } else {
176 set_err(errbuf, errlen, "-11 unknown RelaxEngineParams.kind");
177 return EON_RELAX_UNKNOWN_KIND;
178 }
179 auto neb = root.getNeb();
180 auto &neb_o = ParametersLoadAccess::neb_options(params);
181 neb_o.image_count = static_cast<long>(neb.getImageCount());
182 neb_o.max_iterations = static_cast<long>(neb.getMaxIterations());
183 neb_o.force_tolerance = neb.getForceTolerance();
184 neb_o.climbing_image.enabled = neb.getClimbingImage();
185 neb_o.endpoints.minimize = neb.getMinimizeEndpoints();
186 auto saddle = root.getSaddle();
187 auto &saddle_o = ParametersLoadAccess::saddle_search_options(params);
188 saddle_o.max_iterations = static_cast<long>(saddle.getMaxIterations());
189 saddle_o.converged_force = saddle.getConvergedForce();
190 *epoch = root.getSurfaceEpoch();
191 ParametersLoadAccess::main_options(params).randomSeed =
192 static_cast<long>(root.getRandomSeed());
193 ParametersLoadAccess::gp_surrogate_options(params).uncertainty =
194 root.getUncertainty();
195 return EON_RELAX_OK;
196 } catch (const std::exception &) {
197 set_err(errbuf, errlen, "-13 RelaxEngineParams capnp root");
198 return EON_RELAX_CAPNP_ROOT;
199 } catch (...) {
200 set_err(errbuf, errlen, "-13 RelaxEngineParams capnp root");
201 return EON_RELAX_CAPNP_ROOT;
202 }
203}

207struct EonRelaxEngine {
208 Parameters params;
209 eon_relax_kind_t kind{EON_RELAX_KIND_NEB};
210 uint64_t epoch{0};
211 int last_rc{EON_RELAX_OK};
212 bool unusable{false};
213
214 // Stepper state (eon_relax_step): band, objective, and optimizer
215 // persist across calls so quasi-Newton history accumulates on one
216 // surface epoch; eon_relax_reset drops all of it.
217 std::shared_ptr<eonc::HostSurfacePotential> step_pot;
218 std::unique_ptr<NudgedElasticBand> step_neb;
219 std::shared_ptr<NEBObjectiveFunction> step_objf;
220 std::unique_ptr<Optimizer> step_opt;
221 eon_relax_surface_fn step_surface{nullptr};
222 void *step_user{nullptr};
223 double step_baseline{0.0};
224 long step_iteration{0};
225};
226
227static int stamp_rc(EonRelaxEngine *eng, int rc) {
228 if (eng) {
229 eng->last_rc = rc;
230 }
231 return rc;
232}
233
234static int fail_out(EonRelaxEngine *eng, eon_relax_outcome_t *out, int rc,
235 eon_relax_kind_t kind, uint64_t epoch) {
236 if (out) {
237 stamp_outcome(out, kind, epoch);
238 out->status = -1;
239 }
240 return stamp_rc(eng, rc);
241}

243static void clear_stepper(EonRelaxEngine *eng) {
244 eng->step_opt.reset();
245 eng->step_objf.reset();
246 eng->step_neb.reset();
247 eng->step_pot.reset();
248 eng->step_surface = nullptr;
249 eng->step_user = nullptr;
250 eng->step_baseline = 0.0;
251 eng->step_iteration = 0;
252}

282EonRelaxEngine *eon_relax_create(const void *config, size_t config_len,
283 char *errbuf, size_t errlen) {
284 if (config == nullptr && config_len == 0) {
285 auto *eng = new (std::nothrow) EonRelaxEngine();
286 if (!eng) {
287 set_err(errbuf, errlen, "-19 out of memory");
288 return nullptr;
289 }
290 eng->kind = EON_RELAX_KIND_NEB;
291 return eng;
292 }
293 if (config == nullptr || config_len == 0) {
294 set_err(errbuf, errlen, "-13 RelaxEngineParams capnp root");
295 return nullptr;
296 }
297 auto *eng = new (std::nothrow) EonRelaxEngine();
298 if (!eng) {
299 set_err(errbuf, errlen, "-19 out of memory");
300 return nullptr;
301 }
302 const int prc = apply_relax_params(eng->params, &eng->kind, &eng->epoch,
303 config, config_len, errbuf, errlen);
304 if (prc != EON_RELAX_OK) {
305 delete eng;
306 return nullptr;
307 }
308 return eng;
309}

333int eon_relax_run(EonRelaxEngine *eng, eon_relax_band_t *band,
334 eon_relax_surface_fn surface, void *user,
335 eon_relax_outcome_t *out) {
336 if (!eng) {
337 return EON_RELAX_NULL_ENGINE;
338 }
339 if (eng->unusable) {
340 return fail_out(eng, out, EON_RELAX_SURFACE_FATAL, eng->kind, eng->epoch);
341 }
342 if (!band) {
343 return fail_out(eng, out, EON_RELAX_NULL_BAND, eng->kind, eng->epoch);
344 }
345 if (!surface) {
346 return fail_out(eng, out, EON_RELAX_NULL_SURFACE, eng->kind, eng->epoch);
347 }
348 if (!out) {
349 return stamp_rc(eng, EON_RELAX_NULL_OUTCOME);
350 }
351 clear_stepper(eng);
352 if (band->n_atoms <= 0) {
353 return fail_out(eng, out, EON_RELAX_NATOMS, eng->kind, eng->epoch);
354 }
355 if (!band->positions) {
356 return fail_out(eng, out, EON_RELAX_NULL_POSITIONS, eng->kind, eng->epoch);
357 }
358 if (!band->atomic_nrs) {
359 return fail_out(eng, out, EON_RELAX_NULL_ATOMIC_NRS, eng->kind, eng->epoch);
360 }
361 if (!band->boxes) {
362 return fail_out(eng, out, EON_RELAX_NULL_BOXES, eng->kind, eng->epoch);
363 }
364 if (band->version.major != EON_RELAX_ABI_MAJOR) {
365 return fail_out(eng, out, EON_RELAX_ABI_MISMATCH, eng->kind, eng->epoch);
366 }
367 if (!known_kind(eng->kind)) {
368 return fail_out(eng, out, EON_RELAX_UNKNOWN_KIND, eng->kind, eng->epoch);
369 }
370 const int mass_rc = masses_ok(band);
371 if (mass_rc != EON_RELAX_OK) {
372 return fail_out(eng, out, mass_rc, eng->kind, eng->epoch);
373 }
374 stamp_outcome(out, eng->kind, eng->epoch);
375
376 auto pot =
377 std::make_shared<eonc::HostSurfacePotential>(surface, user, eng->epoch);
378 try {
379 if (eng->kind == EON_RELAX_KIND_NEB) {
380 if (band->n_images < 3) {
381 out->status = -1;
382 return stamp_rc(eng, EON_RELAX_BAND_TOO_SHORT);
383 }
384 const int64_t want =
385 static_cast<int64_t>(eng->params.neb_options().image_count) + 2;
386 if (band->n_images != want) {
387 out->status = -1;
388 return stamp_rc(eng, EON_RELAX_BAND_SIZE);
389 }
390 std::vector<Matter> path;
391 path.reserve(static_cast<size_t>(band->n_images));
392 for (int64_t i = 0; i < band->n_images; ++i) {
393 Matter img(pot, eng->params);
394 fill_matter(img, band, static_cast<long>(i), eng->epoch);
395 path.push_back(std::move(img));
396 }
397 auto neb = std::make_unique<NudgedElasticBand>(std::move(path),
398 eng->params, pot);
399 pot->bindPath(neb->path);
400 dirty_endpoints(*neb);
401 const auto st = neb->compute();
402 if (st == NudgedElasticBand::NEBStatus::INIT ||
403 st == NudgedElasticBand::NEBStatus::RUNNING) {
404 out->status = neb_status(st);
405 return stamp_rc(eng, EON_RELAX_UNKNOWN_STATUS);
406 }
407 out->status = neb_status(st);
408 NEBObjectiveFunction ranked(neb.get(), eng->params);
409 if (ranked.isUncertain()) {
410 out->status = EON_RELAX_NEB_MAX_UNCERTAINTY;
411 }
412 out->iterations = neb->lastIteration();
413 out->climbing_image = neb->climbingImage;
414 out->max_force = neb->convergenceForce();
415 out->surface_epoch = pot->surfaceEpoch();
416 for (int64_t i = 1; i + 1 < band->n_images; ++i) {
417 write_image(band->positions + i * 3 * band->n_atoms,
418 *neb->path[static_cast<size_t>(i)],
419 static_cast<long>(band->n_atoms));
420 }
421 return stamp_rc(eng, EON_RELAX_OK);
422 }
423
424 if (band->n_images != 1) {
425 out->status = -1;
426 return stamp_rc(eng, EON_RELAX_SADDLE_NIMAGES);
427 }
428 if (!band->mode) {
429 out->status = -1;
430 return stamp_rc(eng, EON_RELAX_NULL_MODE);
431 }
432 auto matter = std::make_shared<Matter>(pot, eng->params);
433 fill_matter(*matter, band, 0, eng->epoch);
434 AtomMatrix mode(band->n_atoms, 3);
435 Eigen::Map<const AtomMatrix> mmap(band->mode, band->n_atoms, 3);
436 mode = mmap;
437 const double reactant_e = matter->getPotentialEnergy();
438 MinModeSaddleSearch search(matter, mode, reactant_e, eng->params, pot);
439 const int sst = search.run();
440 write_image(band->positions, *matter, static_cast<long>(band->n_atoms));
441 out->iterations = search.getIterationCount();
442 out->surface_epoch = pot->surfaceEpoch();
443 out->status = sst;
444 return stamp_rc(eng, EON_RELAX_OK);
445 } catch (const eonc::SurfaceRecoverable &rec) {
446 out->status = -1;
447 return stamp_rc(eng, rec.rc);
448 } catch (const std::exception &) {
449 out->status = -1;
450 eng->unusable = true;
451 return stamp_rc(eng, EON_RELAX_SURFACE_FATAL);
452 }
453}

455int eon_relax_step(EonRelaxEngine *eng, eon_relax_band_t *band,
456 eon_relax_surface_fn surface, void *user,
457 eon_relax_outcome_t *out) {
458 if (!eng) {
459 return EON_RELAX_NULL_ENGINE;
460 }
461 if (eng->unusable) {
462 return fail_out(eng, out, EON_RELAX_SURFACE_FATAL, eng->kind, eng->epoch);
463 }
464 if (!band) {
465 return fail_out(eng, out, EON_RELAX_NULL_BAND, eng->kind, eng->epoch);
466 }
467 if (!surface) {
468 return fail_out(eng, out, EON_RELAX_NULL_SURFACE, eng->kind, eng->epoch);
469 }
470 if (!out) {
471 return stamp_rc(eng, EON_RELAX_NULL_OUTCOME);
472 }
473 if (eng->kind == EON_RELAX_KIND_SADDLE) {
474 return fail_out(eng, out, EON_RELAX_INVALID_PARAMETER, eng->kind,
475 eng->epoch);
476 }
477 if (eng->kind != EON_RELAX_KIND_NEB) {
478 return fail_out(eng, out, EON_RELAX_UNKNOWN_KIND, eng->kind, eng->epoch);
479 }
480 if (band->n_atoms <= 0) {
481 return fail_out(eng, out, EON_RELAX_NATOMS, EON_RELAX_KIND_NEB, eng->epoch);
482 }
483 if (!band->positions) {
484 return fail_out(eng, out, EON_RELAX_NULL_POSITIONS, EON_RELAX_KIND_NEB,
485 eng->epoch);
486 }
487 if (!band->atomic_nrs) {
488 return fail_out(eng, out, EON_RELAX_NULL_ATOMIC_NRS, EON_RELAX_KIND_NEB,
489 eng->epoch);
490 }
491 if (!band->boxes) {
492 return fail_out(eng, out, EON_RELAX_NULL_BOXES, EON_RELAX_KIND_NEB,
493 eng->epoch);
494 }
495 if (band->version.major != EON_RELAX_ABI_MAJOR) {
496 return fail_out(eng, out, EON_RELAX_ABI_MISMATCH, EON_RELAX_KIND_NEB,
497 eng->epoch);
498 }
499 const int mass_rc = masses_ok(band);
500 if (mass_rc != EON_RELAX_OK) {
501 return fail_out(eng, out, mass_rc, EON_RELAX_KIND_NEB, eng->epoch);
502 }
503 stamp_outcome(out, EON_RELAX_KIND_NEB, eng->epoch);
504 if (eng->step_neb && (band->n_atoms != eng->step_neb->atoms ||
505 band->n_images != eng->step_neb->numImages + 2)) {
506 out->status = -1;
507 return stamp_rc(eng, EON_RELAX_INVALID_PARAMETER);
508 }
509 if (band->n_images < 3) {
510 out->status = -1;
511 return stamp_rc(eng, EON_RELAX_BAND_TOO_SHORT);
512 }
513 const int64_t want =
514 static_cast<int64_t>(eng->params.neb_options().image_count) + 2;
515 if (band->n_images != want) {
516 out->status = -1;
517 return stamp_rc(eng, EON_RELAX_BAND_SIZE);
518 }
519 if (eng->step_neb &&
520 (eng->step_surface != surface || eng->step_user != user)) {
521 out->status = -1;
522 return stamp_rc(eng, EON_RELAX_INVALID_PARAMETER);
523 }
524
525 try {
526 if (!eng->step_neb) {
527 eng->step_pot = std::make_shared<eonc::HostSurfacePotential>(
528 surface, user, eng->epoch);
529 std::vector<Matter> path;
530 path.reserve(static_cast<size_t>(band->n_images));
531 for (int64_t i = 0; i < band->n_images; ++i) {
532 Matter img(eng->step_pot, eng->params);
533 fill_matter(img, band, static_cast<long>(i), eng->epoch);
534 path.push_back(std::move(img));
535 }
536 eng->step_neb = std::make_unique<NudgedElasticBand>(
537 std::move(path), eng->params, eng->step_pot);
538 eng->step_pot->bindPath(eng->step_neb->path);
539 dirty_endpoints(*eng->step_neb);
540 eng->step_neb->E_ref =
541 std::min(eng->step_neb->path[0]->getPotentialEnergy(),
542 eng->step_neb->path[eng->step_neb->numImages + 1]
543 ->getPotentialEnergy());
544 eng->step_neb->updateForces();
545 eng->step_objf = std::make_shared<NEBObjectiveFunction>(
546 eng->step_neb.get(), eng->params);
547 eng->step_opt = eonc::helpers::create::mkOptim(
548 eng->step_objf, eng->params.neb_options().opt_method, eng->params);
549 eng->step_surface = surface;
550 eng->step_user = user;
551 eng->step_baseline = eng->step_neb->convergenceForce();
552 eng->step_iteration = 0;
553 } else {
554 // The host may have moved images between steps (acquisition,
555 // reparameterization); resync and mark the band dirty.
556 for (int64_t i = 0; i < band->n_images; ++i) {
557 fill_matter(*eng->step_neb->path[static_cast<size_t>(i)], band,
558 static_cast<long>(i), eng->epoch);
559 }
560 eng->step_neb->movedAfterForceCall = true;
561 eng->step_neb->updateForces();
562 }
563
564 NudgedElasticBand &neb = *eng->step_neb;
565 const double convForce = neb.convergenceForce();
566 const auto &ci_opt = eng->params.neb_options().climbing_image;
567 const bool ci_active =
568 ci_opt.enabled &&
569 (convForce < eng->step_baseline * ci_opt.trigger_factor ||
570 convForce < ci_opt.trigger_force);
571 neb.setCIEnabled(ci_active);
572
573 const double force_tol = eng->params.neb_options().force_tolerance;
574 const long max_iter = eng->params.neb_options().max_iterations;
575 int status = EON_RELAX_NEB_RUNNING;
576 if (eng->step_objf && eng->step_objf->isUncertain()) {
577 status = EON_RELAX_NEB_MAX_UNCERTAINTY;
578 } else if (convForce <= force_tol) {
579 status = EON_RELAX_NEB_GOOD;
580 } else if (eng->step_iteration >= max_iter) {
581 status = EON_RELAX_NEB_BAD_MAX_ITERATIONS;
582 } else {
583 eng->step_opt->step(eng->params.optimizer_options().max_move);
584 ++eng->step_iteration;
585 neb.updateForces();
586 if (eng->step_objf && eng->step_objf->isUncertain()) {
587 status = EON_RELAX_NEB_MAX_UNCERTAINTY;
588 } else if (neb.convergenceForce() <= force_tol) {
589 status = EON_RELAX_NEB_GOOD;
590 } else if (eng->step_iteration >= max_iter) {
591 status = EON_RELAX_NEB_BAD_MAX_ITERATIONS;
592 }
593 }
594
595 // Interior images only: eOn never moves the endpoints, and Matter
596 // wraps positions into the cell, so writing them back would hand
597 // the caller a different frame than the one it supplied.
598 for (int64_t i = 1; i + 1 < band->n_images; ++i) {
599 write_image(band->positions + i * 3 * band->n_atoms,
600 *neb.path[static_cast<size_t>(i)],
601 static_cast<long>(band->n_atoms));
602 }
603 out->status = status;
604 out->iterations = eng->step_iteration;
605 out->climbing_image = neb.climbingImage;
606 out->max_force = neb.convergenceForce();
607 out->surface_epoch = eng->step_pot->surfaceEpoch();
608 return stamp_rc(eng, EON_RELAX_OK);
609 } catch (const eonc::SurfaceRecoverable &rec) {
610 out->status = -1;
611 clear_stepper(eng);
612 return stamp_rc(eng, rec.rc);
613 } catch (const std::exception &) {
614 out->status = -1;
615 eng->unusable = true;
616 return stamp_rc(eng, EON_RELAX_SURFACE_FATAL);
617 }
618}

620int eon_relax_reset(EonRelaxEngine *eng) {
621 if (!eng) {
622 return EON_RELAX_NULL_ENGINE;
623 }
624 if (eng->unusable) {
625 return stamp_rc(eng, EON_RELAX_SURFACE_FATAL);
626 }
627 clear_stepper(eng);
628 eng->last_rc = EON_RELAX_OK;
629 return EON_RELAX_OK;
630}
