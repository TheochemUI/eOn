96 void force(std::span<const double> positions, std::span<const int> atomicNrs,
97 std::span<double> forces, double *energy, double *variance,
98 std::span<const double> box) {
99 if (positions.size() % 3 != 0 || positions.size() / 3 != atomicNrs.size() ||
100 forces.size() != positions.size() || box.size() != 9) {
101 throw std::invalid_argument("Potential::force span size mismatch");
102 }
103 force(static_cast<long>(atomicNrs.size()), positions.data(),
104 atomicNrs.data(), forces.data(), energy, variance, box.data());
105 }

234 virtual void forceBatch(long nSystems, long nAtoms,
235 const double *const *positions,
236 const int *const *atomicNrs, double *const *forces,
237 double *energies, double *variances,
238 const double *const *boxes) {
239 for (long i = 0; i < nSystems; i++) {
240 double var = 0;
241 force(nAtoms, positions[i], atomicNrs[i], forces[i], &energies[i], &var,
242 boxes[i]);
243 if (variances)
244 variances[i] = var;
245 recordForceCall();
246 }
247 }
