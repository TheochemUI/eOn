198
199 virtual void forceBatch(long nSystems, long nAtoms,
200 const double *const *positions,
201 const int *const *atomicNrs, double *const *forces,
202 double *energies, double *variances,
203 const double *const *boxes) {
204 for (long i = 0; i < nSystems; i++) {
205 double var = 0;
206 force(nAtoms, positions[i], atomicNrs[i], forces[i], &energies[i], &var,
207 boxes[i]);
208 if (variances)
209 variances[i] = var;
210 forceCallCounter++;
211 registry_.on_force_call(ptype);
212 }
213 }
