152 std::string processSearchString() const {
153 std::ostringstream out;
154 out << status_code << " termination_reason\n";
155 if (!status_text.empty()) {
156 out << status_text << " termination_reason_text\n";
157 }
158 out << random_seed << " random_seed\n";
159 if (!potential_type.empty()) {
160 out << potential_type << " potential_type\n";
161 }
162 out << force_calls << " total_force_calls\n";
163 out << force_calls_minimization << " force_calls_minimization\n";
164 out << force_calls_saddle << " force_calls_saddle\n";
165 out << std::format("{:.12e} potential_energy_saddle\n",
166 potential_energy_saddle);
167 out << std::format("{:.12e} potential_energy_reactant\n",
168 potential_energy_reactant);
169 out << std::format("{:.12e} potential_energy_product\n",
170 potential_energy_product);
171 out << std::format("{:.12e} barrier_reactant_to_product\n",
172 barrier_reactant_to_product);
173 out << std::format("{:.12e} barrier_product_to_reactant\n",
174 barrier_product_to_reactant);
175 out << std::format("{:.12e} displacement_saddle_distance\n",
176 displacement_saddle_distance);
177 if (has_dynamics) {
178 out << std::format("{:.12e} simulation_time\n", simulation_time);
179 out << std::format("{:.12e} md_temperature\n", md_temperature);
180 }
181 out << force_calls_prefactors << " force_calls_prefactors\n";
182 out << std::format("{:.12e} prefactor_reactant_to_product\n",
183 prefactor_reactant_to_product);
184 out << std::format("{:.12e} prefactor_product_to_reactant\n",
185 prefactor_product_to_reactant);
186 return out.str();
187 }

189 std::string toString() const {
190 if (process_search_layout) {
191 return processSearchString();
192 }
193 std::ostringstream out;
194 out << status_code << " termination_reason\n";
195 if (!status_text.empty()) {
196 out << status_text << " termination_reason_text\n";
197 }
198 if (!job_type.empty()) {
199 out << job_type << " job_type\n";
200 }
201 if (!potential_type.empty()) {
202 out << potential_type << " potential_type\n";
203 }
204 if (random_seed >= 0) {
205 out << random_seed << " random_seed\n";
206 }
207 out << force_calls << " total_force_calls\n";
208 if (has_energy) {
209 out << std::format("{:.12e} potential_energy\n", potential_energy);
210 }
211 if (has_saddle) {
212 out << std::format("{:.12e} potential_energy_saddle\n",
213 potential_energy_saddle);
214 }
215 if (has_reactant) {
216 out << std::format("{:.12e} potential_energy_reactant\n",
217 potential_energy_reactant);
218 }
219 if (has_product) {
220 out << std::format("{:.12e} potential_energy_product\n",
221 potential_energy_product);
222 }
223 for (const auto &kv : extras) {
224 out << std::format("{:.12e} {}\n", kv.second, kv.first);
225 }
226 for (const auto &kv : tags) {
227 out << kv.second << " " << kv.first << "\n";
228 }
229 out << provenance.text();
230 return out.str();
231 }

244 static JobResultEnvelope fromMinimization(RunStatus status, PotType pot,
245 std::uint64_t fcalls, bool hasE,
246 double energy) {
247 JobResultEnvelope e;
248 e.job_type = "minimization";
249 e.status_code = static_cast<int>(status);
250 e.status_text = std::string(magic_enum::enum_name<RunStatus>(status));
251 e.potential_type = std::string(magic_enum::enum_name<PotType>(pot));
252 e.force_calls = fcalls;
253 e.has_energy = hasE;
254 e.potential_energy = energy;
255 return e;
256 }

258 static JobResultEnvelope fromProcessSearch(
259 int status, std::string statusText, PotType pot, std::int64_t seed,
260 std::uint64_t fMin, std::uint64_t fSaddle, std::uint64_t fPref,
261 double eSaddle, double eReactant, double eProduct, double barrierFwd,
262 double barrierRev, double displacement, double prefFwd, double prefRev,
263 bool dynamics, double simTime, double mdTemp) {
264 JobResultEnvelope e;
265 e.process_search_layout = true;
266 e.job_type = "process_search";
267 e.status_code = status;
268 e.status_text = std::move(statusText);
269 e.potential_type = std::string(magic_enum::enum_name<PotType>(pot));
270 e.random_seed = seed;
271 e.force_calls_minimization = fMin;
272 e.force_calls_saddle = fSaddle;
273 e.force_calls_prefactors = fPref;
274 e.force_calls = fMin + fSaddle + fPref;
275 e.has_saddle = true;
276 e.has_reactant = true;
277 e.has_product = true;
278 e.potential_energy_saddle = eSaddle;
279 e.potential_energy_reactant = eReactant;
280 e.potential_energy_product = eProduct;
281 e.barrier_reactant_to_product = barrierFwd;
282 e.barrier_product_to_reactant = barrierRev;
283 e.displacement_saddle_distance = displacement;
284 e.prefactor_reactant_to_product = prefFwd;
285 e.prefactor_product_to_reactant = prefRev;
286 e.has_dynamics = dynamics;
287 e.simulation_time = simTime;
288 e.md_temperature = mdTemp;
289 return e;
290 }
