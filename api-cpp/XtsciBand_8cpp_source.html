78// Fixed atoms keep the coordinates they had before the step.
79void pinRows(const AtomMatrix &ref, const Matter &image, AtomMatrix *pos) {
80 for (int j = 0; j < image.numberOfAtoms(); ++j) {
81 if (image.getFixed(j)) {
82 pos->row(j) = ref.row(j);
83 }
84 }
85}

87// Moves an image only when the coordinates differ, so an image that
88// already holds this geometry keeps its evaluation.
89void moveTo(Matter &image, const AtomMatrix &pos) {
90 if (image.getPositions() != pos) {
91 image.setPositions(pos);
92 }
93}

97int xtsciBandSurface(void *user, void *request) {
98 auto *self = static_cast<XtsciBand *>(user);
99 auto *req = static_cast<rgsaddle_surface_request_t *>(request);
100 if (self == nullptr || req == nullptr || req->positions == nullptr ||
101 req->energies == nullptr || req->gradients == nullptr) {
102 return RGSADDLE_SURFACE_FAILED;
103 }
104 try {
105 auto *neb = self->m_neb;
106 const auto band = static_cast<int64_t>(neb->numImages + 2);
107 const auto nAtoms = static_cast<int64_t>(neb->atoms);
108 if (req->version.major != RGSADDLE_ABI_MAJOR) {
109 return RGSADDLE_ABI_MISMATCH;
110 }
111 if (req->n_atoms != nAtoms) {
112 return RGSADDLE_SHAPE;
113 }
114 // Rows carried by this request and the band image of row 0. The
115 // session sends the whole band on its first evaluation, the interior
116 // images 1 .. band - 2 afterwards, or one image with
117 // RGSADDLE_REQ_ONE_IMAGE.
118 int64_t rows = 0;
119 int64_t first = 0;
120 if ((req->flags & RGSADDLE_REQ_ONE_IMAGE) != 0) {
121 if (req->n_images != band || req->image < 0 || req->image >= band) {
122 return RGSADDLE_SHAPE;
123 }
124 rows = 1;
125 first = req->image;
126 } else if (req->n_images == band) {
127 rows = band;
128 } else if (req->n_images == band - 2) {
129 rows = band - 2;
130 first = 1;
131 } else {
132 return RGSADDLE_SHAPE;
133 }
134 const auto dof = static_cast<Eigen::Index>(3 * nAtoms);
135 std::vector<Matter *> carried;
136 carried.reserve(static_cast<size_t>(rows));
137 for (int64_t r = 0; r < rows; ++r) {
138 const auto i = static_cast<size_t>(first + r);
139 AtomMatrix pos = AtomMatrix::Map(req->positions + r * dof, nAtoms, 3);
140 if (i < self->m_fixed.size()) {
141 pinRows(self->m_fixed[i], *neb->path[i], &pos);
142 }
143 // The fixed endpoints and a start point the band already evaluated
144 // come back from the image's own cache.
145 moveTo(*neb->path[i], pos);
146 carried.push_back(neb->path[i].get());
147 }
148 // One batch for every carried image, so calculator groups or a batched
149 // model take them together; a serial potential evaluates them in turn.
150 eonc::evaluateTogether(*carried.front()->getPotential(), carried);
151 for (int64_t r = 0; r < rows; ++r) {
152 Matter &image = *carried[static_cast<size_t>(r)];
153 req->energies[r] = image.getPotentialEnergy();
154 const AtomMatrix &force = image.getForces();
155 double *grad = req->gradients + r * dof;
156 const double *src = force.data();
157 for (Eigen::Index k = 0; k < dof; ++k) {
158 grad[k] = -src[k];
159 }
160 }
161 return RGSADDLE_OK;
162 } catch (...) {
163 return RGSADDLE_SURFACE_FAILED;
164 }
165}

167XtsciBand::XtsciBand(NudgedElasticBand &neb, const Parameters &params)
168 : m_neb{&neb},
169 m_maxMove{params.optimizer_options().max_move} {
170 rgsaddle_version_t stamp{};
171 checkStatus(rgsaddle_abi_stamp(&stamp), "rgsaddle_abi_stamp");
172 if (stamp.major != RGSADDLE_ABI_MAJOR) {
173 throw std::runtime_error("incompatible rgsaddle ABI");
174 }
175 const auto &nebOpt = params.neb_options();
176 const auto nImages = static_cast<int64_t>(neb.numImages + 2);
177 const auto nAtoms = static_cast<int64_t>(neb.atoms);
178 if (nImages < 3 || nAtoms < 1) {
179 throw std::runtime_error("rgsaddle band needs at least 3 images");
180 }
181 if (springKind(nebOpt) == RGSADDLE_SPRING_WEIGHTED) {
182 m_springKs.assign(static_cast<size_t>(nImages - 1), nebOpt.spring.constant);
183 }
184 rgsaddle_band_config_t config{};
185 config.version = RGSADDLE_VERSION_INIT;
186 config.flags = 0;
187 config.tangent = tangentKind(nebOpt);
188 config.spring = springKind(nebOpt);
189 config.projection = projectionKind(nebOpt);
190 config.method = methodKind(params.optimizer_options());
191 config.spring_k = nebOpt.spring.om.enabled
192 ? nebOpt.spring.constant * nebOpt.spring.om.k_scale
193 : nebOpt.spring.constant;
194 config.spring_ks = m_springKs.empty() ? nullptr : m_springKs.data();
195 if (nebOpt.climbing_image.enabled) {
196 config.ci_trigger_factor = nebOpt.climbing_image.trigger_factor > 0.0
197 ? nebOpt.climbing_image.trigger_factor
198 : 1.0e-300;
199 config.ci_trigger_force = nebOpt.climbing_image.trigger_force;
200 } else {
201 config.ci_trigger_factor = 0.0;
202 config.ci_trigger_force = 0.0;
203 }
204 if (!neb.path.empty() && neb.path[0]->getPeriodic()) {
205 const Matrix3d cell = neb.path[0]->getCell();
206 m_cell.assign(cell.data(), cell.data() + 9);
207 config.cell = m_cell.data();
208 }
209 config.force_tol = nebOpt.force_tolerance;
210 config.max_move = m_maxMove > 0.0 ? m_maxMove : 0.2;
211 config.memory = std::max<long>(1, params.optimizer_options().lbfgs.memory);
212
213 std::vector<double> positions(static_cast<size_t>(nImages * 3 * nAtoms), 0.0);
214 for (int64_t i = 0; i < nImages; ++i) {
215 const AtomMatrix &pos = neb.path[i]->getPositions();
216 std::copy(pos.data(), pos.data() + pos.size(),
217 positions.begin() + static_cast<std::ptrdiff_t>(i * pos.size()));
218 }
219 m_band = rgsaddle_band_create(&config, nImages, nAtoms, positions.data());
220 if (m_band == nullptr) {
221 throw std::runtime_error("rgsaddle_band_create failed");
222 }
223}

250void XtsciBand::step(double maxMove) {
251 if (maxMove > 0.0) {
252 m_maxMove = maxMove;
253 }
254 const auto nImages = static_cast<int64_t>(m_neb->numImages + 2);
255 m_fixed.resize(static_cast<size_t>(nImages));
256 for (int64_t i = 0; i < nImages; ++i) {
257 m_fixed[static_cast<size_t>(i)] = m_neb->path[i]->getPositions();
258 }
259 // The session holds the band of the last step. Resending it drops the
260 // session's cached endpoint and start-point evaluations, so only a band
261 // moved outside the session (a reparameterization) is resent.
262#if RGSADDLE_ABI_MINOR >= 5
263 // Unchanged rows cost nothing: the session keeps its cached values.
264 syncFromPath();
265#else
266 if (!sessionMatchesPath()) {
267 syncFromPath();
268 }
269#endif
270 rgsaddle_report_t report{};
271 checkStatus(rgsaddle_band_step(m_band, surfaceCallback, this, &report),
272 "rgsaddle_band_step");
273 std::vector<double> positions(m_fixed.size() * m_fixed[0].size(), 0.0);
274 checkStatus(rgsaddle_band_positions(m_band, positions.data()),
275 "rgsaddle_band_positions");
276 const auto dof = m_fixed[0].size();
277 for (size_t i = 0; i < m_fixed.size(); ++i) {
278 AtomMatrix pos =
279 AtomMatrix::Map(positions.data() + static_cast<std::ptrdiff_t>(i * dof),
280 m_neb->atoms, 3);
281 // Endpoints are fixed by the session. Interior fixed atoms stay put.
282 if (i == 0 || i + 1 == m_fixed.size()) {
283 continue;
284 }
285 pinRows(m_fixed[i], *m_neb->path[i], &pos);
286 // The accepted point is usually the band's last evaluation, which the
287 // image still holds; the band update after the step then costs nothing.
288 moveTo(*m_neb->path[i], pos);
289 }
290#if RGSADDLE_ABI_MINOR >= 5
291 // An image whose last evaluation was elsewhere (the solver backed off to
292 // a point it evaluated earlier) takes the session's evaluation of the
293 // accepted band instead of a fresh force call.
294 std::vector<double> energies(m_fixed.size());
295 std::vector<double> gradients(m_fixed.size() * dof);
296 if (rgsaddle_band_evaluation(m_band, energies.data(), gradients.data(),
297 nullptr) == RGSADDLE_OK) {
298 for (size_t i = 1; i + 1 < m_fixed.size(); ++i) {
299 Matter &image = *m_neb->path[i];
300 if (!image.needsForceUpdate()) {
301 continue;
302 }
303 const AtomMatrix grad = AtomMatrix::Map(
304 gradients.data() + static_cast<std::ptrdiff_t>(i * dof), m_neb->atoms,
305 3);
306 image.setEvaluation(-grad, energies[i]);
307 }
308 }
309#endif
310 m_neb->movedAfterForceCall = true;
311}

313bool XtsciBand::sessionMatchesPath() const {
314 const auto nImages = static_cast<int64_t>(m_neb->numImages + 2);
315 const auto nAtoms = static_cast<int64_t>(m_neb->atoms);
316 std::vector<double> positions(static_cast<size_t>(nImages * 3 * nAtoms));
317 if (rgsaddle_band_positions(m_band, positions.data()) != RGSADDLE_OK) {
318 return false;
319 }
320 const auto dof = static_cast<std::ptrdiff_t>(3 * nAtoms);
321 for (int64_t i = 0; i < nImages; ++i) {
322 const AtomMatrix &pos = m_neb->path[static_cast<size_t>(i)]->getPositions();
323 if (!std::equal(pos.data(), pos.data() + dof,
324 positions.begin() + i * dof)) {
325 return false;
326 }
327 }
328 return true;
329}
