51 [[nodiscard]] std::vector<int>
52 assign(const std::vector<std::int64_t> &keys) {
53 const long m = static_cast<long>(keys.size());
54 const long cap = (m + groups_ - 1) / groups_;
55 std::vector<long> load(static_cast<size_t>(groups_), 0);
56 std::vector<int> route(static_cast<size_t>(m), -1);
57 for (long j = 0; j < m; ++j) {
58 const int home = homeOf(keys[static_cast<size_t>(j)], j);
59 if (load[static_cast<size_t>(home)] < cap) {
60 route[static_cast<size_t>(j)] = home;
61 ++load[static_cast<size_t>(home)];
62 }
63 }
64 for (long j = 0; j < m; ++j) {
65 if (route[static_cast<size_t>(j)] >= 0)
66 continue;
67 const auto least = std::min_element(load.begin(), load.end());
68 const int g = static_cast<int>(least - load.begin());
69 route[static_cast<size_t>(j)] = g;
70 ++*least;
71 }
72 for (long j = 0; j < m; ++j)
73 home_[keys[static_cast<size_t>(j)]] = route[static_cast<size_t>(j)];
74 return route;
75 }

86 [[nodiscard]] int nearest(const double *positions, long n3) const {
87 double best = std::numeric_limits<double>::infinity();
88 int group = 0;
89 for (const auto &[key, stored] : last_) {
90 if (static_cast<long>(stored.size()) != n3)
91 continue;
92 double d2 = 0.0;
93 for (long i = 0; i < n3; ++i) {
94 const double d = stored[static_cast<size_t>(i)] - positions[i];
95 d2 += d * d;
96 }
97 const auto home = home_.find(key);
98 if (d2 < best && home != home_.end()) {
99 best = d2;
100 group = home->second;
101 }
102 }
103 return group;
104 }

107 void settle(std::int64_t key, int group) { home_[key] = group; }
