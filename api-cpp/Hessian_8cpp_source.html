30// phva_atoms entries are *mobile / displaced* atoms for FD (hybrid/PHVA-class
31// active set). Intersect with non-fixed atoms in HessianJob.
32
33// Checkpoint: first line "eon_hess_ckpt <size> <next_col>", then size*size
34// doubles in row-major order matching MatrixXd storage.
35bool loadColumnCheckpoint(const std::string &path, int size, int &nextCol,
36 MatrixXd &H) {
37 std::ifstream in(path);
38 if (!in) {
39 return false;
40 }
41 std::string tag;
42 int fileSize = 0;
43 in >> tag >> fileSize >> nextCol;
44 if (!in || tag != "eon_hess_ckpt" || fileSize != size || nextCol < 0 ||
45 nextCol > size) {
46 return false;
47 }
48 H.resize(size, size);
49 for (int i = 0; i < size; ++i) {
50 for (int j = 0; j < size; ++j) {
51 double v = 0.0;
52 in >> v;
53 if (!in) {
54 return false;
55 }
56 H(i, j) = v;
57 }
58 }
59 return true;
60}
61
62bool saveColumnCheckpoint(const std::string &path, int size, int nextCol,
63 const MatrixXd &H) {
64 std::ofstream out(path);
65 if (!out) {
66 return false;
67 }
68 out << "eon_hess_ckpt " << size << " " << nextCol << "\n";
69 out.precision(17);
70 for (int i = 0; i < size; ++i) {
71 for (int j = 0; j < size; ++j) {
72 out << H(i, j) << (j + 1 == size ? '\n' : ' ');
73 }
74 }
75 return static_cast<bool>(out);
76}

124MobileColoring buildMobileColoring(const Matter &matter, const VectorXi &atoms,
125 double cutoff) {
126 MobileColoring out;
127 const int nAtoms = static_cast<int>(matter.numberOfAtoms());
128 const int nMobile = static_cast<int>(atoms.rows());
129 if (nAtoms <= 0 || nMobile <= 0 || !(cutoff > 0.0) ||
130 !std::isfinite(cutoff)) {
131 return out;
132 }
133 const Matrix3d cell = matter.getCell();
134 if (!(std::abs(cell.determinant()) > 1e-18)) {
135 return out;
136 }
137 for (int a = 0; a < nMobile; ++a) {
138 const long idx = atoms(a);
139 if (idx < 0 || idx >= nAtoms) {
140 return {};
141 }
142 }
143
144 const AtomMatrix &pos = matter.getPositions();
145 VesinNeighbors nl;
146 VesinNeighbors::Options opt;
147 opt.cutoff = cutoff;
148 opt.full = true;
149 opt.return_distances = false;
150 opt.return_vectors = false;
151 opt.periodic = {matter.getPeriodic(), matter.getPeriodic(),
152 matter.getPeriodic()};
153 try {
154 nl.compute(pos.data(), static_cast<std::size_t>(nAtoms), cell.data(), opt);
155 } catch (const std::exception &) {
156 return {};
157 }
158
159 std::vector<std::vector<int>> nbs(static_cast<std::size_t>(nAtoms));
160 for (std::size_t p = 0; p < nl.size(); ++p) {
161 const int i = static_cast<int>(nl.i(p));
162 const int j = static_cast<int>(nl.j(p));
163 if (i == j || i < 0 || j < 0 || i >= nAtoms || j >= nAtoms) {
164 continue;
165 }
166 nbs[static_cast<std::size_t>(i)].push_back(j);
167 }
168 for (auto &row : nbs) {
169 std::sort(row.begin(), row.end());
170 row.erase(std::unique(row.begin(), row.end()), row.end());
171 }
172
173 std::vector<int> local(static_cast<std::size_t>(nAtoms), -1);
174 for (int a = 0; a < nMobile; ++a) {
175 local[static_cast<std::size_t>(atoms(a))] = a;
176 }
177
178 out.closed.assign(static_cast<std::size_t>(nMobile), {});
179 for (int ia = 0; ia < nMobile; ++ia) {
180 auto &nbhd = out.closed[static_cast<std::size_t>(ia)];
181 nbhd.push_back(ia);
182 const int g = static_cast<int>(atoms(ia));
183 for (int nb : nbs[static_cast<std::size_t>(g)]) {
184 const int lb = local[static_cast<std::size_t>(nb)];
185 if (lb >= 0) {
186 nbhd.push_back(lb);
187 }
188 }
189 std::sort(nbhd.begin(), nbhd.end());
190 nbhd.erase(std::unique(nbhd.begin(), nbhd.end()), nbhd.end());
191 }
192
193 // Square of the cutoff graph on the mobile set: atoms whose closed
194 // neighborhoods intersect cannot move in the same finite-difference.
195 std::vector<std::vector<int>> touchers(static_cast<std::size_t>(nMobile));
196 for (int ia = 0; ia < nMobile; ++ia) {
197 for (int k : out.closed[static_cast<std::size_t>(ia)]) {
198 touchers[static_cast<std::size_t>(k)].push_back(ia);
199 }
200 }
201 std::vector<std::vector<int>> adj(static_cast<std::size_t>(nMobile));
202 for (int k = 0; k < nMobile; ++k) {
203 const auto &t = touchers[static_cast<std::size_t>(k)];
204 for (size_t a = 0; a < t.size(); ++a) {
205 for (size_t b = a + 1; b < t.size(); ++b) {
206 adj[static_cast<std::size_t>(t[a])].push_back(t[b]);
207 adj[static_cast<std::size_t>(t[b])].push_back(t[a]);
208 }
209 }
210 }
211 for (auto &row : adj) {
212 std::sort(row.begin(), row.end());
213 row.erase(std::unique(row.begin(), row.end()), row.end());
214 }
215 out.color = greedyColorCutoffGraph(adj);
216 return out;
217}

219void writeMassWeighted(MatrixXd &hessian, const Matter &matter,
220 const VectorXi &atoms, int col, int atomI,
221 const AtomMatrix &forceA, const AtomMatrix &forceB,
222 double denom, const std::vector<int> &owner, int ia) {
223 const int size = static_cast<int>(atoms.rows()) * 3;
224 const double massI = matter.getMass(atomI);
225 for (int j = 0; j < size; ++j) {
226 const long atomJ = atoms(j / 3);
227 double dF = 0.0;
228 if (owner[static_cast<std::size_t>(j / 3)] == ia) {
229 dF = forceA(atomJ, j % 3) - forceB(atomJ, j % 3);
230 }
231 hessian(col, j) = -dF / denom;
232 const double effMass = std::sqrt(matter.getMass(atomJ) * massI);
233 hessian(col, j) = eonc::safemath::safe_div(hessian(col, j), effMass, 0.0);
234 }
235}

239std::vector<int>
240greedyColorCutoffGraph(const std::vector<std::vector<int>> &adj) {
241 const int n = static_cast<int>(adj.size());
242 std::vector<int> color(static_cast<std::size_t>(n), -1);
243 std::vector<int> used(static_cast<std::size_t>(std::max(n, 0)), 0);
244 int epoch = 0;
245 for (int v = 0; v < n; ++v) {
246 ++epoch;
247 for (int u : adj[static_cast<std::size_t>(v)]) {
248 if (u < 0 || u >= n || u == v) {
249 continue;
250 }
251 const int cu = color[static_cast<std::size_t>(u)];
252 if (cu >= 0 && cu < n) {
253 used[static_cast<std::size_t>(cu)] = epoch;
254 }
255 }
256 int c = 0;
257 while (c < n && used[static_cast<std::size_t>(c)] == epoch) {
258 ++c;
259 }
260 color[static_cast<std::size_t>(v)] = c;
261 }
262 return color;
263}

270bool Hessian::calculate() {
271 int nAtoms = matter->numberOfAtoms();
272
273 int size = static_cast<int>(atoms.rows()) * 3;
274 QUILL_LOG_DEBUG(log, "[Hessian] Hessian size: {}\n", size);
275 if (size == 0) {
276 return false;
277 }
278
279 // Mobile-atom polarity: indices in `atoms` are FD-displaced DOF owners.
280 for (int a = 0; a < atoms.rows(); ++a) {
281 const long idx = atoms(a);
282 if (idx < 0 || idx >= nAtoms) {
283 QUILL_LOG_ERROR(log,
284 "[Hessian] atom index {} out of range [0, {}) at list "
285 "entry {}; aborting FD Hessian",
286 idx, nAtoms, a);
287 return false;
288 }
289 }
290
291 double dr = parameters.main_options().finiteDifference;
292 if (!(dr > 0.0) || !std::isfinite(dr)) {
293 QUILL_LOG_ERROR(log, "[Hessian] invalid finiteDifference dr={}\n", dr);
294 return false;
295 }
296
297 const FdScheme scheme = parseFdScheme(parameters.hessian_options().fd_scheme);
298 const std::string &ckptPath = parameters.hessian_options().checkpoint_path;
299
300 hessian.resize(size, size);
301 hessian.setZero();
302
303 // Net-force removal adds the same shift to every atom, so columns are
304 // no longer confined to the cutoff neighborhood. A column checkpoint is
305 // also stored one coordinate at a time.
306 bool anyFixed = false;
307 for (int i = 0; i < nAtoms; ++i) {
308 if (matter->getFixed(i)) {
309 anyFixed = true;
310 break;
311 }
312 }
313 const bool netCoupled =
314 parameters.main_options().removeNetForce && nAtoms > 1 && !anyFixed;
315 double cutoff = 0.0;
316 if (matter->getPotential()) {
317 cutoff = matter->getPotential()->finiteCutoff();
318 }
319 if (!netCoupled && ckptPath.empty() && cutoff > 0.0 &&
320 std::isfinite(cutoff)) {
321 if (calculateColored(cutoff, dr, scheme)) {
322 return true;
323 }
324 hessian.setZero();
325 }
326 return calculateSerial(dr, scheme);
327}

329bool Hessian::calculateColored(double cutoff, double dr, FdScheme scheme) {
330 const int nAtoms = static_cast<int>(matter->numberOfAtoms());
331 const int nMobile = static_cast<int>(atoms.rows());
332 const int size = nMobile * 3;
333 const MobileColoring coloring = buildMobileColoring(*matter, atoms, cutoff);
334 if (static_cast<int>(coloring.color.size()) != nMobile ||
335 static_cast<int>(coloring.closed.size()) != nMobile) {
336 return false;
337 }
338
339 int nColors = 0;
340 for (int c : coloring.color) {
341 if (c < 0) {
342 return false;
343 }
344 nColors = std::max(nColors, c + 1);
345 }
346 if (nColors <= 0) {
347 return false;
348 }
349 QUILL_LOG_DEBUG(log,
350 "[Hessian] cutoff coloring: {} colors for {} mobile atoms\n",
351 nColors, nMobile);
352
353 std::vector<std::vector<int>> members(static_cast<std::size_t>(nColors));
354 for (int ia = 0; ia < nMobile; ++ia) {
355 members[static_cast<std::size_t>(
356 coloring.color[static_cast<std::size_t>(ia)])]
357 .push_back(ia);
358 }
359 std::vector<std::vector<int>> owner(
360 static_cast<std::size_t>(nColors),
361 std::vector<int>(static_cast<std::size_t>(nMobile), -1));
362 for (int c = 0; c < nColors; ++c) {
363 for (int ia : members[static_cast<std::size_t>(c)]) {
364 for (int k : coloring.closed[static_cast<std::size_t>(ia)]) {
365 int &slot =
366 owner[static_cast<std::size_t>(c)][static_cast<std::size_t>(k)];
367 if (slot >= 0 && slot != ia) {
368 return false;
369 }
370 slot = ia;
371 }
372 }
373 }
374
375 Matter matterTemp(*matter);
376 const AtomMatrix pos = matter->getPositions();
377 AtomMatrix posDisplace(nAtoms, 3);
378 AtomMatrix force0 = matterTemp.getForces();
379 if (!force0.allFinite()) {
380 QUILL_LOG_ERROR(log, "[Hessian] non-finite forces at undisplaced geometry; "
381 "aborting FD Hessian");
382 return false;
383 }
384
385 for (int dir = 0; dir < 3; ++dir) {
386 for (int c = 0; c < nColors; ++c) {
387 const auto &group = members[static_cast<std::size_t>(c)];
388 if (group.empty()) {
389 continue;
390 }
391 auto shifted = [&](double scale, const char *side) -> AtomMatrix {
392 posDisplace.setZero();
393 for (int ia : group) {
394 posDisplace(atoms(ia), dir) = scale * dr;
395 }
396 matterTemp.setPositions(pos + posDisplace);
397 AtomMatrix force = matterTemp.getForces();
398 if (!force.allFinite()) {
399 QUILL_LOG_ERROR(log,
400 "[Hessian] non-finite forces for color {} dir {} "
401 "({}); aborting FD Hessian",
402 c, dir, side);
403 return AtomMatrix();
404 }
405 return force;
406 };
407 const AtomMatrix forcePlus = shifted(1.0, "+");
408 if (forcePlus.size() == 0) {
409 return false;
410 }
411 AtomMatrix forceMinus = force0;
412 AtomMatrix forcePlus2 = force0;
413 AtomMatrix forceMinus2 = force0;
414 if (scheme != FdScheme::OneSided) {
415 forceMinus = shifted(-1.0, "-");
416 if (forceMinus.size() == 0) {
417 return false;
418 }
419 }
420 if (scheme == FdScheme::Fourth) {
421 forcePlus2 = shifted(2.0, "+2");
422 if (forcePlus2.size() == 0) {
423 return false;
424 }
425 forceMinus2 = shifted(-2.0, "-2");
426 if (forceMinus2.size() == 0) {
427 return false;
428 }
429 }
430 const AtomMatrix slope = fdForceDerivative(
431 scheme, dr, force0, forcePlus, forceMinus, forcePlus2, forceMinus2);
432 const AtomMatrix zero = AtomMatrix::Zero(nAtoms, 3);
433 for (int ia : group) {
434 const int col = ia * 3 + dir;
435 writeMassWeighted(hessian, *matter, atoms, col,
436 static_cast<int>(atoms(ia)), slope, zero, 1.0,
437 owner[static_cast<std::size_t>(c)], ia);
438 }
439 }
440 }
441 return finalizeHessian(size);
442}

444bool Hessian::calculateSerial(double dr, FdScheme scheme) {
445 const int nAtoms = static_cast<int>(matter->numberOfAtoms());
446 const int size = static_cast<int>(atoms.rows()) * 3;
447 const std::string &ckptPath = parameters.hessian_options().checkpoint_path;
448 const bool wantResume =
449 parameters.hessian_options().resume && !ckptPath.empty();
450
451 AtomMatrix pos = matter->getPositions();
452 AtomMatrix posDisplace(nAtoms, 3);
453 AtomMatrix posTemp(nAtoms, 3);
454 AtomMatrix forcePlus(nAtoms, 3);
455 AtomMatrix forceMinus(nAtoms, 3);
456
457 int startCol = 0;
458 if (wantResume && loadColumnCheckpoint(ckptPath, size, startCol, hessian)) {
459 QUILL_LOG_DEBUG(log, "[Hessian] resume from column {} / {}\n", startCol,
460 size);
461 } else {
462 startCol = 0;
463 hessian.setZero();
464 }
465
466 Matter matterTemp(*matter);
467 AtomMatrix force0 = matterTemp.getForces();
468 if (!force0.allFinite()) {
469 QUILL_LOG_ERROR(log, "[Hessian] non-finite forces at undisplaced geometry; "
470 "aborting FD Hessian");
471 return false;
472 }
473
474 for (int i = startCol; i < size; i++) {
475 posDisplace.setZero();
476 posDisplace(atoms(i / 3), i % 3) = dr;
477
478 posTemp = pos + posDisplace;
479 matterTemp.setPositions(posTemp);
480 forcePlus = matterTemp.getForces();
481 if (!forcePlus.allFinite()) {
482 QUILL_LOG_ERROR(log,
483 "[Hessian] non-finite forces for FD column {} (+); "
484 "aborting FD Hessian",
485 i);
486 return false;
487 }
488
489 if (scheme != FdScheme::OneSided) {
490 posTemp = pos - posDisplace;
491 matterTemp.setPositions(posTemp);
492 forceMinus = matterTemp.getForces();
493 if (!forceMinus.allFinite()) {
494 QUILL_LOG_ERROR(log,
495 "[Hessian] non-finite forces for FD column {} (-); "
496 "aborting FD Hessian",
497 i);
498 return false;
499 }
500 }
501 AtomMatrix forcePlus2 = force0;
502 AtomMatrix forceMinus2 = force0;
503 if (scheme == FdScheme::Fourth) {
504 posDisplace(atoms(i / 3), i % 3) = 2.0 * dr;
505 matterTemp.setPositions(pos + posDisplace);
506 forcePlus2 = matterTemp.getForces();
507 if (!forcePlus2.allFinite()) {
508 QUILL_LOG_ERROR(log,
509 "[Hessian] non-finite forces for FD column {} (+2); "
510 "aborting FD Hessian",
511 i);
512 return false;
513 }
514 matterTemp.setPositions(pos - posDisplace);
515 forceMinus2 = matterTemp.getForces();
516 if (!forceMinus2.allFinite()) {
517 QUILL_LOG_ERROR(log,
518 "[Hessian] non-finite forces for FD column {} (-2); "
519 "aborting FD Hessian",
520 i);
521 return false;
522 }
523 }
524 // H ≈ -dF, mass-weighted. dF is the selected real stencil.
525 const AtomMatrix slope = fdForceDerivative(
526 scheme, dr, force0, forcePlus, forceMinus, forcePlus2, forceMinus2);
527 for (int j = 0; j < size; j++) {
528 hessian(i, j) = -slope(atoms(j / 3), j % 3);
529 const double effMass = std::sqrt(matter->getMass(atoms(j / 3)) *
530 matter->getMass(atoms(i / 3)));
531 hessian(i, j) = eonc::safemath::safe_div(hessian(i, j), effMass, 0.0);
532 }
533
534 if (!ckptPath.empty()) {
535 // next column to compute after a clean interrupt
536 saveColumnCheckpoint(ckptPath, size, i + 1, hessian);
537 }
538 }
539 return finalizeHessian(size);
540}

542bool Hessian::finalizeHessian(int size) {
543 const std::string &ckptPath = parameters.hessian_options().checkpoint_path;
544
545 // Symmetrize (FD noise breaks H=H^T; required for vib analysis)
546 for (int i = 0; i < size; i++) {
547 for (int j = 0; j < i; j++) {
548 hessian(i, j) = (hessian(i, j) + hessian(j, i)) / 2;
549 hessian(j, i) = hessian(i, j);
550 }
551 }
552
553 if (!hessian.allFinite()) {
554 QUILL_LOG_ERROR(log, "[Hessian] non-finite entries after FD assembly; "
555 "aborting eigen solve");
556 return false;
557 }
558
559 if (!parameters.main_options().quiet) {
560 QUILL_LOG_DEBUG(log, "[Hessian] writing hessian\n");
561 }
562 {
563 // A previous case in the same process can still hold hessian.dat on
564 // Windows. Replace it, and try once more after removing the old file.
565 std::ofstream hessfile("hessian.dat", std::ios::out | std::ios::trunc);
566 if (!hessfile) {
567 std::remove("hessian.dat");
568 hessfile.clear();
569 hessfile.open("hessian.dat", std::ios::out | std::ios::trunc);
570 }
571 if (!hessfile) {
572 QUILL_LOG_ERROR(log, "[Hessian] failed to open hessian.dat");
573 return false;
574 }
575 hessfile << hessian;
576 hessfile.close();
577 if (!hessfile) {
578 QUILL_LOG_ERROR(log, "[Hessian] failed to write hessian.dat");
579 return false;
580 }
581 }
582
583 // Completed run: remove checkpoint so a later job does not resume stale cols
584 if (!ckptPath.empty()) {
585 std::remove(ckptPath.c_str());
586 }
587
588 double t0, t1;
589 eonc::helpers::getTime(&t0, nullptr, nullptr);
590 QUILL_LOG_DEBUG(log, "[Hessian] calculating eigen values of the hessian\n");
591 // ColMajor copy for SelfAdjointEigenSolver (eOn MatrixXd is RowMajor)
592 using ColMajorXd =
593 Eigen::Matrix<double, Eigen::Dynamic, Eigen::Dynamic, Eigen::ColMajor>;
594 ColMajorXd hessianCol = hessian;
595 Eigen::SelfAdjointEigenSolver<ColMajorXd> es(hessianCol,
596 Eigen::EigenvaluesOnly);
597 eonc::helpers::getTime(&t1, nullptr, nullptr);
598 QUILL_LOG_DEBUG(log, "[Hessian] eigenvalue problem took {:.4e} seconds\n",
599 t1 - t0);
600 if (es.info() != Eigen::Success) {
601 QUILL_LOG_ERROR(log,
602 "[Hessian] SelfAdjointEigenSolver failed (info={}); "
603 "aborting",
604 static_cast<int>(es.info()));
605 return false;
606 }
607 freqs = es.eigenvalues();
608 if (!freqs.allFinite()) {
609 QUILL_LOG_ERROR(log, "[Hessian] non-finite eigenvalues; aborting");
610 return false;
611 }
612
613 return true;
614}

616VectorXd Hessian::removeZeroFreqs(const VectorXd &freqs) {
617 QUILL_LOG_DEBUG(log, "[Hessian] removing zero frequency modes");
618 int size = freqs.size();
619 if (size != 3 * matter->numberOfAtoms()) {
620 return freqs;
621 }
622 VectorXd newfreqs;
623 newfreqs.resize(size);
624 int nremoved = 0;
625 for (int i = 0; i < size; i++) {
626 if (std::abs(freqs(i)) > parameters.hessian_options().zero_freq_value) {
627 newfreqs(i - nremoved) = freqs(i);
628 } else {
629 nremoved++;
630 }
631 }
632
633 if (nremoved != 6) {
634 QUILL_LOG_ERROR(
635 log, "[Hessian] [error] Found {} trivial eigenmodes instead of 6",
636 nremoved);
637 }
638 return newfreqs.head(size - nremoved);
639}
