79 forces = matter.forces;
80 masses = matter.masses;
81 atomicNrs = matter.atomicNrs;
82 isFixed = matter.isFixed;
83 atomIndex = matter.atomIndex;
84 fileToMatter = matter.fileToMatter;
85 cell = matter.cell;
86 cellInverse = matter.cellInverse;
87 velocities = matter.velocities;
88
89 removeNetForce = matter.removeNetForce;
90 structComp = matter.structComp;
91 parameters = matter.parameters;
92
93 usePeriodicBoundaries = matter.usePeriodicBoundaries;
94 pbcConvention = matter.pbcConvention;
95
96 potential = matter.potential;
97 potentialEnergy = matter.potentialEnergy;
98 energyVariance = matter.energyVariance;
99 forceCalls = matter.forceCalls;
100 recomputePotential = matter.recomputePotential;
101 // Both caches describe the forces this object held before the assignment.
102 // resize() above already raises them; state it here alongside the members
103 // this function owns.
104 recomputeFreeMask = true;
105 recomputeMaskedForces = true;
106
107 // A BondBoost binds to one Matter (BondBoost.h:32 takes a Matter *), so a
108 // copy cannot share the source's: boosting through it would drive the
109 // original. The copy starts without one and re-establishes it through
110 // setBiasPotential. biasForces is zeroed by the resize above, which is the
111 // state that matches having no bias potential.
112 biasPotential = nullptr;
113
114 headerCon = matter.headerCon;
115 // ConFrame is move-only; copy does not retain movie trajectory.
116 movie_frames_.clear();
117
118 return *this;
119}

166bool Matter::compare(const Matter &matter, bool indistinguishable) {
167 if (nAtoms != matter.numberOfAtoms())
168 return false;
169 if (structComp.check_rotation && indistinguishable) {
170 return eonc::geometry::sortedR(*this, matter,
171 structComp.distance_difference);
172 } else if (indistinguishable) {
173 if (this->numberOfFixedAtoms() == 0 and structComp.remove_translation)
174 eonc::geometry::translationRemove(*this, matter);
175 return eonc::geometry::identical(*this, matter,
176 structComp.distance_difference);
177 } else if (structComp.check_rotation) {
178 return eonc::geometry::rotationMatch(*this, matter,
179 structComp.distance_difference);
180 } else {
181 if (this->numberOfFixedAtoms() == 0 and structComp.remove_translation)
182 eonc::geometry::translationRemove(*this, matter);
183 return (structComp.distance_difference) > perAtomNorm(matter);
184 }
185}

195// Returns the maximum distance between two atoms in the Matter objects.
196double Matter::perAtomNorm(const Matter &matter) {
197 long i = 0;
198 double max_distance = 0.0;
199
200 if (matter.numberOfAtoms() == nAtoms) {
201 AtomMatrix diff = pbc(positions - matter.positions);
202 for (i = 0; i < nAtoms; i++) {
203 max_distance = std::max(diff.row(i).norm(), max_distance);
204 }
205 }
206 return max_distance;
207}

209void Matter::resize(const long int length) {
210 if (length < 0) {
211 throw std::invalid_argument("Matter::resize: negative atom count");
212 }
213 // Same-N resize still zeros coordinates. Keep .con column-5 ids and
214 // the file-order map so a later matter2con does not stamp 1..N.
215 const bool keepAtomIds = (length == nAtoms && atomIndex.size() == length &&
216 fileToMatter.size() == static_cast<size_t>(length));
217 // Zero is a real size: leaving nAtoms at the old value there sends
218 // setMasses and every other nAtoms loop off the end of an empty array.
219 nAtoms = length;
220 positions.resize(length, 3);
221 positions.setZero();
222
223 velocities.resize(length, 3);
224 velocities.setZero();
225
226 biasForces.resize(length, 3);
227 biasForces.setZero();
228
229 forces.resize(length, 3);
230 forces.setZero();
231
232 masses.resize(length);
233 masses.setZero();
234
235 atomicNrs.resize(length);
236 atomicNrs.setZero();
237
238 isFixed.resize(length, 3);
239 isFixed.setZero();
240
241 if (!keepAtomIds) {
242 atomIndex.resize(length);
243 fileToMatter.resize(static_cast<size_t>(length));
244 for (long i = 0; i < length; i++) {
245 atomIndex(i) = static_cast<std::int64_t>(i); // default: sequential
246 fileToMatter[static_cast<size_t>(i)] = i;
247 }
248 }
249 recomputePotential = true;
250 recomputeMaskedForces = true;
251 recomputeFreeMask = true;
252}

390// Return forces with fixed atoms zeroed (cached).
391// Note: not thread-safe. Concurrent reads on the same Matter instance
392// may race on the mutable maskedForces/recomputeMaskedForces members.
393const AtomMatrix &Matter::getForces() const {
394 computePotential();
395 if (recomputeMaskedForces) {
396 // Use the cached freeMask (Nx3, 1.0 for free / 0.0 for fixed) to zero
397 // fixed-atom forces in a single vectorized Eigen operation.
398 maskedForces = forces.array() * getFree().array();
399 recomputeMaskedForces = false;
400 }
401 return maskedForces;

538double Matter::getKineticEnergy() const {
539 // 0.5 * sum(mass_i * |v_i,free|^2); a constrained axis does not contribute.
540 AtomMatrix vfree = velocities.array() * getFree().array();
541 Eigen::VectorXd speed2 = vfree.rowwise().squaredNorm();
542 return 0.5 * (masses.array() * speed2.array()).sum();
543}

570void Matter::assertIsolatedMoleculeLayoutSafe() const {
571 if (!potential || !potential->requiresIsolatedMoleculeLayout()) {
572 return;
573 }
574 if (usePeriodicBoundaries) {
575 throw std::runtime_error(
576 "Potential requires an isolated (non-periodic) molecular layout "
577 "(NWChem/ORCA-class). Disable periodic boundaries before optimizing; "
578 "PBC wraps can tear non-centered molecules (issue #188).");
579 }
580}

582void Matter::computePotential() const {
583 if (recomputePotential) {
584 if (!potential) {
585 throw std::runtime_error(
586 "Matter::computePotential called without a potential");
587 }
588 assertIsolatedMoleculeLayoutSafe();
589 if (potential->isSurrogate()) {
590 // Surrogate potential case: uses free-atom subset interface
591 auto surrogatePotential =
592 static_cast<SurrogatePotential *>(potential.get());
593 auto [freePE, freeForces, vari] = surrogatePotential->get_ef_var(
594 this->getPositionsFree(), this->getAtomicNrsFree(), cell);
595 this->potentialEnergy = freePE;
596 this->energyVariance = vari;
597 for (long idx{0}, jdx{0}; idx < nAtoms; idx++) {
598 if (!getFixed(idx)) {
599 forces.row(idx) = freeForces.row(jdx);
600 jdx++;
601 }
602 }
603 } else {
604 // Hot path: call force() directly into member storage.
605 // No intermediate allocation, no tuple, no copy.
606 double var{0};
607 potential->setFixedMask(nAtoms, isFixed.data());
608 const auto n = static_cast<size_t>(nAtoms);
609 // Isolated molecules still store a box for I/O. Pots that infer PBC
610 // from a non-zero cell (GFN2) must see a zero box here.
611 const Matrix3d force_cell =
612 usePeriodicBoundaries ? cell : Matrix3d::Zero();
613 potential->force(std::span<const double>(positions.data(), n * 3),
614 std::span<const int>(atomicNrs.data(), n),
615 std::span<double>(forces.data(), n * 3),
616 &potentialEnergy, &var,
617 std::span<const double>(force_cell.data(), 9));
618 potential->forceCallCounter++;
619 PotRegistry::get().on_force_call(potential->getType());
620 }
621 if (!std::isfinite(potentialEnergy) || !forces.allFinite()) {
622 throw std::runtime_error(
623 "Potential returned non-finite energy or forces");
624 }
625 forceCalls = forceCalls + 1;
626 recomputePotential = false;
627
628 // One free atom: subtracting the mean force is identically zero
629 // (eOn-zjri). NEB would then report immediate GOOD.
630 if (isFixed.maxCoeff() < 0.5 && removeNetForce && nAtoms > 1) {
631 Vector3d tempForce = forces.colwise().sum() / nAtoms;
632 for (long int i = 0; i < nAtoms; i++) {
633 forces.row(i) -= tempForce.transpose();
634 }
635 }
636 }
637}

639// Transform coordinates into the cell using the selected PBC convention (#176).
640// Legacy: fractional [0,1) via fmod (historical). MinimumImage: fractional
641// [-0.5,0.5) via floor (same MIC as eonc::pbc::apply for differences).
642void Matter::applyPeriodicBoundary() {
643 assertIsolatedMoleculeLayoutSafe();
644 // Unset / singular cell (default after Matter construct is Zero) — do not
645 // wipe coordinates; callers set the cell before wrapping makes sense.
646 if (std::abs(cell.determinant()) < 1e-30) {
647 return;
648 }
649 positions =
650 eonc::pbc::applyPositions(positions, cell, cellInverse, pbcConvention);
651}

653double Matter::maxForce() const {
654 // Ensures that the forces are up to date
655 computePotential();
656
657 const AtomMatrix &f = getForces();
658 double maxForce = 0.0;
659 for (int i = 0; i < nAtoms; i++) {
660 if (getFixed(i)) {
661 continue;
662 }
663 maxForce = std::max(f.row(i).norm(), maxForce);
664 }
665 return maxForce;
666}

729void Matter::setPotential(std::shared_ptr<Potential> pot) {
730 this->potential = pot;
731 // Molecular QM backends (NWChem/ORCA) must not use PBC wraps (#188). Auto-off
732 // with a hard fail if code later forces PBC while this pot is attached.
733 if (potential && potential->requiresIsolatedMoleculeLayout() &&
734 usePeriodicBoundaries) {
735 usePeriodicBoundaries = false;
736 // Use EONC_LOG_* (not bare QUILL_LOG_* with eonc::log::get() as arg): the
737 // quill macros expand logger->… and choke on a call expression as the first
738 // macro argument on some GCC versions (CI: expected primary-expression).
739 EONC_LOG_WARNING(
740 "Disabled PBC for isolated-molecule potential (NWChem/ORCA-class); "
741 "re-enabling PBC will throw (issue #188)");
742 }
743 recomputePotential = true;
744 recomputeMaskedForces = true;
745}

747void Matter::setComputedPotential(double energy, double variance) {
748 potentialEnergy = energy;
749 energyVariance = variance;
750 recomputePotential = false;
751 recomputeMaskedForces = true;
752 forceCalls++;
753
754 // Apply the same net force removal as computePotential()
755 if (isFixed.maxCoeff() < 0.5 && removeNetForce && nAtoms > 1) {
756 Vector3d tempForce = forces.colwise().sum() / nAtoms;
757 for (long int i = 0; i < nAtoms; i++) {
758 forces.row(i) -= tempForce.transpose();
759 }
760 }
