53struct LinebSurface {
54 Matter *probe = nullptr;
55 std::vector<long> freeAtoms;
56 int64_t surfaceRows = 0;
57 std::exception_ptr failure{};
58};
59
63rgsaddle_status_t surfaceCallback(void *user, rgsaddle_surface_request_t *req) {
64 auto *ctx = static_cast<LinebSurface *>(user);
65 if (ctx == nullptr || ctx->probe == nullptr || req == nullptr ||
66 req->positions == nullptr || req->energies == nullptr ||
67 req->gradients == nullptr) {
68 return RGSADDLE_SURFACE_FAILED;
69 }
70 const auto nFree = static_cast<int64_t>(ctx->freeAtoms.size());
71 if (req->n_atoms != nFree) {
72 return RGSADDLE_SHAPE;
73 }
74
75 const int64_t nImages =
76 (req->flags & RGSADDLE_REQ_ONE_IMAGE) ? 1 : req->n_images;
77 if (nImages < 1) {
78 return RGSADDLE_SHAPE;
79 }
80
81 try {
82 AtomMatrix pos = ctx->probe->getPositions();
83 for (int64_t image = 0; image < nImages; ++image) {
84 const double *xyz = req->positions + image * nFree * 3;
85 for (int64_t a = 0; a < nFree; ++a) {
86 const long atom = ctx->freeAtoms[static_cast<size_t>(a)];
87 pos(atom, 0) = xyz[3 * a + 0];
88 pos(atom, 1) = xyz[3 * a + 1];
89 pos(atom, 2) = xyz[3 * a + 2];
90 }
91 ctx->probe->setPositions(pos);
92 req->energies[image] = ctx->probe->getPotentialEnergy();
93 const AtomMatrix forces = ctx->probe->getForces();
94 double *grad = req->gradients + image * nFree * 3;
95 for (int64_t a = 0; a < nFree; ++a) {
96 const long atom = ctx->freeAtoms[static_cast<size_t>(a)];
97 grad[3 * a + 0] = -forces(atom, 0);
98 grad[3 * a + 1] = -forces(atom, 1);
99 grad[3 * a + 2] = -forces(atom, 2);
100 }
101 }
102 ctx->surfaceRows += nImages;
103 return RGSADDLE_OK;
104 } catch (...) {
105 ctx->failure = std::current_exception();
106 return RGSADDLE_SURFACE_FAILED;
107 }
108}

111void writeRateOrbit(const std::string &filename, const Matter &seed,
112 const std::vector<long> &freeAtoms,
113 const std::vector<double> &positions,
114 const std::vector<double> &gradients,
115 nlohmann::json metadata) {
116 Matter bead(seed);
117 // Preserve the continuous Cartesian path across periodic cell boundaries.
118 // The copy is only serialized; the potential is never called on it.
119 bead.setPeriodic(false);
120 AtomMatrix xyz = bead.getPositions();
121 const size_t width = 3 * freeAtoms.size();
122 std::vector<readcon::ConFrame> frames;
123 const auto numBeads = positions.size() / width;
124 frames.reserve(numBeads);
125 metadata["temperature_K"] = metadata.at("lineb").at("temperature");
126 auto &observed = metadata["lineb_orbit"];
127 for (const auto atom : freeAtoms) {
128 observed["mobile_atom_ids"].push_back(bead.getAtomIndex(atom));
129 }
130 for (size_t offset = 0; offset < positions.size(); offset += width) {
131 auto &rows = observed["gradients_eV_per_angstrom"];
132 rows = nlohmann::json::array();
133 for (size_t a = 0; a < freeAtoms.size(); ++a) {
134 const size_t k = offset + 3 * a;
135 xyz.row(freeAtoms[a]) =
136 Eigen::Map<const Eigen::RowVector3d>(positions.data() + k);
137 rows.push_back({gradients[k], gradients[k + 1], gradients[k + 2]});
138 }
139 bead.setPositions(xyz);
140 io::ConFrameMetadata frame;
141 frame.frame_index = frames.size();
142 metadata["tau_over_beta"] = static_cast<double>(frames.size()) / numBeads;
143 frame.write_con_forces = false;
144 frame.raw_json = metadata.dump();
145 frames.push_back(io::matterToConFrame(bead, &frame));
146 }
147 if (!io::io_ok(io::writeConFrames(filename, frames))) {
148 throw std::runtime_error("lineb: cannot write " + filename);
149 }
150}

153void unpackPositions(const std::vector<double> &flat,
154 const std::vector<long> &freeAtoms,
155 std::vector<std::shared_ptr<Matter>> &band) {
156 const auto nFree = freeAtoms.size();
157 if (nFree == 0 || flat.size() % (nFree * 3) != 0) {
158 throw std::runtime_error("lineb: optimized path has the wrong length");
159 }
160 const size_t nImages = flat.size() / (nFree * 3);
161 if (nImages != band.size()) {
162 throw std::runtime_error("lineb: optimized path image count changed");
163 }
164 for (size_t i = 0; i < nImages; ++i) {
165 AtomMatrix pos = band[i]->getPositions();
166 for (size_t a = 0; a < nFree; ++a) {
167 const long atom = freeAtoms[a];
168 const size_t base = (i * nFree + a) * 3;
169 pos(atom, 0) = flat[base + 0];
170 pos(atom, 1) = flat[base + 1];
171 pos(atom, 2) = flat[base + 2];
172 }
173 band[i]->setPositions(pos);
174 }
175}

181std::vector<std::string> LinebJob::run(void) {
182 if (!linebAvailable()) {
183 throw std::runtime_error(
184 "lineb: needs -Dwith_rgsaddle=true against a rgsaddle that exports "
185 "rgsaddle_lineb_solve (lineb C API)");
186 }
187
188#ifdef WITH_RGSADDLE_LINEB
189 const auto &o = params.lineb_options();
190 if (!(o.energy < std::numeric_limits<double>::infinity()) ||
191 !std::isfinite(o.energy)) {
192 throw std::runtime_error("lineb: [Lineb] energy must be a finite tunneling "
193 "energy below the barrier");
194 }
195 if (o.max_steps < 1) {
196 throw std::runtime_error("lineb: [Lineb] max_steps must be at least 1");
197 }
198 if (o.beads < 0) {
199 throw std::runtime_error(
200 "lineb: [Lineb] beads must be nonnegative (0 is path only)");
201 }
202
203#ifdef RGPOT_HAS_CACHE
204 std::optional<ScopedRgpotCache> resultCache;
205 if (!o.result_cache.empty()) {
206 if (auto *provider = dynamic_cast<RgpotCacheProvider *>(pot.get()))
207 resultCache.emplace(*provider, o.result_cache);
208 else if (auto *engine = dynamic_cast<RgpotEngineCacheProvider *>(pot.get()))
209 resultCache.emplace(*engine, o.result_cache);
210 else
211 throw std::runtime_error("lineb: result_cache requires an rgpot adapter");
212 }
213 const auto cacheCounts = [&] {
214 return resultCache ? resultCache->counts() : rgpot::cache::EvalCounts{};
215 };
216 const auto cacheAtStart = cacheCounts();
217#else
218 if (!o.result_cache.empty()) {
219 throw std::runtime_error(
220 "lineb: result_cache requires rgpot built with cache support");
221 }
222#endif
223 const auto jobStart = pot->forceCallCounter.load();
224 auto band = lineb::loadSeedBand(params, pot, o.initial_path);
225 std::vector<long> freeAtoms;
226 std::vector<double> masses;
227 lineb::collectFreeAtoms(*band.front(), freeAtoms, masses);
228 for (size_t i = 1; i < band.size(); ++i) {
229 std::vector<long> otherFree;
230 std::vector<double> otherMasses;
231 lineb::collectFreeAtoms(*band[i], otherFree, otherMasses);
232 if (otherFree != freeAtoms) {
233 throw std::runtime_error(
234 "lineb: every seed image must share the same free-atom mask");
235 }
236 }
237
238 std::vector<double> positions;
239 // Unwrapped Cartesian for the C API only; Matter stays on its wrapped seed.
240 lineb::packSeedCartesian(band, freeAtoms, o.molecular, positions);
241 std::vector<double> optimized(positions.size(), 0.0);
242
243 rgsaddle_lineb_config_t cfg{};
244 const rgsaddle_status_t cfgRc =
245 rgsaddle_lineb_config_default(o.opt_tol, o.perp_tol, &cfg);
246 if (cfgRc != RGSADDLE_OK) {
247 throw std::runtime_error(
248 std::string("lineb: rgsaddle_lineb_config_default: ") +
249 rgsaddle_status_name(cfgRc));
250 }
251 cfg.spring = o.spring;
252 cfg.endpoint_scale = o.endpoint_scale;
253 cfg.time_step = o.time_step;
254 cfg.max_move = o.max_move;
255 cfg.turning_width = o.turning_width;
256 cfg.hbar = o.hbar;
257 cfg.boltzmann = o.boltzmann;
258 cfg.quadrature_pairs = o.quadrature_pairs;
259
260 Matter probe(*band.front());
261 LinebSurface ctx{&probe, freeAtoms};
262
263#ifdef RGPOT_HAS_CACHE
264 const auto cacheAtSolve = cacheCounts();
265#endif
266 const auto solveStart = pot->forceCallCounter.load();
267 rgsaddle_lineb_result_t result{};
268#ifndef WITH_RGSADDLE_LINEB_RATE
269 if (o.interpolated_hessian) {
270 throw std::runtime_error("lineb: interpolated_hessian requires "
271 "rgsaddle_lineb_solve_counted_rate");
272 }
273#endif
274 if (o.adjacent_alignment && !o.molecular) {
275 throw std::runtime_error(
276 "lineb: adjacent_alignment requires molecular = true");
277 }
278#ifdef RGSADDLE_LINEB_MASSES_MOLECULE_ADJACENT
279 const int32_t massMode =
280 o.molecular
281 ? (o.adjacent_alignment ? RGSADDLE_LINEB_MASSES_MOLECULE_ADJACENT
282 : RGSADDLE_LINEB_MASSES_MOLECULE)
283 : RGSADDLE_LINEB_MASSES_COORDINATES;
284#else
285 if (o.adjacent_alignment) {
286 throw std::runtime_error("lineb: adjacent_alignment requires rgsaddle with "
287 "RGSADDLE_LINEB_MASSES_MOLECULE_ADJACENT");
288 }
289 const int32_t massMode = o.molecular ? 1 : 0;
290#endif
291#ifndef RGSADDLE_LINEB_INTERPOLATED_FORCE_CLOCK
292 if (o.interpolated_force_clock) {
293 throw std::runtime_error(
294 "lineb: interpolated_force_clock requires rgsaddle with "
295 "RGSADDLE_LINEB_INTERPOLATED_FORCE_CLOCK");
296 }
297#endif
298#ifndef RGSADDLE_LINEB_PROJECTED_VERLET
299 if (o.projected_verlet) {
300 throw std::runtime_error("lineb: projected_verlet requires rgsaddle with "
301 "RGSADDLE_LINEB_PROJECTED_VERLET");
302 }
303#endif
304#ifndef RGSADDLE_LINEB_LINEAR_SEED
305 if (o.linear_seed || o.endpoint_convergence) {
306 throw std::runtime_error("lineb: linear_seed and endpoint_convergence "
307 "require rgsaddle with fixed-energy seed options");
308 }
309#endif
310#ifdef RGSADDLE_LINEB_LINEAR_SEED
311 uint32_t options =
312 (o.linear_seed ? RGSADDLE_LINEB_LINEAR_SEED : 0u) |
313 (o.endpoint_convergence ? RGSADDLE_LINEB_ENDPOINT_CONVERGENCE : 0u);
314#ifdef RGSADDLE_LINEB_PROJECTED_VERLET
315 if (o.projected_verlet) {
316 options |= RGSADDLE_LINEB_PROJECTED_VERLET;
317 }
318#endif
319#ifdef RGSADDLE_LINEB_INTERPOLATED_FORCE_CLOCK
320 if (o.interpolated_force_clock) {
321 options |= RGSADDLE_LINEB_INTERPOLATED_FORCE_CLOCK;
322 }
323#endif
324#ifdef WITH_RGSADDLE_LINEB_COUNTS
325 rgsaddle_lineb_counts_t counts{};
326#ifdef RGSADDLE_HAS_LINEB_RATE_DIAGNOSTICS
327 int64_t negativeModes = -1;
328 int64_t retainedModes = -1;
329#ifdef WITH_RGSADDLE_LINEB_ORBIT
330 const size_t width = 3 * freeAtoms.size();
331 if (static_cast<size_t>(o.beads) > std::vector<double>().max_size() / width) {
332 throw std::length_error("lineb: orbit exceeds addressable memory");
333 }
334 std::vector<double> orbitPositions(static_cast<size_t>(o.beads) * width);
335 std::vector<double> orbitGradients(orbitPositions.size());
336 rgsaddle_lineb_orbit_t orbit{};
337 orbit.capacity = orbitPositions.size();
338 orbit.positions = orbitPositions.data();
339 orbit.gradients = orbitGradients.data();
340#ifdef WITH_RGSADDLE_LINEB_TURNING
341 rgsaddle_lineb_turning_t turning{};
342 const auto solveRate = [&](auto fn, auto... rateOptions) {
343 return fn(&cfg, static_cast<int64_t>(band.size()),
344 static_cast<int64_t>(freeAtoms.size()), positions.data(),
345 masses.data(), massMode, o.energy, o.max_steps, o.beads,
346 surfaceCallback, &ctx, optimized.data(), &result,
347 o.endpoint_kappa > 0.0 ? &o.endpoint_kappa : nullptr, options,
348 &counts, &negativeModes, &retainedModes,
349 o.beads ? &orbit : nullptr, &turning, rateOptions...);
350 };
351#ifdef WITH_RGSADDLE_LINEB_RATE
352 if (o.interpolated_hessian) {
353 options |= RGSADDLE_LINEB_INTERPOLATED_HESSIAN;
354 }
355 rgsaddle_lineb_rate_diagnostics_t rateDiagnostics{};
356 const auto rc =
357 solveRate(rgsaddle_lineb_solve_counted_rate,
358 o.image_hessian_step > 0 ? &o.image_hessian_step : nullptr,
359 &rateDiagnostics);
360#else
361 const auto rc = solveRate(rgsaddle_lineb_solve_counted_turning);
362#endif
363#else
364 const rgsaddle_status_t rc = rgsaddle_lineb_solve_counted_orbit(
365 &cfg, static_cast<int64_t>(band.size()),
366 static_cast<int64_t>(freeAtoms.size()), positions.data(), masses.data(),
367 massMode, o.energy, o.max_steps, o.beads, surfaceCallback, &ctx,
368 optimized.data(), &result,
369 o.endpoint_kappa > 0.0 ? &o.endpoint_kappa : nullptr, options, &counts,
370 &negativeModes, &retainedModes, o.beads ? &orbit : nullptr);
371#endif
372#else
373 const rgsaddle_status_t rc = rgsaddle_lineb_solve_counted_diagnostics(
374 &cfg, static_cast<int64_t>(band.size()),
375 static_cast<int64_t>(freeAtoms.size()), positions.data(), masses.data(),
376 massMode, o.energy, o.max_steps, o.beads, surfaceCallback, &ctx,
377 optimized.data(), &result,
378 o.endpoint_kappa > 0.0 ? &o.endpoint_kappa : nullptr, options, &counts,
379 &negativeModes, &retainedModes);
380#endif
381#else
382 const rgsaddle_status_t rc = rgsaddle_lineb_solve_counted(
383 &cfg, static_cast<int64_t>(band.size()),
384 static_cast<int64_t>(freeAtoms.size()), positions.data(), masses.data(),
385 massMode, o.energy, o.max_steps, o.beads, surfaceCallback, &ctx,
386 optimized.data(), &result,
387 o.endpoint_kappa > 0.0 ? &o.endpoint_kappa : nullptr, options, &counts);
388#endif
389#else
390 const rgsaddle_status_t rc = rgsaddle_lineb_solve_options(
391 &cfg, static_cast<int64_t>(band.size()),
392 static_cast<int64_t>(freeAtoms.size()), positions.data(), masses.data(),
393 massMode, o.energy, o.max_steps, o.beads, surfaceCallback, &ctx,
394 optimized.data(), &result,
395 o.endpoint_kappa > 0.0 ? &o.endpoint_kappa : nullptr, options);
396#endif
397#elif defined(WITH_RGSADDLE_LINEB_KAPPA)
398 const rgsaddle_status_t rc = rgsaddle_lineb_solve_with_kappa(
399 &cfg, static_cast<int64_t>(band.size()),
400 static_cast<int64_t>(freeAtoms.size()), positions.data(), masses.data(),
401 massMode, o.energy, o.max_steps, o.beads, surfaceCallback, &ctx,
402 optimized.data(), &result,
403 o.endpoint_kappa > 0.0 ? &o.endpoint_kappa : nullptr);
404#else
405 if (o.endpoint_kappa > 0.0) {
406 throw std::runtime_error(
407 "lineb: endpoint_kappa requires rgsaddle_lineb_solve_with_kappa");
408 }
409 const rgsaddle_status_t rc = rgsaddle_lineb_solve(
410 &cfg, static_cast<int64_t>(band.size()),
411 static_cast<int64_t>(freeAtoms.size()), positions.data(), masses.data(),
412 massMode, o.energy, o.max_steps, o.beads, surfaceCallback, &ctx,
413 optimized.data(), &result);
414#endif
415
416 if (rc != RGSADDLE_OK) {
417 std::string message =
418 std::string("lineb: rgsaddle_lineb_solve: ") + rgsaddle_status_name(rc);
419 const char *detail = rgsaddle_last_error();
420 if (detail && *detail) {
421 message += std::string(" (") + detail + ")";
422 }
423 if (ctx.failure) {
424 try {
425 std::rethrow_exception(ctx.failure);
426 } catch (const std::exception &e) {
427 message += std::string(": ") + e.what();
428 } catch (...) {
429 message += ": unknown potential exception";
430 }
431 }
432 throw std::runtime_error(message);
433 }
434
435#ifdef RGPOT_HAS_CACHE
436 const auto cacheAtOutput = cacheCounts();
437#endif
438 const auto solveEnd = pot->forceCallCounter.load();
439 unpackPositions(optimized, freeAtoms, band);
440
441 const std::string pathFile = "lineb.con";
442 const std::string resultsFile = "results.dat";
443 std::vector<io::ConFrameMetadata> metas(band.size());
444 for (size_t i = 0; i < band.size(); ++i) {
445 metas[i].energy = band[i]->getPotentialEnergy();
446 }
447#ifdef WITH_RGSADDLE_LINEB_COUNTS
448 std::map<std::string, int64_t> solverCalls{
449 {"seed", counts.seed},
450 {"optimization", counts.optimizer},
451 {"path_preparation", counts.preparation},
452 {"quadrature", counts.quadrature},
453 {"dynamics", counts.dynamics},
454 {"rate", counts.rate}};
455#else
456 std::map<std::string, int64_t> solverCalls{
457 {"optimizer_reported", result.optimizer_calls},
458 {"measure_reported", result.measure_calls},
459 {"dynamics", result.dynamics_calls},
460 {"rate", result.rate_calls}};
461#endif
462 int64_t classified = 0;
463 for (const auto &entry : solverCalls)
464 classified += entry.second;
465#ifdef WITH_RGSADDLE_LINEB_COUNTS
466 if (classified != ctx.surfaceRows)
467 throw std::runtime_error("lineb: solver counts disagree with observed "
468 "configuration evaluations");
469#else
470 solverCalls.emplace("unclassified", ctx.surfaceRows - classified);
471#endif
472 solverCalls.emplace("total", ctx.surfaceRows);
473 const auto outputEnd = pot->forceCallCounter.load();
474 const std::map<std::string, size_t> potentialCalls{
475 {"setup", solveStart - jobStart},
476 {"solve", solveEnd - solveStart},
477 {"output", outputEnd - solveEnd},
478 {"total", outputEnd - jobStart}};
479 const auto provenance = provenanceForJob(params, pot.get());
480 nlohmann::json metadata = {
481 {"evaluation_counts",
482 {{"scope", "job"},
483 {"boundary", "solver_callback_geometry"},
484 {"surface_calls", solverCalls},
485 {"potential_calls", potentialCalls},
486 {"driver_rows_by_provider",
487 {{provenance.rgpot_name, outputEnd - jobStart}}}}},
488 {"lineb",
489 {{"scope", "job"},
490 {"energy", o.energy},
491 {"hbar", o.hbar},
492 {"boltzmann", o.boltzmann},
493 {"molecular", o.molecular},
494 {"adjacent_alignment", o.adjacent_alignment},
495 {"beads", o.beads},
496 {"iterations", result.iterations},
497 {"action", result.action},
498 {"temperature", result.temperature},
499 {"temperature_dynamics", result.temperature_dynamics},
500 {"half_period", result.half_period},
501 {"half_period_dynamics", result.half_period_dynamics},
502 {"log_rate_over_reactant",
503 o.beads ? nlohmann::json(result.log_rate_over_reactant)
504 : nlohmann::json(nullptr)},
505 {"optimizer", o.projected_verlet ? "projected_verlet" : "fire"},
506 {"replica_clock", o.interpolated_force_clock
507 ? "interpolated_force_verlet"
508 : "quadrature"},
509 {"surface_call_unit", "configuration"},
510 {"surface_calls", solverCalls},
511 {"potential_calls", potentialCalls}}}};
512#ifdef WITH_RGSADDLE_LINEB_TURNING
513 if (o.interpolated_force_clock && turning.launch_image >= 0) {
514 metadata["lineb"]["turning"] = {
515 {"launch_image", turning.launch_image},
516 {"path_arc_length", turning.path_arc_length},
517 {"return_arc", turning.return_arc},
518 {"max_cartesian_displacement_angstrom",
519 turning.max_cartesian_displacement},
520 {"return_energy_minus_target_eV", turning.energy_residual},
521 {"return_energy_minus_opposite_image_eV",
522 turning.endpoint_energy_difference}};
523 }
524#endif
525 if (o.beads > 0) {
526 metadata["lineb"]["hessian"] =
527 o.interpolated_hessian ? "interpolated_images" : "beads";
528#ifdef WITH_RGSADDLE_LINEB_RATE
529 metadata["lineb"]["rate_calls"] = {
530 {"curvature", rateDiagnostics.curvature_evaluations},
531 {"bead", rateDiagnostics.bead_evaluations}};
532 if (o.interpolated_hessian) {
533 metadata["lineb"]["image_hessian_step_angstrom"] =
534 rateDiagnostics.image_hessian_step;
535 }
536#endif
537 }
538 if (!provenance.rgpot_backend.empty()) {
539 metadata["potential"] = {{"schema", "eon.rgpot.v1"},
540 {"provider", "rgpot"},
541 {"potential_type", provenance.rgpot_name},
542 {"kernel", "RgpotPot"},
543 {"backend", provenance.rgpot_backend},
544 {"revision", ""}};
545 metadata["potential"]["linked_version"] = provenance.rgpot_version;
546 if (!provenance.rgpot_engine_version.empty())
547 metadata["potential"]["engine_version"] = provenance.rgpot_engine_version;
548 }
549 auto &spectrum = metadata["lineb"]["spectrum"];
550 spectrum = nullptr;
551#ifdef RGSADDLE_HAS_LINEB_RATE_DIAGNOSTICS
552 if (negativeModes >= 0 && retainedModes >= 0) {
553 spectrum = {{"negative_modes", negativeModes}, {"modes", retainedModes}};
554 }
555#endif
556 auto &providerResults = metadata["evaluation_counts"]["rgpot_results"];
557 providerResults = nullptr;
558#ifdef RGPOT_HAS_CACHE
559 if (resultCache) {
560 const auto end = cacheCounts();
561 const auto delta = [](auto first, auto last) {
562 return nlohmann::json{{"computed", last.computed - first.computed},
563 {"served", last.served - first.served}};
564 };
565 providerResults = delta(cacheAtStart, end);
566 providerResults["scope"] = "job";
567 providerResults["phases"] = {{"setup", delta(cacheAtStart, cacheAtSolve)},
568 {"solve", delta(cacheAtSolve, cacheAtOutput)},
569 {"output", delta(cacheAtOutput, end)}};
570 }
571#endif
572 std::vector<std::string> files{resultsFile, pathFile};
573#ifdef WITH_RGSADDLE_LINEB_ORBIT
574 if (o.beads > 0) {
575 const auto component = static_cast<size_t>(orbit.max_coordinate);
576 const long atom = freeAtoms.at(component / 3);
577 metadata["lineb"]["stationarity"] = {
578 {"discretization", "ring_polymer"},
579 {"max_abs_eV_per_angstrom", orbit.max_abs_residual},
580 {"rms_eV_per_angstrom", orbit.rms_residual},
581 {"max_bead", orbit.max_bead},
582 {"max_atom_id", band.front()->getAtomIndex(atom)},
