24JobType job_from_ssot(std::string_view j) {
25 if (j == "process_search")
26 return JobType::Process_Search;
27 if (j == "minimization")
28 return JobType::Minimization;
29 if (j == "saddle_search")
30 return JobType::Saddle_Search;
31 if (j == "basin_hopping")
32 return JobType::Basin_Hopping;
33 if (j == "parallel_replica" || j == "unbiased_parallel_replica")
34 return JobType::Parallel_Replica;
35 if (j == "nudged_elastic_band")
36 return JobType::Nudged_Elastic_Band;
37 if (j == "dynamics" || j == "molecular_dynamics")
38 return JobType::Dynamics;
39 if (j == "hessian")
40 return JobType::Hessian;
41 if (j == "point")
42 return JobType::Point;
43 if (j == "prefactor")
44 return JobType::Prefactor;
45 if (j == "monte_carlo")
46 return JobType::Monte_Carlo;
47 if (j == "structure_comparison")
48 return JobType::Structure_Comparison;
49 if (j == "gp_surrogate")
50 return JobType::GP_Surrogate;
51 if (j == "safe_hyperdynamics")
52 return JobType::Safe_Hyperdynamics;
53 if (j == "tad")
54 return JobType::TAD;
55 if (j == "replica_exchange")
56 return JobType::Replica_Exchange;
57 if (j == "finite_difference" || j == "finite_differences")
58 return JobType::Finite_Difference;
59 if (j == "global_optimization")
60 return JobType::Global_Optimization;
61 return JobType::Process_Search;
62}
