39MetatomicDynPot::MetatomicDynPot(const Parameters &params)
40 : Potential(PotType::METATOMIC) {
41 auto &loader = MetatomicLoader::instance();
42 loader.require_loaded();
43 std::array<char, 1024> err{};
44 auto cfg = config_from_params(params);
45 m_handle = loader.create(&cfg, err.data(), err.size());
46 if (!m_handle) {
47 throw std::runtime_error(std::string("MetatomicDynPot: create failed: ") +
48 err.data());
49 }
50}
51
52MetatomicDynPot::~MetatomicDynPot() {
53 if (m_handle) {
54 MetatomicLoader::instance().destroy(m_handle);
55 m_handle = nullptr;
56 }
57}
