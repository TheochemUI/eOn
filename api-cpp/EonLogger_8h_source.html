58[[nodiscard]] inline quill::ManualBackendWorker *manual_worker_for_caller() {

81inline void flush(quill::Logger *logger) noexcept {
82 if (manual_backend()) {
83 if (auto *worker = detail::manual_worker_for_caller()) {
84 worker->poll();
85 // An idle poll flushes the sinks (sink_min_flush_interval is zero).
86 worker->poll_one();
87 }
88 return;
89 }
90 // The flush record waits for a backend. One that was never started would
91 // block here for good, and one that was stopped cannot answer.
92 if (!quill::Backend::is_running()) {
93 start_client_backend(BackendMode::Thread);
94 if (!quill::Backend::is_running()) {
95 return;
96 }
97 }
98 if (logger) {
99 logger->flush_log();
100 }
101}

108inline void poll_if_due() noexcept {
109 if (!manual_backend()) {
110 return;
111 }
112 auto *worker = detail::manual_worker_for_caller();
113 if (!worker) {
114 return;
115 }
116 const auto now = std::chrono::steady_clock::now().time_since_epoch().count();
117 const auto last = detail::g_last_poll.load(std::memory_order_relaxed);
118 if (now - last <
119 std::chrono::duration_cast<std::chrono::steady_clock::duration>(
120 detail::kManualPollInterval)
121 .count()) {
122 return;
123 }
124 detail::g_last_poll.store(now, std::memory_order_relaxed);
125 worker->poll();
126 worker->poll_one();
127}

130inline void shutdown() noexcept {
131 if (manual_backend()) {
132 flush(nullptr);
133 return;
134 }
135 // Records logged before any backend existed are written by this one.
136 if (!quill::Backend::is_running()) {
137 start_client_backend(BackendMode::Thread);
138 }
139 quill::Backend::stop();
140}

148[[nodiscard]] inline quill::Logger *init_client_loggers() noexcept {
149 static std::once_flag once;
150 std::call_once(once, [] {
151 if (quill::Frontend::get_logger(std::string(detail::kDefaultLoggerName))) {
152 return;
153 }
154 try {
155 auto console_sink =
156 quill::Frontend::create_or_get_sink<quill::ConsoleSink>("console");
157 auto file_sink = quill::Frontend::create_or_get_sink<quill::FileSink>(
158 "client_quill.log",
159 []() {
160 quill::FileSinkConfig cfg;
161 cfg.set_open_mode('w');
162 return cfg;
163 }(),
164 quill::FileEventNotifier{});
165 auto *logger = quill::Frontend::create_or_get_logger(
166 "combi", {std::move(console_sink), std::move(file_sink)},
167 quill::PatternFormatterOptions{"%(message)"},
168 quill::ClockSourceType::System);
169 logger->set_log_level(quill::LogLevel::TraceL3);
170 auto trace_csink =
171 quill::Frontend::create_or_get_sink<quill::ConsoleSink>(
172 "trace_console");
173 auto trace_fsink = quill::Frontend::create_or_get_sink<quill::FileSink>(
174 "client_traceback.log",
175 []() {
176 quill::FileSinkConfig cfg;
177 cfg.set_open_mode('w');
178 return cfg;
179 }(),
180 quill::FileEventNotifier{});
181 quill::Frontend::create_or_get_logger(
182 "_traceback", {std::move(trace_csink), std::move(trace_fsink)},
183 quill::PatternFormatterOptions{
184 " [%(log_level)] [%(source_location)] [%(caller_function)] \n "
185 "%(message)\n[end %(log_level)]"},
186 quill::ClockSourceType::System);
187 // Build the logger and thread context singletons before the backend
188 // one. Statics are destroyed in reverse order, so the manual worker
189 // then drains its queue at exit while both are still alive.
190 quill::Frontend::preallocate();
191 } catch (...) {
192 }
193 });
194 return quill::Frontend::get_logger(std::string(detail::kDefaultLoggerName));
195}

203inline BackendMode start_client_backend(BackendMode mode) noexcept {
204 static std::once_flag once;
205 static BackendMode effective{BackendMode::Thread};
206 std::call_once(once, [mode] {
207 if (quill::Backend::is_running() || manual_backend()) {
208 return;
209 }
210 try {
211 quill::BackendOptions backend_options;
212 backend_options.transit_event_buffer_initial_capacity = 2048;
213 backend_options.log_timestamp_ordering_grace_period =
214 std::chrono::microseconds{0};
215 backend_options.check_printable_char = {};
216 if (mode == BackendMode::Manual) {
217 backend_options.sink_min_flush_interval = std::chrono::milliseconds{0};
218 auto *worker = quill::Backend::acquire_manual_backend_worker();
219 worker->init(backend_options);
220 detail::g_manual_owner.store(std::this_thread::get_id());
221 detail::g_manual_worker.store(worker, std::memory_order_release);
222 std::atexit([] { shutdown(); });
223 effective = BackendMode::Manual;
224 return;
225 }
226 // The backend thread sleeps this long when its queues are empty. At
227 // 10 us it woke about 13,000 times a second beside the force loop;
228 // at 1 ms a line reaches its sink at most 1 ms late. Order is kept
229 // either way, and Backend::stop and flush_log drain the queues.
230 backend_options.sleep_duration = detail::kBackendSleep;
231 backend_options.sink_min_flush_interval = std::chrono::milliseconds{100};
232 quill::Backend::start(backend_options);
233 } catch (...) {
234 }
235 });
236 return effective;
237}

255[[nodiscard]] inline quill::Logger *get() noexcept {
256 static std::atomic<quill::Logger *> cached_logger{nullptr};
257 if (quill::Logger *l = cached_logger.load(std::memory_order_relaxed)) {
258 return l;
259 }
260
261 if (auto *logger = quill::Frontend::get_logger(
262 std::string(detail::kDefaultLoggerName))) {
263 cached_logger.store(logger, std::memory_order_relaxed);
264 return logger;
265 }

321[[nodiscard]] inline quill::Logger *
322get_file(std::string_view name, std::string_view filename,
323 std::string_view pattern = "%(message)") {
324 // If the logger already exists, return it immediately.
325 // Quill strictly checks configuration matches on recreation and throws if
326 // PatternFormatterOptions differ, which crashes instances (like
327 // Dimer/Optimizers) that are created multiple times in loops (e.g., NEB).
328 if (auto *existing = quill::Frontend::get_logger(std::string(name))) {
329 return existing;
330 }
331 static std::mutex file_logger_mutex;
332 std::lock_guard<std::mutex> lock(file_logger_mutex);
333 if (auto *existing = quill::Frontend::get_logger(std::string(name))) {
334 return existing;
335 }
336
337 try {
338 // Durable absolute path: optimizers open "_lbfgs.log" etc. under CWD.
339 // Job integration fixtures chdir into /tmp/eon_test_job_* then delete that
340 // tree when the case ends, while Quill's process-lifetime FileSink still
341 // flushes — "fopen failed after 3 attempts". Keep logs under a stable
342 // temp subdir instead of the disposable workdir.
343 namespace fs = std::filesystem;
344 fs::path path{std::string(filename)};
345 if (!path.is_absolute()) {
346 static const fs::path log_dir = [] {
347 fs::path d = fs::temp_directory_path() / "eon_logs";
348 std::error_code ec;
349 fs::create_directories(d, ec);
350 return d;
351 }();
352 path = log_dir / path.filename();
353 }
354 quill::FileSinkConfig cfg;
355 cfg.set_open_mode('w');
356 auto file_sink = quill::Frontend::create_or_get_sink<quill::FileSink>(
357 path.string(), cfg);
358 return quill::Frontend::create_or_get_logger(
359 std::string(name), std::move(file_sink),
360 quill::PatternFormatterOptions{std::string(pattern)},
361 quill::ClockSourceType::System);
362 } catch (...) {
363 return get(); // Fallback to default combi logger if creation fails
364 }
365}

382struct Scoped {

385 Scoped() noexcept : logger(get()) {}
