42[[nodiscard]] inline quill::Logger *init_client() noexcept {
43 static std::once_flag once;
44 std::call_once(once, [] {
45 if (quill::Frontend::get_logger(std::string(detail::kDefaultLoggerName))) {
46 return;
47 }
48 try {
49 quill::BackendOptions backend_options;
50 backend_options.sleep_duration = std::chrono::microseconds{10};
51 backend_options.transit_event_buffer_initial_capacity = 2048;
52 backend_options.log_timestamp_ordering_grace_period =
53 std::chrono::microseconds{0};
54 backend_options.sink_min_flush_interval = std::chrono::milliseconds{100};
55 backend_options.check_printable_char = {};
56 try {
57 quill::Backend::start(backend_options);
58 } catch (...) {
59 }
60 auto console_sink =
61 quill::Frontend::create_or_get_sink<quill::ConsoleSink>("console");
62 auto file_sink = quill::Frontend::create_or_get_sink<quill::FileSink>(
63 "client_quill.log",
64 []() {
65 quill::FileSinkConfig cfg;
66 cfg.set_open_mode('w');
67 return cfg;
68 }(),
69 quill::FileEventNotifier{});
70 auto *logger = quill::Frontend::create_or_get_logger(
71 "combi", {std::move(console_sink), std::move(file_sink)},
72 quill::PatternFormatterOptions{"%(message)"},
73 quill::ClockSourceType::System);
74 logger->set_log_level(quill::LogLevel::TraceL3);
75 auto trace_csink =
76 quill::Frontend::create_or_get_sink<quill::ConsoleSink>(
77 "trace_console");
78 auto trace_fsink = quill::Frontend::create_or_get_sink<quill::FileSink>(
79 "client_traceback.log",
80 []() {
81 quill::FileSinkConfig cfg;
82 cfg.set_open_mode('w');
83 return cfg;
84 }(),
85 quill::FileEventNotifier{});
86 quill::Frontend::create_or_get_logger(
87 "_traceback", {std::move(trace_csink), std::move(trace_fsink)},
88 quill::PatternFormatterOptions{
89 " [%(log_level)] [%(source_location)] [%(caller_function)] \n "
90 "%(message)\n[end %(log_level)]"},
91 quill::ClockSourceType::System);
92 } catch (...) {
93 }
94 });
95 return quill::Frontend::get_logger(std::string(detail::kDefaultLoggerName));
96}

106[[nodiscard]] inline quill::Logger *get() noexcept {
107 static std::atomic<quill::Logger *> cached_logger{nullptr};
108 if (quill::Logger *l = cached_logger.load(std::memory_order_relaxed)) {
109 return l;
110 }
111
112 if (auto *logger = quill::Frontend::get_logger(
113 std::string(detail::kDefaultLoggerName))) {
114 cached_logger.store(logger, std::memory_order_relaxed);
115 return logger;
116 }
117
118 // Fallback for Python bindings/tests that didn't run ClientEON's main()
119 static quill::Logger *fallback = []() -> quill::Logger * {
120 try {
121 quill::Backend::start();
122 } catch (...) {
123 } // Ignore if backend already started
124
125 try {
126 auto console_sink =
127 quill::Frontend::create_or_get_sink<quill::ConsoleSink>(
128 "fallback_console");
129 return quill::Frontend::create_or_get_logger(
130 "combi", std::move(console_sink),
131 quill::PatternFormatterOptions{"%(message)"},
132 quill::ClockSourceType::System);
133 } catch (...) {
134 return nullptr;
135 }
136 }();
137 if (fallback)
138 cached_logger.store(fallback, std::memory_order_relaxed);
139 return fallback;
140}

150[[nodiscard]] inline quill::Logger *traceback() noexcept {
151 static std::atomic<quill::Logger *> cached{nullptr};
152 if (quill::Logger *l = cached.load(std::memory_order_relaxed)) {
153 return l;
154 }
155 if (auto *logger = quill::Frontend::get_logger(
156 std::string(detail::kTracebackLoggerName))) {
157 cached.store(logger, std::memory_order_relaxed);
158 return logger;
159 }
160 return nullptr;
161}

172[[nodiscard]] inline quill::Logger *
173get_file(std::string_view name, std::string_view filename,
174 std::string_view pattern = "%(message)") {
175 // If the logger already exists, return it immediately.
176 // Quill strictly checks configuration matches on recreation and throws if
177 // PatternFormatterOptions differ, which crashes instances (like
178 // Dimer/Optimizers) that are created multiple times in loops (e.g., NEB).
179 if (auto *existing = quill::Frontend::get_logger(std::string(name))) {
180 return existing;
181 }
182 static std::mutex file_logger_mutex;
183 std::lock_guard<std::mutex> lock(file_logger_mutex);
184 if (auto *existing = quill::Frontend::get_logger(std::string(name))) {
185 return existing;
186 }
187
188 try {
189 // Durable absolute path: optimizers open "_lbfgs.log" etc. under CWD.
190 // Job integration fixtures chdir into /tmp/eon_test_job_* then delete that
191 // tree when the case ends, while Quill's process-lifetime FileSink still
192 // flushes — "fopen failed after 3 attempts". Keep logs under a stable
193 // temp subdir instead of the disposable workdir.
194 namespace fs = std::filesystem;
195 fs::path path{std::string(filename)};
196 if (!path.is_absolute()) {
197 static const fs::path log_dir = [] {
198 fs::path d = fs::temp_directory_path() / "eon_logs";
199 std::error_code ec;
200 fs::create_directories(d, ec);
201 return d;
202 }();
203 path = log_dir / path.filename();
204 }
205 quill::FileSinkConfig cfg;
206 cfg.set_open_mode('w');
207 auto file_sink = quill::Frontend::create_or_get_sink<quill::FileSink>(
208 path.string(), cfg);
209 return quill::Frontend::create_or_get_logger(
210 std::string(name), std::move(file_sink),
211 quill::PatternFormatterOptions{std::string(pattern)},
212 quill::ClockSourceType::System);
213 } catch (...) {
214 return get(); // Fallback to default combi logger if creation fails
215 }
216}

233struct Scoped {
234 [[no_unique_address]] quill::Logger *logger{nullptr};
235
236 Scoped() noexcept : logger(get()) {}
237
238 // Implicit conversion to Logger* for seamless use with QUILL_LOG_* macros
239 operator quill::Logger *() const noexcept { return logger; }
240
241 // Support QUILL_LOG_ macro pointer dereference
242 quill::Logger *operator->() const noexcept { return logger; }
243
244 // Allow dynamic re-assignment (e.g. switching to _traceback logger)
245 Scoped &operator=(quill::Logger *l) noexcept {
246 logger = l;
247 return *this;
248 }
249
250 // Explicit access if needed
251 [[nodiscard]] quill::Logger *ptr() const noexcept { return logger; }
252};
