117std::tuple<double, AtomMatrix> Potential::get_ef(const AtomMatrix &pos,
118 const VectorXi &atmnrs,
119 const Matrix3d &box) {
120 double energy{std::numeric_limits<double>::infinity()};
121 long nAtoms = static_cast<long>(pos.rows());
122 AtomMatrix forces{MatrixXd::Zero(nAtoms, 3)};
123 double var{0}; // no variance for true potentials
124 this->force(nAtoms, pos.data(), atmnrs.data(), forces.data(), &energy, &var,
125 box.data());
126 forceCallCounter++;
127 PotRegistry::get().on_force_call(ptype);
128
129 return std::make_tuple(energy, forces);
130}

153std::unique_ptr<rgpot::PotentialBase> make_expr_term(const std::string &raw,
154 const Parameters &params) {
155 const std::string name = lower_copy(raw);
156 if (name == "lj") {
157 return std::make_unique<rgpot::LJPot>(rgpot::LJConfig{});
158 }
159 if (name == "ljcluster") {
160 return std::make_unique<rgpot::LJClusterPot>(rgpot::LJClusterConfig{});
161 }
162 if (name == "morse" || name == "morse_pt") {
163 return std::make_unique<rgpot::MorsePot>(rgpot::MorseConfig{});
164 }
165 if (name == "zbl") {
166 return std::make_unique<rgpot::ZBLPot>(rgpot::ZBLConfig{
167 .cut_inner = params.zbl_options.cut_inner,
168 .cut_global = params.zbl_options.cut_global,
169 });
170 }
171#ifdef RGPOT_HAS_DFTD3
172 if (name == "d3" || name == "dftd3") {
173 return std::make_unique<rgpot::D3Pot>(rgpot::D3Config{
174 .damping = d3_damping_from_params(params),
175 .functional = params.dftd_options.functional,
176 .atm = params.dftd_options.atm,
177 });
178 }
179#endif
180#ifdef RGPOT_HAS_DFTD4
181 if (name == "d4" || name == "dftd4") {
182 return std::make_unique<rgpot::D4Pot>(rgpot::D4Config{
183 .functional = params.dftd_options.functional,
184 .charge = params.dftd_options.d4_charge,
185 .atm = params.dftd_options.atm,
186 });
187 }
188#endif
189 if (name == "mopac") {
190 return std::make_unique<rgpot::MOPACPot>(rgpot::MOPACPot::Config{
191 .charge = params.mopac_options.charge,
192 .spin = params.mopac_options.spin,
193 .model = params.mopac_options.model,
194 .engine_path = params.mopac_options.engine_path,
195 });
196 }
197 throw std::runtime_error(
198 "ExprPot unknown term '" + raw +
199 "' (lj, ljcluster, morse, zbl, d3/dftd3, d4/dftd4, mopac)");
200}

202std::vector<rgpot::ExprPot::Term> parse_expr_terms(const Parameters &params) {
203 std::vector<rgpot::ExprPot::Term> terms;
204 std::string buf = params.expr_options.terms;
205 std::string token;
206 auto flush = [&]() {
207 while (!token.empty() &&
208 std::isspace(static_cast<unsigned char>(token.front()))) {
209 token.erase(token.begin());
210 }
211 while (!token.empty() &&
212 std::isspace(static_cast<unsigned char>(token.back()))) {
213 token.pop_back();
214 }
215 if (!token.empty()) {
216 terms.emplace_back(token, make_expr_term(token, params));
217 token.clear();
218 }
219 };
220 for (char c : buf) {
221 if (c == ',') {
222 flush();
223 } else {
224 token.push_back(c);
225 }
226 }
227 flush();
228 if (terms.empty()) {
229 throw std::runtime_error(
230 "ExprPot needs [ExprPot] terms (comma-separated names used in "
231 "expression)");
232 }
233 return terms;
234}

246 const Parameters &params) {
