43GenericEngineLoader::GenericEngineLoader(const GenericEngineOptions &opt)
44 : m_tag(opt.tag.empty() ? opt.library : opt.tag) {
45 std::vector<std::string> paths;
46 if (!opt.engine_path.empty())
47 paths.push_back(opt.engine_path);
48 if (!opt.env_var.empty())
49 if (const char *e = std::getenv(opt.env_var.c_str()))
50 if (e && *e)
51 paths.emplace_back(e);
52 paths.push_back(opt.library);
53 auto add_dirs = [&](const char *env) {
54 if (!env)
55 return;
56 std::string s(env);
57 size_t i = 0;
58 while (i < s.size()) {
59 const auto j = s.find(':', i);
60 const auto dir = s.substr(i, j == std::string::npos ? j : j - i);
61 if (!dir.empty())
62 paths.push_back(dir + "/" + opt.library);
63 if (j == std::string::npos)
64 break;
65 i = j + 1;
66 }
67 };
68 add_dirs(std::getenv("EON_POTENTIALS_PATH"));
69
70 std::string last_dlerr;
71 for (const auto &p : paths) {
72 m_lib = open_lib(p.c_str());
73 if (m_lib)
74 break;
75#ifndef _WIN32
76 if (const char *e = dlerror())
77 last_dlerr = e;
78#endif
79 }
80 if (!m_lib) {
81 std::string msg =
82 "RGPOT(" + m_tag + "): " + opt.library + " not found (set " +
83 (opt.env_var.empty() ? std::string("engine_path") : opt.env_var) +
84 " or EON_POTENTIALS_PATH)";
85 if (!last_dlerr.empty())
86 msg += std::string("; last dlerror: ") + last_dlerr;
87 throw std::runtime_error(msg);
88 }
89
90 auto abi =
91 reinterpret_cast<int (*)()>(load_sym(m_lib, "rgpot_engine_abi_version"));
92 m_create =
93 reinterpret_cast<create_fn>(load_sym(m_lib, "rgpot_engine_create"));
94 m_destroy =
95 reinterpret_cast<destroy_fn>(load_sym(m_lib, "rgpot_engine_destroy"));
96 m_force = reinterpret_cast<force_fn>(load_sym(m_lib, "rgpot_engine_force"));
97 if (!abi || !m_create || !m_destroy || !m_force ||
98 abi() != RGPOT_ENGINE_ABI_VERSION) {
99 close_lib(m_lib);
100 m_lib = nullptr;
101 throw std::runtime_error("RGPOT(" + m_tag +
102 "): engine C ABI missing/mismatch");
103 }
104 char err[1024]{};
105 m_pot = m_create(opt.config.data(), opt.config.size(), err, sizeof err);
106 if (!m_pot) {
107 close_lib(m_lib);
108 m_lib = nullptr;
109 throw std::runtime_error("RGPOT(" + m_tag +
110 "): create failed: " + std::string(err));
111 }
112}

114GenericEngineLoader::~GenericEngineLoader() {
115 // Destroy the pot while the engine is still mapped. Do not dlclose the
116 // engine: torch static teardown after dlclose routinely SEGV on exit.
117 if (m_pot && m_destroy)
118 m_destroy(m_pot);
119 m_pot = nullptr;
120 m_lib = nullptr;
121}
