45std::string strip_nl(const std::string &s) {
46 std::string str(s);
47 while (!str.empty() && (str.back() == '\n' || str.back() == '\r'))
48 str.pop_back();
49 return str;
50}

52std::string canonical_generator_header(const std::string &header) {
53 auto stripped = strip_nl(header);
54 if (!stripped.empty()) {
55 return stripped;
56 }
57 return "Generated by eOn";
58}

60std::string ensure_extension(std::string filename, std::string_view ext) {
61 fs::path path(filename);
62 const auto name = path.filename().string();
63 const bool has_compound =
64 name.size() > ext.size() && (name.ends_with(std::string(ext) + ".gz") ||
65 name.ends_with(std::string(ext) + ".zst"));
66 if (!has_compound && path.extension() != ext) {
67 path += ext;
68 }
69 return path.string();
70}

78 "flat_row_major assumes AtomMatrix RowMajor Nx3 layout");

147std::vector<size_t> matter_order(const std::vector<readcon::Atom> &atoms) {
148 const size_t n = atoms.size();
149 std::vector<size_t> order(n);
150 std::iota(order.begin(), order.end(), size_t{0});
151 if (n < 2) {
152 return order;
153 }
154 const bool ascending =
155 std::is_sorted(atoms.begin(), atoms.end(),
156 [](const readcon::Atom &a, const readcon::Atom &b) {
157 return a.atom_id < b.atom_id;
158 }) &&
159 std::adjacent_find(atoms.begin(), atoms.end(),
160 [](const readcon::Atom &a, const readcon::Atom &b) {
161 return a.atom_id == b.atom_id;
162 }) == atoms.end();
163 if (ascending) {
164 return order;
165 }
166 std::vector<uint64_t> ids;
167 ids.reserve(n);
168 for (const auto &atom : atoms) {
169 ids.push_back(atom.atom_id);
170 }
171 std::vector<uint64_t> sorted(ids);
172 std::sort(sorted.begin(), sorted.end());
173 if (std::adjacent_find(sorted.begin(), sorted.end()) != sorted.end()) {
174 return order;
175 }
176 std::stable_sort(order.begin(), order.end(),
177 [&ids](size_t a, size_t b) { return ids[a] < ids[b]; });
178 return order;
179}

190std::optional<FileStamp> stamp_file(const fs::path &path) {
191 std::error_code ec;
192 const auto size = fs::file_size(path, ec);
193 if (ec) {
194 return std::nullopt;
195 }
196 const auto mtime = fs::last_write_time(path, ec);
197 if (ec) {
198 return std::nullopt;
199 }
200 return FileStamp{size, mtime};

241bool tail_is_ours(const std::string &key, const fs::path &path) {
242 const auto it = append_stamps().find(key);
243 if (it == append_stamps().end()) {
244 return false;
245 }
246 const auto stamp = stamp_file(path);
247 return stamp.has_value() && *stamp == it->second;

252std::optional<long> last_xyz_natoms(const fs::path &path) {
253 std::ifstream in(path);
254 if (!in) {
255 return std::nullopt;
256 }
257 std::optional<long> last;
258 std::string line;
259 while (std::getline(in, line)) {
260 if (line.empty()) {
261 continue;
262 }
263 long n = 0;
264 try {
265 n = std::stol(line);
266 } catch (const std::exception &) {
267 return std::nullopt;
268 }
269 if (n < 0) {
270 return std::nullopt;
271 }
272 if (!std::getline(in, line)) {
273 return std::nullopt;
274 }
275 for (long i = 0; i < n; ++i) {
276 if (!std::getline(in, line)) {
277 return std::nullopt;
278 }
279 }
280 last = n;
281 }
282 return last;
283}

286bool ends_with_newline(const fs::path &path) {
287 std::ifstream in(path, std::ios::binary | std::ios::ate);
288 if (!in) {
289 return true;
290 }
291 const auto len = static_cast<std::streamoff>(in.tellg());
292 if (len <= 0) {
293 return true;
294 }
295 in.seekg(len - 1);
296 char last = '\n';
297 if (!in.get(last)) {
298 return true;
299 }
300 return last == '\n';
301}

312eonc::io::IoStatus append_frames(const fs::path &path,
313 const std::vector<readcon::ConFrame> &frames,
314 uint8_t precision) {
315 static std::atomic<uint64_t> scratch_counter{0};
316 const auto scratch =
317 path.parent_path() /
318 std::format(".{}.eon-append-{}.tmp", path.filename().string(),
319 scratch_counter.fetch_add(1, std::memory_order_relaxed));
320
321 std::string bytes;
322 try {
323 {
324 readcon::ConFrameWriter writer(
325 scratch, readcon::ConFrameWriter::Compression::None, precision);
326 writer.extend(frames);
327 }
328 std::ifstream in(scratch, std::ios::binary | std::ios::ate);
329 if (!in) {
330 throw std::runtime_error("serialized frame not readable back");
331 }
332 const auto len = static_cast<std::streamoff>(in.tellg());
333 if (len <= 0) {
334 throw std::runtime_error("serialized frame is empty");
335 }
336 bytes.resize(static_cast<size_t>(len));
337 in.seekg(0);
338 if (!in.read(bytes.data(), static_cast<std::streamsize>(len))) {
339 throw std::runtime_error("short read of serialized frame");
340 }
341 } catch (const std::exception &e) {
342 std::error_code ec;
343 fs::remove(scratch, ec);
344 EONC_LOG_ERROR("Failed to serialize frame for {}: {}", path.string(),
345 e.what());
346 return eonc::io::IoStatus::WriteError;
347 }
348 std::error_code ec;
349 fs::remove(scratch, ec);
350
351 std::ofstream out(path, std::ios::binary | std::ios::app);
352 if (!out) {
353 EONC_LOG_ERROR("Failed to open {} for append", path.string());
354 return eonc::io::IoStatus::WriteError;
355 }
356 // A frame header must start its own line. eOn's own frames end in a
357 // newline; a hand-written or foreign tail may not.
358 if (!ends_with_newline(path)) {
359 out.put('\n');
360 }
361 out.write(bytes.data(), static_cast<std::streamsize>(bytes.size()));
362 out.flush();
363 if (!out) {
364 EONC_LOG_ERROR("Failed to append to {}", path.string());
365 return eonc::io::IoStatus::WriteError;
366 }
367 return eonc::io::IoStatus::Ok;
368}

372readcon::ConFrameBuilder seed_builder(eonc::Matter &m,
373 const std::array<std::string, 2> &prebox,
374 const std::array<std::string, 2> &postbox,
375 const std::vector<uint64_t> &atom_ids) {
376 auto [lengths, angles_deg] = eonc::io::cell_to_lengths_angles(m);
377 readcon::ConFrameBuilder builder(
378 {lengths[0], lengths[1], lengths[2]},
379 {angles_deg[0], angles_deg[1], angles_deg[2]}, prebox, postbox);
380
381 const long n = m.numberOfAtoms();
382 if (atom_ids.size() != static_cast<size_t>(n)) {
383 throw std::invalid_argument("seed_builder: atom_ids size mismatch");
384 }
385 for (long i = 0; i < n; ++i) {
386 const auto mask = m.getFixedMask(i);
387 builder.add_atom(symbol_for_z(m.getAtomicNr(i)), 0.0, 0.0, 0.0, mask,
388 atom_ids[static_cast<size_t>(i)], m.getMass(i));
389 }
390 return builder;
391}

393bool should_write_forces(const Matter &m,
394 const eonc::io::ConFrameMetadata *metadata) {
395 // Per-write metadata wins so two callers can disagree without a race on
396 // the process-wide flag. Parameters next, so pyeonclient setting the
397 // field on the bound Parameters object is enough. The atomic global is
398 // last (INI parse and the explicit setter).
399 if (metadata && metadata->write_con_forces.has_value()) {
400 return *metadata->write_con_forces;
401 }
402 if (m.getWriteConForces()) {
403 return true;
404 }
405 return eonc::io::write_con_forces();
406}

408void apply_geometry(readcon::ConFrameBuilder &builder, Matter &m,
409 bool with_velocities,
410 const eonc::io::ConFrameMetadata *metadata) {
411 const long n = m.numberOfAtoms();
412 if (n <= 0) {
413 return;
414 }
415
416 // Prefer bulk flat setters (declared sections, no pointer-lifetime hazards
417 // under ASAN). AtomMatrix is RowMajor Nx3 matching the flat layout.
418 builder.set_positions_from_flat(flat_row_major(m.getPositions()));
419
420 // getForcesRaw() may trigger pot evaluation if recomputePotential is set;
421 // only enter when the pot is already clean so we serialize cached forces.
422 // Gated: force sections are opt-in ([Main] write_con_forces) because
423 // ASE-class readers reject frames that carry them.
424 if (should_write_forces(m, metadata) && !m.needsForceUpdate()) {
425 builder.set_forces_from_flat(flat_row_major(m.getForcesRaw()));
426 }
427
428 if (with_velocities) {
429 const AtomMatrix vel = m.getVelocities();
430 for (long i = 0; i < n; ++i) {
431 builder.set_atom_velocity(static_cast<size_t>(i),
432 {vel(i, 0), vel(i, 1), vel(i, 2)});
433 }
434 }
435}

437void collect_ids_headers(Matter &m, std::vector<uint64_t> &atom_ids,
438 std::array<std::string, 2> &prebox,
439 std::array<std::string, 2> &postbox) {
440 const long n = m.numberOfAtoms();
441 atom_ids.resize(static_cast<size_t>(n));
442 for (long i = 0; i < n; ++i) {
443 atom_ids[static_cast<size_t>(i)] = static_cast<uint64_t>(m.getAtomIndex(i));
444 }
445 const auto &hdr = m.getHeaderCon();
446 prebox = {canonical_generator_header(hdr[0]), strip_nl(hdr[1])};
447 postbox = {strip_nl(hdr[3]), strip_nl(hdr[4])};
448}

450readcon::ConFrame frame_from_matter(Matter &m,
451 const eonc::io::ConFrameMetadata *metadata,
452 bool with_velocities) {
453 std::vector<uint64_t> atom_ids;
454 std::array<std::string, 2> prebox;
455 std::array<std::string, 2> postbox;
456 collect_ids_headers(m, atom_ids, prebox, postbox);
457
458 auto builder = seed_builder(m, prebox, postbox, atom_ids);
459
460 eonc::io::ConFrameMetadata auto_meta;
461 const eonc::io::ConFrameMetadata *meta_ptr = metadata;
462 if (!m.needsForceUpdate()) {
463 if (metadata == nullptr) {
464 auto_meta.energy = m.getPotentialEnergy();
465 meta_ptr = &auto_meta;
466 } else if (!metadata->energy) {
467 auto_meta = *metadata;
468 auto_meta.energy = m.getPotentialEnergy();
469 meta_ptr = &auto_meta;
470 }
471 }
472 apply_frame_metadata(builder, meta_ptr);
473 apply_geometry(builder, m, with_velocities, meta_ptr);
474 return builder.build();
475}

516std::pair<std::array<double, 3>, std::array<double, 3>>
517cell_to_lengths_angles(const Matter &m) {
518 const Matrix3d cell = m.getCell();
519 std::array<double, 3> lengths;
520 lengths[0] = cell.row(0).norm();
521 lengths[1] = cell.row(1).norm();
522 lengths[2] = cell.row(2).norm();
523 // CON header line 4 is alpha beta gamma:
524 // alpha = angle(b,c), beta = angle(a,c), gamma = angle(a,b).
525 std::array<double, 3> angles;
526 angles[0] = eonc::safemath::safe_acos(eonc::safemath::safe_div(
527 cell.row(1).dot(cell.row(2)), lengths[1] * lengths[2])) *
528 180.0 / eonc::helpers::pi;
529 angles[1] = eonc::safemath::safe_acos(eonc::safemath::safe_div(
530 cell.row(0).dot(cell.row(2)), lengths[0] * lengths[2])) *
531 180.0 / eonc::helpers::pi;
532 angles[2] = eonc::safemath::safe_acos(eonc::safemath::safe_div(
533 cell.row(0).dot(cell.row(1)), lengths[0] * lengths[1])) *
534 180.0 / eonc::helpers::pi;
535 return {lengths, angles};
536}

543IoStatus matter2con(Matter &m, std::string filename, bool append,
544 const ConFrameMetadata *metadata) {
545 filename = ensure_extension(std::move(filename), ".con");
546
547 m.applyPeriodicBoundaryIfEnabled();
548
549 const fs::path path(filename);
550 const auto compression =
551 readcon::ConFrameWriter::compression_from_extension(path);
552 const bool streamable =
553 compression == readcon::ConFrameWriter::Compression::None;
554
555 // The lock covers the stamp table and the read-then-extend window on one
556 // path. Nothing under it evaluates a potential: frame_from_matter reads
557 // cached energy and forces only while the pot is clean.
558 const std::lock_guard<std::mutex> guard(append_mutex());
559 const auto key = append_key(path);
560 const bool exists = fs::exists(path);
561 const bool concatenate = append && exists && streamable;
562
563 // ConFrame is move-only (no default ctor), so frames carries the new frame
564 // either way. A gzip or zstd target cannot be concatenated: readcon-core
565 // reads a single gzip member, so appended members would be invisible
566 // on read-back, and the streaming writers cannot be flushed frame by frame.
567 // Those targets keep the whole-file rewrite, which stays O(N) per append.
568 std::vector<readcon::ConFrame> frames;
569 if (append && exists && !streamable) {
570 try {
571 // Prefer read_all_frames (single ownership hand-off) over the iterator
572 // path for append rewrite; simpler lifetime for ASAN/CI envs.
573 frames = readcon::read_all_frames(path);
574 } catch (const std::exception &e) {
575 EONC_LOG_ERROR("Failed to append to {}: {}", filename, e.what());
576 return IoStatus::AppendError;
577 }
578 } else if (concatenate && !tail_is_ours(key, path)) {
579 // Refuse to extend a file eOn cannot parse, matching the rewrite path:
580 // the target keeps its bytes and the caller sees AppendError. Frames this
581 // process wrote and nobody touched since need no such check.
582 try {
583 [[maybe_unused]] const auto existing = readcon::read_all_frames(path);
584 } catch (const std::exception &e) {
585 EONC_LOG_ERROR("Failed to append to {}: {}", filename, e.what());
586 return IoStatus::AppendError;
587 }
588 }
589 try {
590 frames.push_back(frame_from_matter(m, metadata, /*with_velocities=*/false));
591 } catch (const std::exception &e) {
592 EONC_LOG_ERROR("Failed to build frame for {}: {}", filename, e.what());
593 return IoStatus::InvalidArgument;
594 }
595
596 const auto status = concatenate ? append_frames(path, frames, kConPrecision)
597 : write_frames(path, frames, kConPrecision);
598 if (io_ok(status)) {
599 remember_stamp(key, path);
600 if (!concatenate) {
601 mirror_con_corpus(path.string());
602 }
603 } else {
604 append_stamps().erase(key);
605 }
606 return status;
607}

624IoStatus con2matter(Matter &m, const readcon::ConFrame &frame,
625 ConFrameMetadata *out_metadata) {
626 try {
627 const auto &atoms = frame.atoms();
628 const auto &lengths = frame.cell();
629 const auto &angles_deg = frame.angles();
630 const auto &prebox = frame.prebox_header();
631 const auto &postbox = frame.postbox_header();
632
633 m.headerCon[0] = prebox[0] + "\n";
634 m.headerCon[1] = prebox[1] + "\n";
635 m.headerCon[3] = postbox[0] + "\n";
636 m.headerCon[4] = postbox[1] + "\n";
637
638 double angles[3] = {angles_deg[0], angles_deg[1], angles_deg[2]};
639 if (angles[0] == 90.0 && angles[1] == 90.0 && angles[2] == 90.0) {
640 Matrix3d cell = Matrix3d::Zero();
641 cell(0, 0) = lengths[0];
642 cell(1, 1) = lengths[1];
643 cell(2, 2) = lengths[2];
644 m.setCell(cell);
645 } else {
646 angles[0] *= eonc::helpers::pi / 180.0;
647 angles[1] *= eonc::helpers::pi / 180.0;
648 angles[2] *= eonc::helpers::pi / 180.0;
649 const double alpha = angles[0];
650 const double beta = angles[1];
651 const double gamma = angles[2];
652
653 Matrix3d cell = Matrix3d::Zero();
654 cell(0, 0) = 1.0;
655 cell(1, 0) = cos(gamma);
656 cell(1, 1) = sin(gamma);
657 cell(2, 0) = cos(beta);
658 cell(2, 1) = (cos(alpha) - cell(1, 0) * cell(2, 0)) / cell(1, 1);
659 cell(2, 2) = eonc::safemath::safe_sqrt(1.0 - pow(cell(2, 0), 2) -
660 pow(cell(2, 1), 2));
661
662 cell(0, 0) *= lengths[0];
663 cell(1, 0) *= lengths[1];
664 cell(1, 1) *= lengths[1];
665 cell(2, 0) *= lengths[2];
666 cell(2, 1) *= lengths[2];
667 cell(2, 2) *= lengths[2];
668 m.setCell(cell);
669 }
670 m.headerCon[2] =
671 std::format("{} {} {}\n", angles_deg[0], angles_deg[1], angles_deg[2]);
672
673 const auto n = static_cast<Eigen::Index>(atoms.size());
674 m.resize(static_cast<long>(atoms.size()));
675
676 // Undo the species grouping the .con format imposes, so an index into
677 // Matter addresses the same atom as the matching row of mode.dat.
678 const std::vector<size_t> order = matter_order(atoms);
679 std::vector<long> file_to_matter(static_cast<size_t>(n));
680 for (Eigen::Index i = 0; i < n; ++i) {
681 file_to_matter[order[static_cast<size_t>(i)]] = static_cast<long>(i);
682 }
683 m.setFileToMatter(std::move(file_to_matter));
684
685 AtomMatrix positions = AtomMatrix::Zero(n, 3);
686 AtomMatrix forces = AtomMatrix::Zero(n, 3);
687 AtomMatrix velocities = AtomMatrix::Zero(n, 3);
688 VectorXd masses = VectorXd::Zero(n);
689 VectorXi atomic_nrs = VectorXi::Zero(n);
690 bool any_force = false;
691 bool any_velocity = false;
692
693 for (Eigen::Index i = 0; i < n; ++i) {
694 const auto &atom = atoms[order[static_cast<size_t>(i)]];
695 positions(i, 0) = atom.x;
696 positions(i, 1) = atom.y;
697 positions(i, 2) = atom.z;
698 masses(i) = atom.mass;
699 atomic_nrs(i) = static_cast<int>(atom.atomic_number);
700 const auto fixed = atom.fixed_mask();
701 m.setFixedMask(static_cast<long>(i), {fixed[0], fixed[1], fixed[2]});
702 m.setAtomIndex(static_cast<long>(i),
703 static_cast<std::int64_t>(atom.atom_id));
704
705 if (auto vel = atom.velocity()) {
706 any_velocity = true;
707 velocities(i, 0) = (*vel)[0];
708 velocities(i, 1) = (*vel)[1];
709 velocities(i, 2) = (*vel)[2];
710 }
711 if (auto force = atom.force()) {
712 any_force = true;
713 forces(i, 0) = (*force)[0];
714 forces(i, 1) = (*force)[1];
715 forces(i, 2) = (*force)[2];
716 }
717 }
718
719 m.setMasses(masses);
720 m.setAtomicNrs(atomic_nrs);
721 m.setPositions(positions);
722
723 if (any_velocity || frame.has_velocities()) {
724 m.setVelocities(velocities);
725 }
726
727 const auto meta = metadata_from_frame(frame);
728 if (out_metadata != nullptr) {
729 *out_metadata = meta;
730 }
731
732 // Trust file energy+forces only when both are present. Energy-only must not
733 // mark the pot clean with a zero force matrix (optimizer footgun). Prefer
734 // writing raw forces (friend) so fixed-atom components survive RT; then
735 // mark clean without setComputedPotential's net-force adjustment on zeros.
736 const bool has_force_section = any_force || frame.has_forces();
737 if (meta.energy && has_force_section) {
738 m.forces = forces;
739 m.potentialEnergy = *meta.energy;
740 m.energyVariance = 0.0;
741 m.recomputePotential = false;
742 m.recomputeMaskedForces = true;
743 } else if (has_force_section) {
744 m.forces = forces;
745 m.recomputePotential = true;
746 m.recomputeMaskedForces = true;
747 } else {
748 // Classic geometry-only files: always recompute pot (main-era behavior).
749 m.recomputePotential = true;
750 }
751
752 // setPositions already applied PBC when enabled; no second wrap here.
753 return IoStatus::Ok;
754 } catch (const std::exception &e) {
755 EONC_LOG_ERROR("Failed to convert frame to matter: {}", e.what());
756 return IoStatus::ReadError;
757 }
758}

795IoStatus matter2xyz(Matter &m, std::string filename, bool append) {
796 filename = ensure_extension(std::move(filename), ".xyz");
797
798 m.applyPeriodicBoundaryIfEnabled();
799 const long n = m.numberOfAtoms();
800
801 if (append && fs::exists(filename)) {
802 std::error_code ec;
803 const auto sz = fs::file_size(filename, ec);
804 if (!ec && sz > 0) {
805 const auto prev = last_xyz_natoms(filename);
806 if (!prev) {
807 EONC_LOG_ERROR("matter2xyz: cannot parse existing {}", filename);
808 return IoStatus::AppendError;
809 }
810 if (*prev != n) {
811 EONC_LOG_ERROR(
812 "matter2xyz: append atom count {} != last frame {} in {}", n, *prev,
813 filename);
814 return IoStatus::InvalidArgument;
815 }
816 }
817 }
818
819 std::ofstream out;
820 out.open(filename,
821 append ? (std::ios::out | std::ios::app | std::ios::binary)
822 : (std::ios::out | std::ios::trunc | std::ios::binary));
823 if (!out) {
824 EONC_LOG_ERROR("matter2xyz: cannot open {}", filename);
825 return IoStatus::OpenError;
826 }
827
828 // A frame header must start its own line. eOn's own frames end in a
829 // newline; a hand-written or foreign tail may not.
830 if (append && !ends_with_newline(filename)) {
831 out.put('\n');
832 }
833
834 const Matrix3d cell = m.getCell();
835 out << std::format(
836 "{}\nLattice=\"{:.17g} {:.17g} {:.17g} {:.17g} {:.17g} {:.17g} "
837 "{:.17g} {:.17g} {:.17g}\" Properties=species:S:1:pos:R:3 Generated "
838 "by eOn\n",
839 n, cell(0, 0), cell(0, 1), cell(0, 2), cell(1, 0), cell(1, 1), cell(1, 2),
840 cell(2, 0), cell(2, 1), cell(2, 2));
841 const AtomMatrix pos = m.getPositions();
842 for (long i = 0; i < n; ++i) {
843 out << std::format("{}\t{:.17g}\t{:.17g}\t{:.17g}\n",
844 symbol_for_z(m.getAtomicNr(i)), pos(i, 0), pos(i, 1),
845 pos(i, 2));
846 }
847 // Close before testing: the destructor's flush is where a full disk or a
848 // short write surfaces, and by then the state is gone.
849 out.close();
850 if (!out) {
851 EONC_LOG_ERROR("matter2xyz: failed to write {}", filename);
852 return IoStatus::WriteError;
853 }
854 return IoStatus::Ok;
855}

857IoStatus writeTibble(Matter &m, std::string fname) {
858 // Debug table, not a structure format. getForces()/getPotentialEnergy()
859 // run computePotential() on a dirty pot, which would charge a dump to
860 // the force-call count in results.dat. Cached values only; the header
861 // drops the columns it cannot fill. atmID is the CON column-5 id.
862 const bool have_forces = !m.needsForceUpdate();
863 if (!have_forces) {
864 EONC_LOG_WARNING(
865 "writeTibble: pot is dirty, {} omits the force and energy columns",
866 fname);
867 }
868 const AtomMatrix pos = m.getPositions();
869 std::ofstream out(fname);
870 if (!out) {
871 EONC_LOG_ERROR("writeTibble: cannot open {}", fname);
872 return IoStatus::OpenError;
873 }
874 out << (have_forces ? "x y z fx fy fz energy mass symbol atmID fixed\n"
875 : "x y z mass symbol atmID fixed\n");
876 const AtomMatrix fSys = have_forces ? m.getForces() : AtomMatrix();
877 const double eSys = have_forces ? m.getPotentialEnergy() : 0.0;
878 for (long idx = 0; idx < m.numberOfAtoms(); ++idx) {
879 out << std::format("{} {} {}", pos(idx, 0), pos(idx, 1), pos(idx, 2));
880 if (have_forces) {
881 out << std::format(" {} {} {} {}", fSys(idx, 0), fSys(idx, 1),
882 fSys(idx, 2), eSys);
883 }
884 const auto mask = m.getFixedMask(idx);
885 const int fixed_bits =
886 (mask[0] ? 1 : 0) | (mask[1] ? 2 : 0) | (mask[2] ? 4 : 0);
887 out << std::format(" {} {} {} {}\n", m.getMass(idx),
888 symbol_for_z(m.getAtomicNr(idx)), m.getAtomIndex(idx),
889 fixed_bits);
890 }
891 out.close();
892 if (!out) {
893 EONC_LOG_ERROR("writeTibble: failed to write {}", fname);
894 return IoStatus::WriteError;
895 }
896 return IoStatus::Ok;
897}

900buildNebPathFrames(const std::vector<std::shared_ptr<Matter>> &path,
901 const std::vector<ConFrameMetadata> &metadata_per_image) {
902 std::vector<readcon::ConFrame> frames;
903 if (path.empty() || path.size() != metadata_per_image.size()) {
904 EONC_LOG_ERROR(
905 "buildNebPathFrames: path/metadata size mismatch (path={}, meta={})",
906 path.size(), metadata_per_image.size());
907 return frames;
908 }
909 for (const auto &img : path) {
910 if (!img) {
911 EONC_LOG_ERROR("buildNebPathFrames: null Matter in path");
912 return {};
913 }
914 }
915
916 Matter &template_m = *path.front();
917 template_m.applyPeriodicBoundaryIfEnabled();
918
919 std::vector<uint64_t> atom_ids;
920 std::array<std::string, 2> prebox;
921 std::array<std::string, 2> postbox;
922 collect_ids_headers(template_m, atom_ids, prebox, postbox);
923
924 frames.reserve(path.size());
925 try {
926 auto seed = seed_builder(template_m, prebox, postbox, atom_ids);
927 for (size_t i = 0; i < path.size(); ++i) {
928 Matter &img = *path[i];
929 img.applyPeriodicBoundaryIfEnabled();
930 if (img.numberOfAtoms() != template_m.numberOfAtoms()) {
931 EONC_LOG_ERROR(
932 "buildNebPathFrames: image {} atom count {} != template {}", i,
933 img.numberOfAtoms(), template_m.numberOfAtoms());
934 return {};
935 }
936 auto builder = seed.clone();
937 apply_frame_metadata(builder, &metadata_per_image[i]);
938 apply_geometry(builder, img, /*with_velocities=*/false,
939 &metadata_per_image[i]);
940 frames.push_back(builder.build());
941 }
942 } catch (const std::exception &e) {
943 EONC_LOG_ERROR("buildNebPathFrames failed: {}", e.what());
944 return {};
945 }
946 return frames;
947}
