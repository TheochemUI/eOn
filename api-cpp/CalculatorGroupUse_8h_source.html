28struct CalculatorGroupUse {
29 std::vector<double> busy;
30 std::vector<double> systems;
31 double wall{0.0};
32 double singleWall{0.0};
33 long batches{0};
34 long singles{0};
35
36 [[nodiscard]] std::size_t groups() const noexcept { return busy.size(); }
37
38 [[nodiscard]] double meanBusy() const noexcept {
39 if (busy.empty()) {
40 return 0.0;
41 }
42 return std::accumulate(busy.begin(), busy.end(), 0.0) /
43 static_cast<double>(busy.size());
44 }
45
46 [[nodiscard]] double maxBusy() const noexcept {
47 return busy.empty() ? 0.0 : *std::max_element(busy.begin(), busy.end());
48 }
49
51 [[nodiscard]] double loadBalance() const noexcept {
52 const double m = maxBusy();
53 return m > 0.0 ? meanBusy() / m : 1.0;
54 }
55
59 [[nodiscard]] double communicationEfficiency() const noexcept {
60 return wall > 0.0 ? std::min(1.0, maxBusy() / wall) : 1.0;
61 }
62
63 [[nodiscard]] double parallelEfficiency() const noexcept {
64 return loadBalance() * communicationEfficiency();
65 }
66
68 [[nodiscard]] double idleFraction(std::size_t group) const noexcept {
69 if (group >= busy.size() || wall <= 0.0) {
70 return 0.0;
71 }
72 return std::clamp(1.0 - busy[group] / wall, 0.0, 1.0);
73 }
74
75 [[nodiscard]] std::string table() const {
76 std::string out = std::format(
77 "calculator groups: {} groups, {} batches and {} single requests, "
78 "{:.1f} s grouped wall ({:.1f} s single)\n",
79 groups(), batches, singles, wall, singleWall);
80 out += std::format("{:>6} {:>8} {:>10} {:>6}\n", "group", "systems",
81 "busy_s", "idle");
82 for (std::size_t g = 0; g < busy.size(); ++g) {
83 const double n = g < systems.size() ? systems[g] : 0.0;
84 out += std::format("{:>6} {:>8.0f} {:>10.2f} {:>5.1f}%\n", g, n, busy[g],
85 100.0 * idleFraction(g));
86 }
87 out += std::format("load balance {:.3f}, communication efficiency {:.3f}, "
88 "parallel efficiency {:.3f}\n",
89 loadBalance(), communicationEfficiency(),
90 parallelEfficiency());
91 return out;
92 }
93};
