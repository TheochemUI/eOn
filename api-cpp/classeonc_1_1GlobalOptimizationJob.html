377 vat(iat, 1) -= vtot[1];

383 throw std::invalid_argument("GlobalOptimizationJob::velopt: no free atoms");
384 }

386 double kB = params.constants().kB;
387 double kinT = (2.0 * kinE / nFreeCoords / kB);
388 if (!(kinT > 0.0)) {
389 throw std::runtime_error(
390 "GlobalOptimizationJob::velopt: zero kinetic temperature");
391 }
