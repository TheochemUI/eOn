◆ eon_relax_step()

int eon_relax_step ( EonRelaxEngine * eng,
eon_relax_band_t * band,
eon_relax_surface_fn surface,
void * user,
eon_relax_outcome_t * out )

One band optimizer step (kind=NEB only).

The first call on a fresh or reset engine initializes the band state from the caller's positions and captures the baseline force; every call syncs band->positions in (the host may have moved images between steps), assembles NEB forces on the host surface, takes ONE optimizer step, and writes the stepped INTERIOR positions back (endpoints are the caller's and are never rewritten, so the caller's frame survives eOn's own cell wrapping). Climbing-image activation follows the engine's own trigger rule; saddle POLICY (MMF bursts, resampling, acquisition) stays with the host. outcome.status is RUNNING until convergenceForce() reaches the configured tolerance (GOOD), host variance exceeds the GP-surrogate threshold (MAX_UNCERTAINTY), or neb.max_iterations is hit (BAD_MAX_ITERATIONS). surface and user must not change across steps without an eon_relax_reset.

547 eng->step_opt = eonc::helpers::create::mkOptim(
548 eng->step_objf, eng->params.neb_options().opt_method, eng->params);
549 eng->step_surface = surface;
550 eng->step_user = user;
551 eng->step_baseline = eng->step_neb->convergenceForce();

553 } else {

555 // reparameterization); resync and mark the band dirty.
556 for (int64_t i = 0; i < band->n_images; ++i) {
557 fill_matter(*eng->step_neb->path[static_cast<size_t>(i)], band,
558 static_cast<long>(i), eng->epoch);

560 eng->step_neb->movedAfterForceCall = true;

565 const double convForce = neb.convergenceForce();

567 const bool ci_active =
568 ci_opt.enabled &&
569 (convForce < eng->step_baseline * ci_opt.trigger_factor ||
570 convForce < ci_opt.trigger_force);
571 neb.setCIEnabled(ci_active);
572
573 const double force_tol = eng->params.neb_options().force_tolerance;
574 const long max_iter = eng->params.neb_options().max_iterations;
575 int status = EON_RELAX_NEB_RUNNING;
576 if (eng->step_objf && eng->step_objf->isUncertain()) {
577 status = EON_RELAX_NEB_MAX_UNCERTAINTY;
578 } else if (convForce <= force_tol) {
579 status = EON_RELAX_NEB_GOOD;
580 } else if (eng->step_iteration >= max_iter) {
581 status = EON_RELAX_NEB_BAD_MAX_ITERATIONS;
582 } else {
583 eng->step_opt->step(eng->params.optimizer_options().max_move);
584 ++eng->step_iteration;
585 neb.updateForces();
586 if (eng->step_objf && eng->step_objf->isUncertain()) {
587 status = EON_RELAX_NEB_MAX_UNCERTAINTY;
588 } else if (neb.convergenceForce() <= force_tol) {
589 status = EON_RELAX_NEB_GOOD;
590 } else if (eng->step_iteration >= max_iter) {
591 status = EON_RELAX_NEB_BAD_MAX_ITERATIONS;
592 }
593 }
594
595 // Interior images only: eOn never moves the endpoints, and Matter
596 // wraps positions into the cell, so writing them back would hand
597 // the caller a different frame than the one it supplied.
598 for (int64_t i = 1; i + 1 < band->n_images; ++i) {
599 write_image(band->positions + i * 3 * band->n_atoms,
600 *neb.path[static_cast<size_t>(i)],
601 static_cast<long>(band->n_atoms));
602 }
603 out->status = status;
604 out->iterations = eng->step_iteration;
605 out->climbing_image = neb.climbingImage;
606 out->max_force = neb.convergenceForce();
607 out->surface_epoch = eng->step_pot->surfaceEpoch();
608 return stamp_rc(eng, EON_RELAX_OK);
609 } catch (const eonc::SurfaceRecoverable &rec) {
610 out->status = -1;
611 clear_stepper(eng);
612 return stamp_rc(eng, rec.rc);
613 } catch (const std::exception &) {
614 out->status = -1;
615 eng->unusable = true;
616 return stamp_rc(eng, EON_RELAX_SURFACE_FATAL);
617 }
