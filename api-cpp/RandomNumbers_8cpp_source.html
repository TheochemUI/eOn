80double eonc::rng::gaussRandom(double avg, double std) {
81 double r = 2, v1, v2, l, result;
82 while (r >= 1.0 || r < 1e-300) {
83 v1 = 2.0 * randomDouble() - 1.0;
84 v2 = 2.0 * randomDouble() - 1.0;
85 r = v1 * v1 + v2 * v2;
86 }
87 l = v1 * sqrt(-2.0 * ::log(r) / r);
88 result = avg + std * l;
89 return (result);
90}
