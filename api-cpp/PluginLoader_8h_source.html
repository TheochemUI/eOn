43// ---------------------------------------------------------------------------
44class IPluginLoader {
45public:
49 virtual void add_config_paths(const std::string &colon_paths) = 0;
50
54 [[nodiscard]] virtual bool lib_present(const char *lib_base) const = 0;
55
57 [[nodiscard]] virtual const std::vector<std::string> &
58 search_paths() const noexcept = 0;
59
61 [[noreturn]] virtual void throw_not_found(const char *lib_base,
62 const char *description) const = 0;
63
68 template <typename Fn>
69 Fn load_sym(const char *lib_base, const char *sym_name) {
70 dynlib::Handle h = open_lib(lib_base);
71 if (!h)
72 return nullptr;
73 return dynlib::loadSym<Fn>(h, sym_name);
74 }
75
76 virtual ~IPluginLoader() = default;
77
78 IPluginLoader(const IPluginLoader &) = delete;
79 IPluginLoader &operator=(const IPluginLoader &) = delete;
80
81protected:
82 IPluginLoader() = default;
83 virtual dynlib::Handle open_lib(const char *lib_base) = 0;
84};
