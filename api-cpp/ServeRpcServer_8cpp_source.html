62void fillServerCapabilities(::Capabilities::Builder caps) {
63 caps.setProtocolFamily(eonc::rpc::kProtocolFamily);
64 caps.setProtocolMajor(eonc::rpc::kProtocolMajor);
65 caps.setProtocolMinor(eonc::rpc::kProtocolMinor);
66 caps.setSchemaId(eonc::rpc::kSchemaId);
67 caps.setBridgeAbiMajor(eonc::rpc::kBridgeAbiMajor);
68 caps.setBridgeAbiMinor(eonc::rpc::kBridgeAbiMinor);
69 caps.setBridgeLayout(eonc::rpc::kBridgeLayout);
70 caps.setDlpackMajor(eonc::rpc::kDlpackMajor);
71 caps.setDlpackMinor(eonc::rpc::kDlpackMinor);
72 caps.setBridgeFeatures(eonc::rpc::kBridgeFeatures);
73 caps.setBackendName(eonc::rpc::kBackendName);
74 caps.setBackendVersion(EON_BUILD_VERSION);
75 caps.setAvailable(true);
76 caps.setBuildVersion(EON_BUILD_VERSION);
77 caps.setBuildRevision(EON_BUILD_REVISION);
78 auto ops = caps.initOperations(2);
79 ops.set(0, ::Capabilities::Operation::ENERGY);
80 ops.set(1, ::Capabilities::Operation::FORCES);
81}

88RpcCapabilitiesView readOne(kj::AsyncIoContext &io,
89 capnp::Capability::Client bootstrap) {
90 auto pipe = io.provider->newTwoWayPipe();
91 capnp::TwoPartyClient serverSide(*pipe.ends[0], kj::mv(bootstrap),
92 capnp::rpc::twoparty::Side::SERVER);
93 capnp::TwoPartyClient clientSide(*pipe.ends[1]);
94 auto pot = clientSide.bootstrap().castAs<Potential>();
95 auto response = pot.getCapabilitiesRequest().send().wait(io.waitScope);
96 auto caps = response.getCapabilities();
97
98 RpcCapabilitiesView view;
99 view.protocolFamily = caps.getProtocolFamily().cStr();
100 view.protocolMajor = caps.getProtocolMajor();
101 view.protocolMinor = caps.getProtocolMinor();
102 view.schemaId = caps.getSchemaId().cStr();
103 view.bridgeAbiMajor = caps.getBridgeAbiMajor();
104 view.bridgeAbiMinor = caps.getBridgeAbiMinor();
105 view.bridgeLayout = caps.getBridgeLayout();
106 view.dlpackMajor = caps.getDlpackMajor();
107 view.dlpackMinor = caps.getDlpackMinor();
108 view.bridgeFeatures = caps.getBridgeFeatures();
109 view.backendName = caps.getBackendName().cStr();
110 view.available = caps.getAvailable();
111 view.buildVersion = caps.getBuildVersion().cStr();
112 view.buildRevision = caps.getBuildRevision().cStr();
113 for (auto op : caps.getOperations()) {
114 if (op == ::Capabilities::Operation::ENERGY) {
115 view.servesEnergy = true;
116 }
117 if (op == ::Capabilities::Operation::FORCES) {
118 view.servesForces = true;
119 }
120 }
121 return view;
122}

135class CallbackPotImpl final : public Potential::Server {
136public:
137 explicit CallbackPotImpl(ForceCallback cb)
138 : m_callback(std::move(cb)) {}
139
140 kj::Promise<void> getCapabilities(GetCapabilitiesContext context) override {
141 fillServerCapabilities(context.getResults().initCapabilities());
142 return kj::READY_NOW;
143 }
144
145 kj::Promise<void> calculate(CalculateContext context) override {
146 auto fip = context.getParams().getFip();
147 const size_t numAtoms = fip.getPos().size() / 3;
148
149 KJ_REQUIRE(fip.getAtmnrs().size() == numAtoms, "AtomNumbers size mismatch");
150
151 // Extract flat arrays from capnp
152 std::vector<double> positions(numAtoms * 3);
153 auto capnpPos = fip.getPos();
154 for (size_t i = 0; i < numAtoms * 3; ++i) {
155 positions[i] = capnpPos[i];
156 }
157
158 std::vector<int> atomicNrs(numAtoms);
159 auto capnpAtmnrs = fip.getAtmnrs();
160 for (size_t i = 0; i < numAtoms; ++i) {
161 atomicNrs[i] = capnpAtmnrs[i];
162 }
163
164 double box[9] = {};
165 auto capnpBox = fip.getBox();
166 for (size_t i = 0; i < 9 && i < capnpBox.size(); ++i) {
167 box[i] = capnpBox[i];
168 }
169
170 // Call the force callback
171 std::vector<double> forces(numAtoms * 3, 0.0);
172 double energy = 0.0;
173 m_callback(static_cast<long>(numAtoms), positions.data(), atomicNrs.data(),
174 forces.data(), &energy, box);
175
176 // Serialize result back to capnp
177 auto result = context.getResults();
178 auto pres = result.initResult();
179 pres.setEnergy(energy);
180
181 auto forcesList = pres.initForces(numAtoms * 3);
182 for (size_t i = 0; i < numAtoms * 3; ++i) {
183 forcesList.set(i, forces[i]);
184 }
185
186 return kj::READY_NOW;
187 }
188
189private:
190 ForceCallback m_callback;
191};

217class PooledCallbackPotImpl final : public Potential::Server {
218public:
219 explicit PooledCallbackPotImpl(std::vector<ForceCallback> pool)
220 : m_pool(std::move(pool)),
221 m_mutexes(m_pool.size()),
222 m_next(0) {}
223
224 kj::Promise<void> getCapabilities(GetCapabilitiesContext context) override {
225 fillServerCapabilities(context.getResults().initCapabilities());
226 return kj::READY_NOW;
227 }
228
229 kj::Promise<void> calculate(CalculateContext context) override {
230 size_t idx = m_next.fetch_add(1, std::memory_order_relaxed) % m_pool.size();
231
232 auto fip = context.getParams().getFip();
233 const size_t numAtoms = fip.getPos().size() / 3;
234
235 KJ_REQUIRE(fip.getAtmnrs().size() == numAtoms, "AtomNumbers size mismatch");
236
237 std::vector<double> positions(numAtoms * 3);
238 auto capnpPos = fip.getPos();
239 for (size_t i = 0; i < numAtoms * 3; ++i) {
240 positions[i] = capnpPos[i];
241 }
242
243 std::vector<int> atomicNrs(numAtoms);
244 auto capnpAtmnrs = fip.getAtmnrs();
245 for (size_t i = 0; i < numAtoms; ++i) {
246 atomicNrs[i] = capnpAtmnrs[i];
247 }
248
249 double box[9] = {};
250 auto capnpBox = fip.getBox();
251 for (size_t i = 0; i < 9 && i < capnpBox.size(); ++i) {
252 box[i] = capnpBox[i];
253 }
254
255 std::vector<double> forces(numAtoms * 3, 0.0);
256 double energy = 0.0;
257
258 std::lock_guard<std::mutex> lock(m_mutexes[idx]);
259 m_pool[idx](static_cast<long>(numAtoms), positions.data(), atomicNrs.data(),
260 forces.data(), &energy, box);
261
262 auto result = context.getResults();
263 auto pres = result.initResult();
264 pres.setEnergy(energy);
265
266 auto forcesList = pres.initForces(numAtoms * 3);
267 for (size_t i = 0; i < numAtoms * 3; ++i) {
268 forcesList.set(i, forces[i]);
269 }
270
271 return kj::READY_NOW;
272 }
273
274private:
275 std::vector<ForceCallback> m_pool;
276 std::vector<std::mutex> m_mutexes;
277 std::atomic<size_t> m_next;
278};

295RpcServerCapabilitiesPair readRpcServerCapabilities() {
