23// Cores this process may run on: the affinity mask under Slurm or taskset,
24// else hardware_concurrency(). A band on 4 allocated cores of a 32-core
25// node then gets 4 threads, not 32 that share 4 cores.
26inline long usableCores() {
27#if defined(__linux__)
28 cpu_set_t set;
29 CPU_ZERO(&set);
30 if (sched_getaffinity(0, sizeof(set), &set) == 0) {
31 const int n = CPU_COUNT(&set);
32 if (n > 0)
33 return n;
34 }
35#endif
36 return std::max(1L, static_cast<long>(std::thread::hardware_concurrency()));
37}

39// Workers start once and wait for the next band, so a band costs a
40// condition-variable round trip instead of creating and joining threads.
41// One band runs at a time. A call made while a band runs (from inside a
42// work item, or from a second thread) runs serially on its caller, so the
43// pool never waits on itself.

64 // Calls work(1) .. work(n), the caller taking indices too, and rethrows
65 // the first exception after every index has run.
66 void run(long n, const std::function<void(long)> &work) {
67 if (n <= 0)
68 return;
69 std::unique_lock<std::mutex> busy(runMutex_, std::try_to_lock);
70 if (!busy.owns_lock() || threads_.empty() || n == 1) {
71 serial(n, work);
72 return;
73 }
74 {
75 std::lock_guard<std::mutex> lock(mutex_);
76 work_ = &work;
77 n_ = n;
78 next_.store(1);
79 failure_ = nullptr;
80 // Only as many helpers as there are images beyond the caller's.
81 helpers_ = std::min<long>(static_cast<long>(threads_.size()), n - 1);
82 active_ = helpers_;
83 ++generation_;
84 }
85 wake_.notify_all();
86 drain();
87 std::unique_lock<std::mutex> lock(mutex_);
88 done_.wait(lock, [this] { return active_ == 0; });
89 work_ = nullptr;
90 if (failure_)
91 std::rethrow_exception(failure_);
92 }

134 void loop(long index) {
135 std::size_t seen = 0;
136 for (;;) {
137 {
138 std::unique_lock<std::mutex> lock(mutex_);
139 wake_.wait(lock, [&] {
140 return stop_ || (generation_ != seen && index < helpers_);
141 });
142 if (stop_)
143 return;
144 seen = generation_;
145 }
146 drain();
147 {
148 std::lock_guard<std::mutex> lock(mutex_);
149 if (--active_ == 0)

151 }
152 }

172// Calls work(1) .. work(n) on at most usableCores() threads of a pool that
173// lives for the process. Each thread takes the next index until none are
174// left, so a band with more images than cores does not oversubscribe. The
175// first exception thrown by any call is rethrown here after every index has
176// run.
177template <typename Work> inline void forEachImage(long n, Work &&work) {
178 const std::function<void(long)> f = std::ref(work);
179 detail::ImagePool::instance().run(n, f);
180}
