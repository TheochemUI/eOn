152void RgpotPot::sendStop() {
153 if (!impl_ || !driver_ || stopped_ || impl_->calculatorWorld() <= 1)
154 return;
155 std::int64_t hdr[4] = {kStop, 0, 0, 0};
156 impl_->broadcastFromDriver(hdr, sizeof(hdr));
157 stopped_ = true;
158 if (g_driver == this)
159 g_driver = nullptr;

162void RgpotPot::exchangeGroupUse() {
163 const int groups = impl_->calculatorGroups();
164 use_.busy.assign(static_cast<size_t>(groups), 0.0);
165 use_.systems.assign(static_cast<size_t>(groups), 0.0);
166 for (int g = 0; g < groups; g++) {
167 double tally[3] = {busy_, systemsDone_, 0.0};
168 double unused = 0.0;
169 std::string error;
170 (void)impl_->shareResult(g, 1, tally, &unused, true, error);
171 use_.busy[static_cast<size_t>(g)] = tally[0];
172 use_.systems[static_cast<size_t>(g)] = tally[1];
173 }
174}

176void RgpotPot::stopAndDrop() {
177 if (!impl_)
178 return;
179 const bool grouped = driver_ && impl_->calculatorWorld() > 1;
180 if (grouped && !stopped_) {
181 sendStop();
182 try {
183 exchangeGroupUse();
184 std::cout << "RgpotPot: " << use_.table() << std::flush;
185 } catch (const std::exception &ex) {
186 std::cerr << "RgpotPot: no calculator-group summary: " << ex.what()
187 << std::endl;
188 }
189 }
190 if (!dropped_) {
191 impl_->shutdownModule();
192 dropped_ = true;
193 }
194 if (grouped && !acked_) {
195 std::int64_t hdr[4] = {kDown, 0, 0, 0};
196 impl_->broadcastFromDriver(hdr, sizeof(hdr));
197 acked_ = true;
198 }
199}

201void RgpotPot::releaseWorkersAtExit() {
202 if (g_driver || !impl_ || impl_->calculatorWorld() <= 1)
203 return;
204 // Runs before the MPI_Finalize handler registered in the constructor.
205 // After a failed engine call the workers can sit in a collective that
206 // the stop broadcast never meets; the grouped exit handler aborts the
207 // world instead.
208 g_driver = this;
209 std::atexit([] {
210 if (g_driver && !RGPotEngine::mpiAbortRequested())
211 g_driver->stopAndDrop();
212 });
213}

216// Runs one evaluation and keeps its error instead of throwing, so the
217// rank still joins every broadcast that follows.
218bool try_force(const RGPotEngine &engine, long N, const double *R,
219 const int *atomicNrs, double *F, double *U, const double *box,
220 std::string &error) {
221 try {
222 engine.force(N, R, atomicNrs, F, U, box);
223 return true;
224 } catch (const std::exception &ex) {
225 error = ex.what();
226 return false;
227 }
228}
229
230// The orbital key of system j of a batch: its owner hint (a NEB image, a
231// ring bead) when it has one, else its batch position, kept apart from
232// the hints. A single evaluation has its own key. cpmdc keeps converged
233// orbitals per key, so a calculator that evaluates several systems in
234// turn starts each SCF from that system's own previous orbitals.
235constexpr std::int64_t kSingleKey = -1;
236std::int64_t orbital_key(const std::int64_t *owners, long j) {
237 if (owners && owners[j] >= 0)
238 return owners[j];
239 return -2 - static_cast<std::int64_t>(j);
240}

243 const std::string &error) {
244 std::string msg = "RGPOT: calculator " + std::to_string(owner) +
245 " failed on system " + std::to_string(system);
246 if (!error.empty())
247 msg += ": " + error;
248 throw std::runtime_error(msg);
249}

252bool RgpotPot::evaluate(long N, const double *R, const int *atomicNrs,
253 double *F, double *U, const double *box,
254 std::string &error) {
255 const auto t0 = std::chrono::steady_clock::now();
256 const bool ok = try_force(*impl_, N, R, atomicNrs, F, U, box, error);
257 busy_ += std::chrono::duration<double>(std::chrono::steady_clock::now() - t0)
258 .count();
259 systemsDone_ += 1.0;
260 return ok;
261}

263void RgpotPot::computeSingle(long N, const double *R, const int *atomicNrs,
264 double *F, double *U, const double *box,
265 int group) {
266 // Every calculator evaluates the structure, because the engine agrees
267 // on errors across MPI_COMM_WORLD after each call and a calculator that
268 // skipped the call would never join that agreement. The first rank of
269 // `group`, the one whose stored orbitals are nearest, sends its result
270 // to every rank.
271 std::string error;
272 // Only that group's call counts toward the calculator usage report.
273 impl_->selectOrbitals(kSingleKey);
274 const bool ok = impl_->calculatorIndex() == group
275 ? evaluate(N, R, atomicNrs, F, U, box, error)
276 : try_force(*impl_, N, R, atomicNrs, F, U, box, error);
277 if (!impl_->shareResult(group, N, F, U, ok, error))
278 raise_failure(0, group, error);
279}

281void RgpotPot::computeBatch(long nSystems, long nAtoms,
282 const double *const *positions,
283 const int *const *atomicNrs, double *const *forces,
284 double *energies, const double *const *boxes,
285 const std::int64_t *owners,
286 const std::int64_t *route) {
287 const int groups = impl_->calculatorGroups();
288 const int mine = impl_->calculatorIndex();
289 auto ownerOf = [&](long j) { return static_cast<int>(route[j]); };
290 if (impl_->calculatorWorld() <= 1) {
291 for (long j = 0; j < nSystems; j++) {
292 impl_->selectOrbitals(orbital_key(owners, j));
293 impl_->force(nAtoms, positions[j], atomicNrs[j], forces[j], &energies[j],
294 boxes[j]);
295 }
296 return;
297 }
298 std::vector<char> ok(static_cast<size_t>(nSystems), 1);
299 std::vector<std::string> errors(static_cast<size_t>(nSystems));
300 std::vector<long> owned(static_cast<size_t>(groups), 0);
301 long last = -1;
302 for (long j = 0; j < nSystems; j++) {
303 owned[static_cast<size_t>(ownerOf(j))]++;
304 if (ownerOf(j) == mine) {
305 last = j;
306 impl_->selectOrbitals(orbital_key(owners, j));
307 ok[static_cast<size_t>(j)] =
308 evaluate(nAtoms, positions[j], atomicNrs[j], forces[j], &energies[j],
309 boxes[j], errors[static_cast<size_t>(j)]);
310 }
311 }
312 // Every calculator makes the same number of engine calls: the engine
313 // agrees on errors across MPI_COMM_WORLD after each one. A calculator
314 // that owns fewer systems repeats one into scratch buffers, its last
315 // own system or system 0 when it owns none. The repeat runs under that
316 // system's key, so a repeat of the last own system is the session's
317 // stored result and costs no SCF.
318 const long most = *std::max_element(owned.begin(), owned.end());
319 const long pad = most - owned[static_cast<size_t>(mine)];
320 if (pad > 0) {
321 const long j = last >= 0 ? last : 0;
322 impl_->selectOrbitals(orbital_key(owners, j));
323 std::vector<double> scratchF(static_cast<size_t>(3 * nAtoms));
324 double scratchU = 0.0;
325 std::string scratchError;
326 for (long k = 0; k < pad; k++)
327 try_force(*impl_, nAtoms, positions[j], atomicNrs[j], scratchF.data(),
328 &scratchU, boxes[j], scratchError);
329 }
330 // Every share runs before any rank raises, so all ranks leave together.
331 long failed = -1;
332 for (long j = 0; j < nSystems; j++) {
333 const int owner = ownerOf(j);
334 if (!impl_->shareResult(owner, nAtoms, forces[j], &energies[j],
335 ok[static_cast<size_t>(j)] != 0,
336 errors[static_cast<size_t>(j)]) &&
337 failed < 0)
338 failed = j;
339 }
340 if (failed >= 0)
341 raise_failure(failed, ownerOf(failed), errors[static_cast<size_t>(failed)]);
342}

344void RgpotPot::serveWorker() {
345 for (;;) {
346 std::int64_t hdr[4] = {kStop, 0, 0, 0};
347 impl_->broadcastFromDriver(hdr, sizeof(hdr));
348 if (hdr[0] == kStop) {
349 // Drop CPMD, wait until the driver has dropped it too, then exit 0.
350 // MPI_Finalize is collective and runs from the exit handler.
351 try {
352 exchangeGroupUse();
353 } catch (const std::exception &) {
354 }
355 if (!dropped_) {
356 impl_->shutdownModule();
357 dropped_ = true;
358 }
359 std::int64_t ack[4] = {kDown, 0, 0, 0};
360 impl_->broadcastFromDriver(ack, sizeof(ack));
361 std::exit(0);
362 }
363 const long n = static_cast<long>(hdr[1]);
364 const long m = hdr[0] == kBatch ? static_cast<long>(hdr[2]) : 1;
365 std::vector<double> R(static_cast<size_t>(3 * n * m));
366 std::vector<int> Z(static_cast<size_t>(n * m));
367 std::vector<double> box(static_cast<size_t>(9 * m));
368 impl_->broadcastFromDriver(R.data(), R.size() * sizeof(double));
369 impl_->broadcastFromDriver(Z.data(), Z.size() * sizeof(int));
370 impl_->broadcastFromDriver(box.data(), box.size() * sizeof(double));
371 std::vector<std::int64_t> ids(static_cast<size_t>(m), -1);
372 std::vector<std::int64_t> route(static_cast<size_t>(m), 0);
373 if (hdr[0] == kBatch) {
374 impl_->broadcastFromDriver(ids.data(), ids.size() * sizeof(std::int64_t));
375 impl_->broadcastFromDriver(route.data(),
376 route.size() * sizeof(std::int64_t));
377 }
378 std::vector<double> F(R.size()), U(static_cast<size_t>(m));
379 // A failed request raised on the driver too; the driver decides
380 // whether the job goes on, so a worker keeps serving.
381 if (hdr[0] == kSingle) {
382 try {
383 computeSingle(n, R.data(), Z.data(), F.data(), U.data(), box.data(),
384 static_cast<int>(hdr[3]));
385 } catch (const std::runtime_error &) {
386 }
387 continue;
388 }
389 std::vector<const double *> pos(static_cast<size_t>(m)), bx(pos.size());
390 std::vector<const int *> nrs(pos.size());
391 std::vector<double *> frc(pos.size());
392 for (long j = 0; j < m; j++) {
393 pos[static_cast<size_t>(j)] = R.data() + 3 * n * j;
394 nrs[static_cast<size_t>(j)] = Z.data() + n * j;
395 frc[static_cast<size_t>(j)] = F.data() + 3 * n * j;
396 bx[static_cast<size_t>(j)] = box.data() + 9 * j;
397 }
398 try {
399 computeBatch(m, n, pos.data(), nrs.data(), frc.data(), U.data(),
400 bx.data(), ids.data(), route.data());
401 } catch (const std::runtime_error &) {
402 }
403 }
404}

406void RgpotPot::force(long N, const double *R, const int *atomicNrs, double *F,
407 double *U, double *variance, const double *box) {
408 if (variance)
409 *variance = 0.0;
410 if (impl_->calculatorWorld() <= 1) {
411 impl_->selectOrbitals(kSingleKey);
412 impl_->force(N, R, atomicNrs, F, U, box);
413 return;
414 }
415 const auto t0 = std::chrono::steady_clock::now();
416 if (!schedule_)
417 schedule_ =
418 std::make_unique<eonc::GroupSchedule>(impl_->calculatorGroups());
419 // The group whose last geometry is nearest holds the warmest orbitals.
420 const int group = schedule_->nearest(R, 3 * N);
421 schedule_->settle(kSingleKey, group);
422 schedule_->record(kSingleKey, R, 3 * N);
423 std::int64_t hdr[4] = {kSingle, N, 1, group};
424 impl_->broadcastFromDriver(hdr, sizeof(hdr));
425 impl_->broadcastFromDriver(const_cast<double *>(R), 3 * N * sizeof(double));
426 impl_->broadcastFromDriver(const_cast<int *>(atomicNrs), N * sizeof(int));
427 impl_->broadcastFromDriver(const_cast<double *>(box), 9 * sizeof(double));
428 // Before the evaluation, so a failure that ends the job still stops
429 // the workers.
430 releaseWorkersAtExit();
431 computeSingle(N, R, atomicNrs, F, U, box, group);
432 const double dt =
433 std::chrono::duration<double>(std::chrono::steady_clock::now() - t0)
434 .count();
435 use_.wall += dt;
436 use_.singleWall += dt;
437 use_.singles++;
438}

457void RgpotPot::forceBatchOwned(long nSystems, long nAtoms,
458 const double *const *positions,
459 const int *const *atomicNrs,
460 double *const *forces, double *energies,
461 double *variances, const double *const *boxes,
462 const long *owners) {
463 std::vector<std::int64_t> ids(static_cast<size_t>(nSystems), -1);
464 if (owners) {
465 for (long j = 0; j < nSystems; j++)
466 ids[static_cast<size_t>(j)] = owners[j];
467 }
468 const bool grouped = impl_->calculatorWorld() > 1;
469 const auto t0 = std::chrono::steady_clock::now();
470 std::vector<std::int64_t> route(static_cast<size_t>(nSystems), 0);
471 if (grouped) {
472 if (!schedule_)
473 schedule_ =
474 std::make_unique<eonc::GroupSchedule>(impl_->calculatorGroups());
475 std::vector<std::int64_t> keys(static_cast<size_t>(nSystems));
476 for (long j = 0; j < nSystems; j++)
477 keys[static_cast<size_t>(j)] = orbital_key(ids.data(), j);
478 const std::vector<int> groupOf = schedule_->assign(keys);
479 for (long j = 0; j < nSystems; j++) {
480 route[static_cast<size_t>(j)] = groupOf[static_cast<size_t>(j)];
481 schedule_->record(keys[static_cast<size_t>(j)], positions[j], 3 * nAtoms);
482 }
483 std::int64_t hdr[4] = {kBatch, nAtoms, nSystems, 0};
484 impl_->broadcastFromDriver(hdr, sizeof(hdr));
485 std::vector<double> R(static_cast<size_t>(3 * nAtoms * nSystems));
486 std::vector<int> Z(static_cast<size_t>(nAtoms * nSystems));
487 std::vector<double> box(static_cast<size_t>(9 * nSystems));
488 for (long j = 0; j < nSystems; j++) {
489 std::copy(positions[j], positions[j] + 3 * nAtoms,
490 R.begin() + 3 * nAtoms * j);
491 std::copy(atomicNrs[j], atomicNrs[j] + nAtoms, Z.begin() + nAtoms * j);
492 std::copy(boxes[j], boxes[j] + 9, box.begin() + 9 * j);
493 }
494 impl_->broadcastFromDriver(R.data(), R.size() * sizeof(double));
495 impl_->broadcastFromDriver(Z.data(), Z.size() * sizeof(int));
496 impl_->broadcastFromDriver(box.data(), box.size() * sizeof(double));
497 impl_->broadcastFromDriver(ids.data(), ids.size() * sizeof(std::int64_t));
498 impl_->broadcastFromDriver(route.data(),
499 route.size() * sizeof(std::int64_t));
500 }
501 releaseWorkersAtExit();
502 computeBatch(nSystems, nAtoms, positions, atomicNrs, forces, energies, boxes,
503 ids.data(), route.data());
504 if (grouped) {
505 use_.wall +=
506 std::chrono::duration<double>(std::chrono::steady_clock::now() - t0)
507 .count();
508 use_.batches++;
509 }
510 for (long j = 0; j < nSystems; j++) {
511 if (variances)
512 variances[j] = 0.0;
513 forceCallCounter++;
514 eonc::PotRegistry::get().on_force_call(ptype);
515 }
