19int FIRE::step(double a_maxMove) {
20 double P = 0;
21 // Check convergence.
22 if (m_objf->isConverged()) {
23 return 1;
24 }
25
26 // Velocity Verlet
27 Eigen::VectorXd f = -m_objf->getGradient();
28 Eigen::VectorXd x = m_objf->getPositions();
29
30 m_vel += f * m_dt;
31 Eigen::VectorXd dx = m_vel * m_dt;
32
33 dx = eonc::helpers::maxAtomMotionAppliedV(dx, m_max_move);
34 m_objf->setPositions(x + dx);
35
36 f = -m_objf->getGradient();
37 Eigen::VectorXd f_unit = f / f.norm();
38
39 // FIRE
40 P = f.dot(m_vel);
41 m_vel = (1 - m_alpha) * m_vel + m_alpha * f_unit * m_vel.norm();
42 QUILL_LOG_DEBUG(m_log,
43 "P: {:.4f}, v: {:.4f}, m_dt: {:.4f}, m_alpha: {:.4f}, N: {}",
44 P, m_vel.norm(), m_dt, m_alpha, m_N);
45 if (P >= 0) {
46 m_N++;
47 if (m_N > m_N_min) {
48 m_dt = std::min(m_dt * m_f_inc, m_dt_max);
49 m_alpha = m_alpha * m_f_a;
50 }
51 } else {
52 m_dt = m_dt * m_f_dec;
53 m_vel.setZero();
54 m_alpha = m_alpha_start;
55 m_N = 0;
56 }
57
58 // add a sanity check on m_dt
59 if (m_dt < 1e-6) {
60 QUILL_LOG_CRITICAL(m_log, "[FIRE] [critical] m_dt is too small: {:.4f}",
61 m_dt);
62 throw std::runtime_error(
63 std::format("[FIRE] m_dt is too small: {:.4f}", m_dt));
64 }
65
66 m_iteration++;
67 return m_objf->isConverged() ? 1 : 0;
68}

70int FIRE::run(size_t a_maxIterations, double a_maxMove) {
71 while (!m_objf->isConverged() && m_iteration < a_maxIterations) {
72 step(a_maxMove);
73 }
74 return m_objf->isConverged() ? 1 : 0;
