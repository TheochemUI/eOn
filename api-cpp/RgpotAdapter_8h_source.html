117 void forceBatch(long nSystems, long nAtoms, const double *const *positions,
118 const int *const *atomicNrs, double *const *forces,
119 double *energies, double *variances,
120 const double *const *boxes) override {
121 const auto n = static_cast<size_t>(nSystems);
122 std::vector<rgpot::ForceInput> in;
123 std::vector<rgpot::ForceOut> out;
124 in.reserve(n);
125 out.reserve(n);
126 for (size_t i = 0; i < n; ++i) {
127 in.push_back(rgpot::ForceInput{static_cast<size_t>(nAtoms), positions[i],
128 atomicNrs[i], boxes[i]});
129 out.push_back(rgpot::ForceOut{forces[i], 0.0, 0.0});
130 }
131
132 pot_.forceBatchImpl(
133 rgpot::ForceBatch{.nSystems = n, .in = in.data(), .out = out.data()});
134
135 haveStress_ = false;
136 for (size_t i = 0; i < n; ++i) {
137 energies[i] = out[i].energy;
138 if (variances != nullptr) {
139 variances[i] = out[i].variance;
140 }
141 if (out[i].has_stress) {
142 storeStress(out[i]);
143 }
144 forceCallCounter++;
145 eonc::PotRegistry::get().on_force_call(ptype);
146 }
147 }

179 void finishConstruct(const eonc::Parameters &params) {

181 validateCaps();

183
184 void validateCaps() const {
185 const auto caps = pot_.caps();
186 using R = rgpot::Reentrancy;
187 if (caps.reentrancy != R::SharedInstance &&
188 caps.reentrancy != R::PerInstance &&
189 caps.reentrancy != R::ProcessSerial) {
190 throw std::runtime_error("rgpot kernel advertised an unknown reentrancy");
191 }
192 }
193
194 void storeStress(const rgpot::ForceOut &out) {
195 haveStress_ = out.has_stress != 0;
196 if (!haveStress_) {
197 return;
198 }
199 for (int row = 0; row < 3; ++row) {
200 for (int col = 0; col < 3; ++col) {
201 stress_(row, col) = out.stress[row * 3 + col];
202 }
203 }
204 }
