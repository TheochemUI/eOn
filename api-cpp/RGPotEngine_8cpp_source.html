45// Serialized Cap'n Proto message (standard segment framing, word-aligned) as
46// written by `capnp encode` or messageToFlatArray.
47std::vector<::capnp::word> read_params_file(const std::string &path) {
48 std::ifstream in(path, std::ios::binary | std::ios::ate);
49 if (!in)
50 throw std::runtime_error("RGPOT: cannot open params_path: " + path);
51 const std::streamsize bytes = in.tellg();
52 if (bytes <= 0 || (static_cast<size_t>(bytes) % sizeof(::capnp::word)) != 0)
53 throw std::runtime_error(
54 "RGPOT: params_path is not a capnp flat message: " + path);
55 std::vector<::capnp::word> words(static_cast<size_t>(bytes) /
56 sizeof(::capnp::word));
57 in.seekg(0);
58 in.read(reinterpret_cast<char *>(words.data()), bytes);
59 if (!in)
60 throw std::runtime_error("RGPOT: short read on params_path: " + path);
61 return words;
62}

78bool looks_like_dft_xc(const std::string &s) {
79 if (s.empty())
80 return false;
81 static const char *k[] = {"b3lyp", "blyp", "pbe", "pw91", "bp86",
82 "hcth", "ft97", "hfexch", "xperpbe", nullptr};
83 for (int i = 0; k[i]; ++i) {
84 if (s.size() >= std::char_traits<char>::length(k[i]) &&
85 s.compare(0, std::char_traits<char>::length(k[i]), k[i]) == 0)
86 return true;
87 }
88 return false;
89}

93 const std::string p = to_lower(paramset);

130// librgpot_pot_mpi, the MPI half of the calculator groups, or nullptr.
131// Loaded on first use only, so a process started without an MPI launcher
132// never maps libmpi. Search: EON_RGPOT_MPI_LIBRARY, the file next to this
133// library, then the linker path.
134const EonRgpotGroupMpi *group_mpi() {
135#if defined(__linux__)
136 static const EonRgpotGroupMpi *api = [] {
137 std::vector<std::string> candidates;
138 if (const char *e = std::getenv("EON_RGPOT_MPI_LIBRARY"); e && *e)
139 candidates.emplace_back(e);
140 Dl_info info{};
141 if (dladdr(reinterpret_cast<void *>(&group_mpi), &info) != 0 &&
142 info.dli_fname != nullptr) {
143 std::string self(info.dli_fname);
144 const auto slash = self.rfind('/');
145 if (slash != std::string::npos)
146 candidates.push_back(self.substr(0, slash + 1) + "librgpot_pot_mpi.so");
147 }
148 candidates.emplace_back("librgpot_pot_mpi.so");
149 for (const auto &path : candidates) {
150 void *h = dlopen(path.c_str(), RTLD_NOW | RTLD_GLOBAL);
151 if (h == nullptr)
152 continue;
153 using Entry = const EonRgpotGroupMpi *(*)();
154 auto entry = reinterpret_cast<Entry>(dlsym(h, "eon_rgpot_group_mpi_v1"));
155 const EonRgpotGroupMpi *table = entry ? entry() : nullptr;
156 if (table != nullptr && table->version == EON_RGPOT_GROUP_MPI_VERSION)
157 return table;
158 }
159 return static_cast<const EonRgpotGroupMpi *>(nullptr);
160 }();
161 return api;
162#else
163 return nullptr;
164#endif
165}

167// Every rank calls this when started under an MPI launcher. False when any
168// rank failed; message is then the lowest failing rank's text.
169bool agree_construction(std::string &message) {
170 const EonRgpotGroupMpi *api = group_mpi();
171 if (api == nullptr)
172 return message.empty();
173 std::array<char, 4096> shared{};
174 if (api->agree(message.c_str(), shared.data(), shared.size()) != 0)
175 return true;
176 message.assign(shared.data());
177 return false;
178}

183::CPMDParams::Builder fillCpmdParams(::capnp::MallocMessageBuilder &msg,
184 const RGPotEngineOptions &opt) {
185 ::CPMDParams::Builder params = msg.initRoot<::CPMDParams>();
186 if (!opt.params_path.empty()) {
187 const auto words = read_params_file(opt.params_path);
188 ::capnp::FlatArrayMessageReader reader(
189 kj::arrayPtr(words.data(), words.size()));
190 msg.setRoot(reader.getRoot<::CPMDParams>());
191 params = msg.getRoot<::CPMDParams>();
192 } else {
193 params.setFunctional(opt.functional);
194 params.setCutOffRy(opt.cutoff_ry);
195 params.setCharge(opt.charge);
196 params.setMultiplicity(opt.multiplicity);
197 if (!opt.title.empty())
198 params.setTitle(opt.title);
199 if (opt.memory_mb > 0)
200 params.setMemoryMb(static_cast<uint32_t>(opt.memory_mb));
201 }
202 if (!opt.engine_path.empty())
203 params.setEnginePath(opt.engine_path);
204 else if (!opt.engine_library.empty())
205 params.setEnginePath(opt.engine_library);
206 if (!opt.engine_root.empty())
207 params.setCpmdRoot(opt.engine_root);
208 if (!opt.scratch_dir.empty())
209 params.setScratchDir(opt.scratch_dir);
210 if (!opt.permanent_dir.empty())
211 params.setPermanentDir(opt.permanent_dir);
212
213 std::string block = opt.input_block;
214 if (block.empty()) {
215 if (const char *env = std::getenv("RGPOT_CPMD_INPUT_BLOCK"))
216 block = env;
217 }
218 if (!block.empty()) {
219 std::vector<std::string> kept;
220 {
221 auto existing = params.getInputBlocks();
222 kept.reserve(existing.size());
223 for (auto text : existing)
224 kept.emplace_back(text.cStr());
225 }
226 auto blocks = params.initInputBlocks(kept.size() + 1);
227 for (std::size_t i = 0; i < kept.size(); ++i)
228 blocks.set(i, kept[i]);
229 blocks.set(kept.size(), block);
230 }
231 return params;
232}

250RGPotEngine::RGPotEngine(const RGPotEngineOptions &opt)
251 : impl_(std::make_unique<Impl>()) {
252 backend_ = to_lower(opt.backend);
253 if (backend_ == "nwchem" || backend_ == "nwchemc" ||
254 backend_ == "nwchempot") {
255 backend_ = "nwchemc";
256 impl_->backend = Impl::Backend::Nwchemc;
257 ::capnp::MallocMessageBuilder msg;
258 auto params = msg.initRoot<::NWChemParams>();
259 params.setBasis(opt.basis);
260 params.setTheory(opt.theory);
261 params.setScfType(opt.scf_type);
262 params.setCharge(opt.charge);
263 params.setMultiplicity(opt.multiplicity);
264 if (!opt.engine_path.empty())
265 params.setEnginePath(opt.engine_path);
266 else if (!opt.engine_library.empty())
267 params.setEnginePath(opt.engine_library);
268 if (!opt.engine_root.empty())
269 params.setNwchemRoot(opt.engine_root);
270 if (!opt.title.empty())
271 params.setTitle(opt.title);
272 if (opt.memory_mb > 0)
273 params.setMemoryMb(static_cast<uint32_t>(opt.memory_mb));
274 if (!opt.scratch_dir.empty())
275 params.setScratchDir(opt.scratch_dir);
276 // DFT XC as inputBlocks when theory=dft and scfType is an XC label
277 std::string block = opt.input_block;
278 if (block.empty()) {
279 if (const char *env = std::getenv("RGPOT_NWCHEM_INPUT_BLOCK"))
280 block = env;
281 }
282 if (block.empty() && (opt.theory == "dft" || opt.theory == "DFT") &&
283 looks_like_dft_xc(opt.scf_type)) {
284 block = "dft\n xc " + opt.scf_type + "\n mult " +
285 std::to_string(opt.multiplicity) + "\nend";
286 } else if (block.empty() && looks_like_dft_xc(opt.theory)) {
287 block = "dft\n xc " + opt.theory + "\n mult " +
288 std::to_string(opt.multiplicity) + "\nend";
289 }
290 if (!block.empty()) {
291 auto blocks = params.initInputBlocks(1);
292 blocks.set(0, block);
293 }
294 impl_->nwchem = std::make_unique<rgpot::NWChemPot>(params.asReader());
295 if (!impl_->nwchem->available())
296 throw std::runtime_error(
297 "RGPOT(nwchemc): engine not available (set NWCHEMC_LIBRARY / "
298 "RGPOT_NWCHEMC_ENGINE or [RgpotPot] engine_path)");
299 } else if (backend_ == "cpmd" || backend_ == "cpmdc" ||
300 backend_ == "cpmdpot") {
301 backend_ = "cpmdc";
302 impl_->backend = Impl::Backend::Cpmdc;
303 std::string local_error;
304 try {
305 ::capnp::MallocMessageBuilder msg;
306 ::CPMDParams::Builder params = eon::fillCpmdParams(msg, opt);
307 impl_->cpmd = std::make_unique<rgpot::CPMDPot>(params.asReader());
308 pin_cpmd_library(opt.engine_path.empty() ? opt.engine_library
309 : opt.engine_path);
310 if (!impl_->cpmd->available())
311 throw std::runtime_error(
312 "RGPOT(cpmdc): engine not available (set CPMDC_LIBRARY / "
313 "RGPOT_CPMDC_ENGINE or [RgpotPot] engine_path)");
314 } catch (const std::exception &ex) {
315 local_error = ex.what();
316 }
317 // Several ranks: agree before MPI_Comm_split. One rank that cannot
318 // read params_path must not leave the others inside the split.
319 if (mpi_world_hint() <= 1 && !local_error.empty())
320 throw std::runtime_error(local_error);
321 // rgpot reports MPI only once MPI_Init has run, and agree_construction
322 // is what initialises it, so the agreement keys on the launch size.
323 if (mpi_world_hint() > 1 && !agree_construction(local_error)) {
324 armGroupedExit();
325 throw std::runtime_error(
326 local_error.empty()
327 ? std::string(
328 "RGPOT(cpmdc): a rank failed before the calculator split")
329 : local_error);
330 }
331 if (!local_error.empty())
332 throw std::runtime_error(local_error);
333 if (::rgpot::calculatorsUseMpi()) {
334 // Collective on MPI_COMM_WORLD, before the first force: the engine
335 // installs the group communicator ahead of CPMD's mp_start.
336 // ranks_per_image = 0 is one calculator on the whole world.
337 const rgpot::CalculatorGroup g =
338 ::rgpot::bindCalculators(opt.ranks_per_image);
339 ::rgpot::finalizeMpiAtExit();
340 if (mpi_world_hint() > 1)
341 armGroupedExit();
342 if (g.index < 0)
343 throw std::runtime_error(
344 "RGPOT(cpmdc): ranks_per_image=" +
345 std::to_string(opt.ranks_per_image) +
346 " does not divide the MPI world into calculator groups");
347 impl_->groups = ::rgpot::calculatorCount();
348 impl_->group = g.index;
349 impl_->world = ::rgpot::calculatorWorldSize();
350 } else if (opt.ranks_per_image > 0) {
351 throw std::runtime_error(
352 "RGPOT(cpmdc): ranks_per_image needs rgpot built with MPI "
353 "(-Drgpot:with_mpi=enabled)");
354 }
355 } else if (backend_ == "metatomic" || backend_ == "mta" ||
356 backend_ == "metatomicpot") {
357 backend_ = "metatomic";
358 impl_->backend = Impl::Backend::Metatomic;
359 MetatomicEngineOptions mopt;
360 mopt.model_path = opt.model_path;
361 mopt.device = opt.device;
362 mopt.length_unit = opt.length_unit;
363 mopt.extensions_directory = opt.extensions_directory;
364 mopt.check_consistency = opt.check_consistency;
365 mopt.uncertainty_threshold = opt.uncertainty_threshold;
366 mopt.engine_path =
367 !opt.engine_path.empty() ? opt.engine_path : opt.engine_library;
368 mopt.torch_determinism_strict = opt.torch_determinism_strict;
369 impl_->metatomic = std::make_unique<MetatomicEngineLoader>(mopt);
370 if (!impl_->metatomic->available())
371 throw std::runtime_error(
372 "RGPOT(metatomic): engine not available (set RGPOT_METATOMIC_ENGINE "
373 "or [RgpotPot] engine_path to librgpot_metatomic_engine.so)");
374 } else if (backend_ == "uma" || backend_ == "omol" || backend_ == "umapot") {
375 backend_ = "uma";
376 impl_->backend = Impl::Backend::Uma;
377 GenericEngineOptions gopt;
378 gopt.library = "libuma_engine.so";
379 gopt.env_var = "RGPOT_UMA_ENGINE";
380 gopt.engine_path =
381 !opt.engine_path.empty() ? opt.engine_path : opt.engine_library;
382 gopt.tag = "uma";
383 {
384 ::capnp::MallocMessageBuilder msg;
385 auto params = msg.initRoot<::UmaParams>();
386 params.setModelPath(opt.model_path);
387 params.setTaskName(opt.task_name);
388 params.setDevice(opt.device);
389 params.setCharge(opt.charge);
390 params.setSpin(opt.multiplicity);
391 const auto words = ::capnp::messageToFlatArray(msg);
392 const auto bytes = words.asBytes();
393 gopt.config.assign(bytes.begin(), bytes.end());
394 }
395 impl_->uma = std::make_unique<GenericEngineLoader>(gopt);
396 if (!impl_->uma->available())
397 throw std::runtime_error(
398 "RGPOT(uma): engine not available (set RGPOT_UMA_ENGINE or "
399 "[RgpotPot] engine_path to libuma_engine.so)");
400 } else if (backend_ == "xtb" || backend_ == "xtbpot" || backend_ == "gfn" ||
401 backend_ == "gfnxtb") {
402 backend_ = "xtb";
403 impl_->backend = Impl::Backend::Xtb;
404 XTBEngineOptions xopt;
405 xopt.method = xtb_method_from_paramset(opt.xtb_paramset);
406 xopt.accuracy = opt.xtb_accuracy;
407 xopt.electronic_temperature = opt.xtb_electronic_temperature;
408 xopt.max_iterations = opt.xtb_max_iterations;
409 xopt.charge = opt.xtb_charge;
410 xopt.uhf = opt.xtb_uhf;
411 xopt.engine_path =
412 !opt.engine_path.empty() ? opt.engine_path : opt.engine_library;
413 impl_->xtb = std::make_unique<XTBEngineLoader>(xopt);
414 if (!impl_->xtb->available())
415 throw std::runtime_error(
416 "RGPOT(xtb): engine not available (set RGPOT_XTB_ENGINE or "
417 "[RgpotPot] engine_path to librgpot_xtb_engine.so)");
418 } else {
419 throw std::runtime_error(
420 "RGPOT: unknown backend '" + opt.backend +
421 "' (expected nwchemc, cpmdc, metatomic, uma, or xtb)");
422 }

489bool RGPotEngine::shareResult(int owner, long N, double *F, double *U, bool ok,
490 std::string &error) const {
491 if (!impl_ || impl_->world <= 1) {
492 (void)error;
493 return ok;
494 }
495 // Slot 0 carries the owner's status, so a failed evaluation reaches
496 // every rank through the same broadcast as a good one.
497 std::vector<double> buf(static_cast<size_t>(3 * N + 2));
498 if (owner == impl_->group) {
499 buf[0] = ok ? 1.0 : 0.0;
500 buf[1] = *U;
501 std::copy(F, F + 3 * N, buf.begin() + 2);
502 }
503 if (::rgpot::shareFromCalculator(owner, buf.data(),
504 buf.size() * sizeof(double)) == 0)
505 throw std::runtime_error("RGPOT: could not share a calculator result");
506 *U = buf[1];
507 std::copy(buf.begin() + 2, buf.end(), F);
508 // The owner's engine text travels with the status. Other ranks hold an
509 // empty string until this broadcast.
510 std::array<char, 512> msg{};
511 if (owner == impl_->group && !error.empty()) {
512 const auto n = std::min(error.size(), msg.size() - 1);
513 std::memcpy(msg.data(), error.data(), n);
514 }
515 if (::rgpot::shareFromCalculator(owner, msg.data(), msg.size()) == 0)
516 throw std::runtime_error("RGPOT: could not share a calculator error");
517 if (msg[0] != '\0')
518 error.assign(msg.data());
519 return buf[0] == 1.0;
520}

564void RGPotEngine::force(long N, const double *R, const int *atomicNrs,
565 double *F, double *U, const double *box) const {
566 forceEngine(N, R, atomicNrs, F, U, box);
567 // Set on one rank by the multi-rank failure test. The text is the

569 // the engine call, where an engine error surfaces, so the rank has
570 // already taken part in every collective the engine runs.
571 if (const char *fail = std::getenv("RGPOT_FORCE_FAIL")) {
572 if (fail[0] != '\0')
573 throw std::runtime_error(fail);

577void RGPotEngine::forceEngine(long N, const double *R, const int *atomicNrs,
578 double *F, double *U, const double *box) const {
579 if (N <= 0)
580 throw std::runtime_error("RGPotEngine::force called with N <= 0");
581
582 AtomMatrix positions(static_cast<int>(N), 3);
583 for (long i = 0; i < N; ++i) {
584 const int ii = static_cast<int>(i);
585 positions(ii, 0) = R[3 * i + 0];
586 positions(ii, 1) = R[3 * i + 1];
587 positions(ii, 2) = R[3 * i + 2];
588 }
589 std::vector<int> atmtypes(atomicNrs, atomicNrs + N);
590 const auto cell = box_from_row_major(box);
591
592 if (impl_->backend == Impl::Backend::Metatomic) {
593 impl_->metatomic->force(N, R, atomicNrs, F, U, nullptr, box);
594 return;
595 }
596 if (impl_->backend == Impl::Backend::Uma) {
597 // A molecular .con may omit the cell. A 25 A diagonal stands in, so
598 // a singular box does not reach the neighbor list.
599 const bool got_cell = box && box[0] > 0.0 && box[4] > 0.0 && box[8] > 0.0;
600 std::array<double, 9> cell_sub{};
601 const double *effective_box = box;
602 if (!got_cell) {
603 cell_sub[0] = cell_sub[4] = cell_sub[8] = 25.0;
604 effective_box = cell_sub.data();
605 }
606 impl_->uma->force(N, R, atomicNrs, F, U, nullptr, effective_box);
607 return;
608 }
609 if (impl_->backend == Impl::Backend::Xtb) {
610 impl_->xtb->force(N, R, atomicNrs, F, U, nullptr, box);
611 return;
612 }
613
614 // rgpot >= 2.5.0: operator() returns (energy, forces, variance).
615 std::tuple<double, AtomMatrix, double> result;
616 if (impl_->backend == Impl::Backend::Nwchemc)
617 result = (*impl_->nwchem)(positions, atmtypes, cell);
618 else
619 result = (*impl_->cpmd)(positions, atmtypes, cell);
620
621 *U = std::get<0>(result);
622 const auto &forces = std::get<1>(result);
623 for (long i = 0; i < N; ++i) {
624 const int ii = static_cast<int>(i);
625 F[3 * i + 0] = forces(ii, 0);
626 F[3 * i + 1] = forces(ii, 1);
627 F[3 * i + 2] = forces(ii, 2);
628 }
629}
