35double massWeightedDistance(const Matter &a, const Matter &b) {
36 if (a.numberOfAtoms() != b.numberOfAtoms()) {
37 throw std::invalid_argument("the structures hold different atom counts");
38 }
39 const AtomMatrix dr = a.pbc(b.getPositions() - a.getPositions());
40 const auto mass = a.getMasses();
41 double sum = 0.0;
42 for (long i = 0; i < a.numberOfAtoms(); ++i) {
43 if (mass(i) <= 0.0) {
44 throw std::invalid_argument(
45 "every atom needs a positive mass for a mass-weighted path");
46 }
47 sum += mass(i) * dr.row(i).squaredNorm();
48 }
49 return std::sqrt(sum);
50}

62Profile::Profile(std::vector<double> s, std::vector<double> v)
63 : s_(std::move(s)),
64 v_(std::move(v)),
65 m_(s_.size(), 0.0) {
66 const size_t n = s_.size();
67 if (n < 2 || v_.size() != n) {
68 throw std::invalid_argument(
69 "a profile needs matching s and V with two points or more");
70 }
71 for (size_t k = 1; k < n; ++k) {
72 if (!(s_[k] > s_[k - 1])) {
73 throw std::invalid_argument(
74 "the path coordinate must increase along the band");
75 }
76 }
77 for (size_t k = 1; k + 1 < n; ++k) {
78 const double h0 = s_[k] - s_[k - 1];
79 const double h1 = s_[k + 1] - s_[k];
80 const double d0 = (v_[k] - v_[k - 1]) / h0;
81 const double d1 = (v_[k + 1] - v_[k]) / h1;
82 if (d0 * d1 <= 0.0) {
83 m_[k] = 0.0;
84 } else {
85 const double w1 = 2.0 * h1 + h0;
86 const double w2 = h1 + 2.0 * h0;
87 m_[k] = (w1 + w2) / (w1 / d0 + w2 / d1);
88 }
89 }
90}

92double Profile::operator()(double x) const {
93 x = std::clamp(x, s_.front(), s_.back());
94 auto it = std::upper_bound(s_.begin(), s_.end(), x);
95 size_t k = static_cast<size_t>(std::distance(s_.begin(), it));
96 k = std::clamp<size_t>(k == 0 ? 0 : k - 1, 0, s_.size() - 2);
97 const double h = s_[k + 1] - s_[k];
98 const double t = (x - s_[k]) / h;
99 const double t2 = t * t;
100 const double t3 = t2 * t;
101 return (2 * t3 - 3 * t2 + 1) * v_[k] + (t3 - 2 * t2 + t) * h * m_[k] +
102 (-2 * t3 + 3 * t2) * v_[k + 1] + (t3 - t2) * h * m_[k + 1];
103}

105double wellCurvature(const Profile &p, bool leftEnd) {
106 const auto &s = p.s();
107 const auto &v = p.v();
108 const size_t n = s.size();
109 const double top = *std::max_element(v.begin(), v.end());
110 const double floor = leftEnd ? v.front() : v.back();
111 const double half = 0.5 * (top - floor);
112 // Sums for the normal equations of y = a x^2 + b x^3.
113 double s44 = 0, s45 = 0, s55 = 0, sy2 = 0, sy3 = 0;
114 size_t used = 0;
115 for (size_t j = 1; j < n; ++j) {
116 const size_t i = leftEnd ? j : n - 1 - j;
117 const double x = leftEnd ? s[i] - s.front() : s.back() - s[i];
118 const double y = v[i] - floor;
119 if (y > half) {
120 break;
121 }
122 const double x2 = x * x;
123 s44 += x2 * x2;
124 s45 += x2 * x2 * x;
125 s55 += x2 * x2 * x2;
126 sy2 += y * x2;
127 sy3 += y * x2 * x;
128 ++used;
129 }
130 if (used == 0) {
131 // The next image already stands above half the barrier: the parabola
132 // through it is all the band says about this well.
133 const size_t i = leftEnd ? 1 : n - 2;
134 const double x = leftEnd ? s[i] - s.front() : s.back() - s[i];
135 return 2.0 * (v[i] - floor) / (x * x);
136 }
137 if (used == 1) {
138 return 2.0 * sy2 / s44;
139 }
140 const double det = s44 * s55 - s45 * s45;
141 const double a = (sy2 * s55 - sy3 * s45) / det;
142 return 2.0 * a;
143}

152double wkbAction(const Profile &p, double energy, int points) {
153 const double a = p.s().front();
154 const double b = p.s().back();
155 const double h = (b - a) / (points - 1);
156 double sum = 0.0;
157 for (int i = 0; i < points; ++i) {
158 const double gap = p(a + i * h) - energy;
159 const double f = gap > 0.0 ? std::sqrt(2.0 * gap) : 0.0;
160 sum += (i == 0 || i == points - 1) ? 0.5 * f : f;
161 }
162 return sum * h / kHbar;
163}

167Splitting wkbSplitting(const Profile &p, double hwReactant, double hwProduct) {
168 const auto &v = p.v();
169 Splitting out;
170 const double top = *std::max_element(v.begin(), v.end());
171 out.delta = v.back() - v.front();
172 out.barrier = top - v.front();
173 out.hwReactant = hwReactant;
174 out.hwProduct = hwProduct;
175 out.referenceEnergy =
176 std::max(v.front() + 0.5 * hwReactant, v.back() + 0.5 * hwProduct);
177 out.action = wkbAction(p, out.referenceEnergy);
178 const double hw = std::sqrt(hwReactant * hwProduct);
179 out.delta0 = hw / std::numbers::pi * std::exp(-out.action);
180 out.deepWells =
181 (top - v.front()) > hwReactant && (top - v.back()) > hwProduct;
182 return out;
183}

187std::vector<double> arcLengths(const std::vector<VectorXd> &path) {
188 std::vector<double> s(path.size(), 0.0);
189 for (size_t k = 1; k < path.size(); ++k) {
190 s[k] = s[k - 1] + (path[k] - path[k - 1]).norm();
191 }
192 return s;
193}

195VectorXd atArcLength(const std::vector<VectorXd> &path,
196 const std::vector<double> &s, double target) {
197 const auto it = std::upper_bound(s.begin(), s.end(), target);
198 const size_t k = std::clamp<size_t>(
199 static_cast<size_t>(std::distance(s.begin(), it)), 1, path.size() - 1);
200 const double seg = s[k] - s[k - 1];
201 const double t = seg > 0.0 ? (target - s[k - 1]) / seg : 0.0;
202 return path[k - 1] + std::clamp(t, 0.0, 1.0) * (path[k] - path[k - 1]);
203}

205// Crossings V(s) = e nearest the barrier top, one on each side.
206std::pair<double, double> turningPoints(const Profile &p, double sTop,
207 double energy) {
208 const double s0 = p.s().front();
209 const double s1 = p.s().back();
210 auto cross = [&](double from, double to) {
211 const int steps = 2000;
212 double a = from;
213 double b = to;
214 for (int k = 1; k <= steps; ++k) {
215 const double sk = from + (to - from) * static_cast<double>(k) / steps;
216 if (p(sk) <= energy) {
217 a = from + (to - from) * static_cast<double>(k - 1) / steps;
218 b = sk;
219 break;
220 }
221 if (k == steps) {
222 return to;
223 }
224 }
225 for (int k = 0; k < 60; ++k) {
226 const double m = 0.5 * (a + b);
227 if (p(m) > energy) {
228 a = m;
229 } else {
230 b = m;
231 }
232 }
233 return 0.5 * (a + b);
234 };
235 return {cross(sTop, s0), cross(sTop, s1)};
236}

238// int_{s-}^{s+} ds / sqrt(2 (V - E)). The cosine substitution keeps the
239// integrand finite at the turning points. Optional tables are the running
240// integral and the arc length at the end of each panel.
241double halfPeriod(const Profile &p, double sMinus, double sPlus, double energy,
242 std::vector<double> *cumulative = nullptr,
243 std::vector<double> *positions = nullptr) {
244 const int panels = 4000;
245 const double mid = 0.5 * (sMinus + sPlus);
246 const double half = 0.5 * (sPlus - sMinus);
247 double total = 0.0;
248 if (cumulative != nullptr) {
249 cumulative->assign(1, 0.0);
250 positions->assign(1, sMinus);
251 }
252 for (int k = 0; k < panels; ++k) {
253 const double phi =
254 std::numbers::pi * (static_cast<double>(k) + 0.5) / panels;
255 const double s = mid - half * std::cos(phi);
256 const double under = 2.0 * (p(s) - energy);
257 const double integrand =
258 half * std::sin(phi) / std::sqrt(std::max(under, 1e-300));
259 total += integrand * std::numbers::pi / panels;
260 if (cumulative != nullptr) {
261 cumulative->push_back(total);
262 positions->push_back(
263 mid - half * std::cos(std::numbers::pi * (k + 1.0) / panels));
264 }
265 }
266 return total;
267}

271std::vector<VectorXd> ringFromPath(const std::vector<VectorXd> &path,
272 const std::vector<double> &energies,
273 double betaHbar, long beads,
274 RingSeed *seed) {
275 if (path.size() < 3 || path.size() != energies.size() || beads < 4 ||
276 !(betaHbar > 0.0)) {
277 throw std::invalid_argument(
278 "ringFromPath: a path of at least three points with energies, "
279 "N >= 4 and beta hbar > 0");
280 }
281 const long width = path.front().size();
282 for (const auto &q : path) {
283 if (q.size() != width) {
284 throw std::invalid_argument("ringFromPath: the path changes dimension");
285 }
286 }
287 const std::vector<double> s = arcLengths(path);
288 const Profile profile(s, energies);
289 double sTop = s.front();
290 double vTop = -std::numeric_limits<double>::infinity();
291 const int grid = 4000;
292 for (int k = 0; k <= grid; ++k) {
293 const double sk =
294 s.front() + (s.back() - s.front()) * static_cast<double>(k) / grid;
295 if (profile(sk) > vTop) {
296 vTop = profile(sk);
297 sTop = sk;
298 }
299 }
300 const double vLow = std::max(energies.front(), energies.back());
301 if (!(vTop > vLow)) {
302 throw std::invalid_argument("ringFromPath: the path has no barrier");
303 }
304 auto period = [&](double energy) {
305 const auto [sMinus, sPlus] = turningPoints(profile, sTop, energy);
306 return 2.0 * halfPeriod(profile, sMinus, sPlus, energy);
307 };
308 // The crossover along the path from a parabola through the three input
309 // points around the barrier top, not from the interpolant, whose slope is
310 // clamped to zero at the top node.
311 {
312 size_t top = 0;
313 for (size_t k = 1; k < energies.size(); ++k) {
314 if (energies[k] > energies[top]) {
315 top = k;
316 }
317 }
318 if (top == 0 || top + 1 == energies.size()) {
319 throw std::invalid_argument(
320 "ringFromPath: the barrier top is an end of the path");
321 }
322 const double h1 = s[top] - s[top - 1], h2 = s[top + 1] - s[top];
323 const double curvature =
324 2.0 *
325 (h1 * energies[top + 1] - (h1 + h2) * energies[top] +
326 h2 * energies[top - 1]) /
327 (h1 * h2 * (h1 + h2));
328 if (!(curvature < 0.0)) {
329 throw std::invalid_argument("ringFromPath: no curvature at the top");
330 }
331 const double tc = kHbar * std::sqrt(-curvature) / (2.0 * std::numbers::pi);
332 if (!(kHbar / betaHbar < tc)) {
333 throw std::invalid_argument(
334 "ringFromPath: the temperature is at or above the crossover along "
335 "this path");
336 }
337 }
338 // Orbit energies on a geometric grid above the lower end: the period
339 // grows only logarithmically as E approaches a well bottom. The period
340 // need not fall monotonically with E: a shoulder on the path, where V'
341 // nearly vanishes away from a well, makes it spike, and a bisection over
342 // the whole range then converges onto the spike for every beta hbar it
343 // spans. Every crossing of beta hbar on the grid is refined, and the
344 // one kept minimises theta(E) + beta E, theta = 2 x the WKB action: it
345 // dominates the steepest-descent integral the instanton evaluates.
346 const double span = vTop - vLow;
347 const double eBottom = vLow + 1e-14 * span;
348 // Geometric in the distance from the top over the upper half, where a
349 // flat-topped barrier keeps the crossing within a small fraction of the
350 // span, and geometric in the height above the lower end over the lower
351 // half, where the period diverges only logarithmically.
352 constexpr int kHalf = 200;
353 constexpr int kScan = 2 * kHalf + 1;
354 std::vector<double> levels(kScan + 1);
355 std::vector<double> gap(kScan + 1);
356 for (int k = 0; k <= kHalf; ++k) {
357 const double u = static_cast<double>(k) / kHalf;
358 levels[static_cast<size_t>(k)] =
359 vTop - span * std::pow(10.0, -9.0 + u * (9.0 + std::log10(0.5)));
360 levels[static_cast<size_t>(kScan - k)] =
361 vLow + span * std::pow(10.0, -14.0 + u * (14.0 + std::log10(0.5)));
362 }
363 for (int k = 0; k <= kScan; ++k) {
364 gap[static_cast<size_t>(k)] =
365 period(levels[static_cast<size_t>(k)]) - betaHbar;
366 }
367 const double beta = betaHbar / kHbar;
368 double energy = eBottom;
369 double best = std::numeric_limits<double>::infinity();
370 for (int k = 0; k < kScan; ++k) {
371 double a = levels[static_cast<size_t>(k)];
372 double b = levels[static_cast<size_t>(k + 1)];
373 double ga = gap[static_cast<size_t>(k)];
374 if ((ga > 0.0) == (gap[static_cast<size_t>(k + 1)] > 0.0)) {
375 continue;
376 }
377 for (int it = 0; it < 200 && std::abs(a - b) > 1e-15 * span; ++it) {
378 const double m = vLow + std::sqrt((a - vLow) * (b - vLow));
379 const double gm = period(m) - betaHbar;
380 if ((gm > 0.0) == (ga > 0.0)) {
381 a = m;
382 ga = gm;
383 } else {
384 b = m;
385 }
386 }
387 const double root = 0.5 * (a + b);
388 // A bracket around a jump of the turning point refines onto the jump,
389 // where the period is not beta hbar.
390 if (!(std::abs(period(root) - betaHbar) <= 1e-6 * betaHbar)) {
391 continue;
392 }
393 const double w = 2.0 * wkbAction(profile, root) + beta * root;
394 if (w < best) {
395 best = w;
396 energy = root;
397 }
398 }
399 // With no crossing the path does not reach a long enough orbit; the
400 // lowest one it holds is the start.
401 if (seed != nullptr) {
402 seed->energy = energy;
403 seed->period = period(energy);
404 seed->pathLow = vLow;
405 seed->reached = std::abs(seed->period - betaHbar) <= 1e-6 * betaHbar;
406 }
407 const auto [sMinus, sPlus] = turningPoints(profile, sTop, energy);
408 std::vector<double> tau;
409 std::vector<double> pos;
410 const double half = halfPeriod(profile, sMinus, sPlus, energy, &tau, &pos);
411 if (tau.size() < 2 || tau.size() != pos.size()) {
412 throw std::invalid_argument("ringFromPath: the orbit has no length");
413 }
414 // Bead j sits at imaginary time j * beta hbar / N on the way from the
415 // reactant-side turning point to the other side. The return repeats it.
416 std::vector<VectorXd> ring(static_cast<size_t>(beads), VectorXd::Zero(width));
417 for (long j = 0; j <= beads / 2; ++j) {
418 const double t =
419 std::min(half, half * 2.0 * static_cast<double>(j) / beads);
420 const auto it = std::upper_bound(tau.begin(), tau.end(), t);
421 const size_t k = std::clamp<size_t>(
422 static_cast<size_t>(std::distance(tau.begin(), it)), 1, tau.size() - 1);
423 const double seg = tau[k] - tau[k - 1];
424 const double w = seg > 0.0 ? (t - tau[k - 1]) / seg : 0.0;
425 const double sj =
426 pos[k - 1] + std::clamp(w, 0.0, 1.0) * (pos[k] - pos[k - 1]);
427 ring[static_cast<size_t>(j)] = atArcLength(path, s, sj);
428 if (j > 0 && j < beads - j) {
429 ring[static_cast<size_t>(beads - j)] = ring[static_cast<size_t>(j)];
430 }
431 }
432 return ring;
433}

435double wkbLogRateAlongPath(const Profile &profile, double beta,
436 double hwReactant) {
437 if (!(beta > 0.0) || !(hwReactant > 0.0)) {
438 throw std::invalid_argument(
439 "wkbLogRateAlongPath: beta and hbar omega must be positive");
440 }
441 const double vReactant = profile.v().front();
442 const double s0 = profile.s().front();
443 const double s1 = profile.s().back();
444 double vTop = vReactant;
445 double sTop = s0;
446 const int grid = 2000;
447 for (int k = 0; k <= grid; ++k) {
448 const double sk = s0 + (s1 - s0) * static_cast<double>(k) / grid;
449 const double vk = profile(sk);
450 if (vk >= vTop) {
451 vTop = vk;
452 sTop = sk;
453 }
454 }
455 const double barrier = vTop - vReactant;
456 if (!(barrier > 0.0)) {
457 throw std::invalid_argument(
458 "wkbLogRateAlongPath: no barrier above the reactant");
459 }
460 double ds = 1e-3 * (s1 - s0);
461 ds = std::min(ds, std::min(sTop - s0, s1 - sTop));
462 if (!(ds > 0.0)) {
463 throw std::invalid_argument(
464 "wkbLogRateAlongPath: the barrier top is at an end of the path");
465 }
466 const double curvature =
467 std::max(1e-12, -(profile(sTop + ds) - 2.0 * vTop + profile(sTop - ds)) /
468 (ds * ds));
469 const double hwBarrier = kHbar * std::sqrt(curvature);
470 // int P(E) exp(-beta E) dE from the reactant up to where the Boltzmann
471 // factor has died. E is measured from the reactant.
472 const double eMax = barrier + 40.0 / beta;
473 const int points = 600;
474 auto logAdd = [](double a, double b) {
475 if (a == -std::numeric_limits<double>::infinity()) {
476 return b;
477 }
478 const double m = std::max(a, b);
479 return m + std::log(std::exp(a - m) + std::exp(b - m));
480 };
481 double logTerms = -std::numeric_limits<double>::infinity();
482 double prevLog = -std::numeric_limits<double>::infinity();
483 double prevE = 0.0;
484 for (int k = 0; k <= points; ++k) {
485 const double energy = eMax * static_cast<double>(k) / points;
486 double theta = 0.0;
487 if (energy < barrier) {
488 theta = wkbAction(profile, vReactant + energy);
489 } else {
490 theta = -std::numbers::pi * (energy - barrier) / hwBarrier;
491 }
492 const double logP =
493 theta > 20.0 ? -2.0 * theta : -std::log1p(std::exp(2.0 * theta));
494 const double logF = logP - beta * energy;
495 if (k > 0) {
496 const double segment =
497 std::log(0.5 * (energy - prevE)) + logAdd(prevLog, logF);
498 logTerms = logAdd(logTerms, segment);
499 }
500 prevLog = logF;
501 prevE = energy;
502 }
503 const double logFlux = logTerms - std::log(2.0 * std::numbers::pi * kHbar);
504 return logFlux + std::log(2.0 * std::sinh(0.5 * beta * hwReactant));
505}

507Splitting bandSplitting(const std::vector<std::shared_ptr<Matter>> &band,
508 double referenceEnergy) {
509 std::vector<double> v;
510 v.reserve(band.size());

514 const Profile p(massWeightedPath(band), std::move(v));

516 hbarOmega(wellCurvature(p, false)));
517}

531using ColMajorXd =
532 Eigen::Matrix<double, Eigen::Dynamic, Eigen::Dynamic, Eigen::ColMajor>;
533
534// J = c T (x) I + blockdiag(A_k), T the tridiagonal (2, -1) spring matrix;
535// the diagonal blocks hold 2c already, the off-diagonal blocks are -c I.
536// Block LU: D_1 = A_1, D_k = A_k - c^2 D_{k-1}^-1.
537class BlockChain {
538public:
539 BlockChain(double c, const std::vector<MatrixXd> &diag)
540 : c_(c) {
541 lu_.reserve(diag.size());
542 for (size_t k = 0; k < diag.size(); ++k) {
543 ColMajorXd d = diag[k];
544 if (k > 0) {
545 d -= c_ * c_ * lu_.back().inverse();
546 }
547 lu_.emplace_back(std::move(d));
548 const auto &f = lu_.back();
549 sign_ *= static_cast<int>(std::lround(f.permutationP().determinant()));
550 for (long i = 0; i < d.rows(); ++i) {
551 const double u = f.matrixLU()(i, i);
552 if (u == 0.0) {
553 throw std::runtime_error("instanton: singular chain Hessian block");
554 }
555 sign_ *= u < 0.0 ? -1 : 1;
556 logAbsDet_ += std::log(std::abs(u));
557 }
558 }
559 }
560 double logAbsDet() const { return logAbsDet_; }
561 int sign() const { return sign_; }
562 // x = J^-1 b, b and x stacked by bead.
563 std::vector<VectorXd> solve(const std::vector<VectorXd> &b) const {
564 const size_t m = b.size();
565 std::vector<VectorXd> y(m), x(m);
566 y[0] = b[0];
567 for (size_t k = 1; k < m; ++k) {
568 y[k] = b[k] + c_ * lu_[k - 1].solve(y[k - 1]);
569 }
570 x[m - 1] = lu_[m - 1].solve(y[m - 1]);
571 for (size_t k = m - 1; k-- > 0;) {
572 x[k] = lu_[k].solve(y[k] + c_ * x[k + 1]);
573 }
574 return x;
575 }
576 // Same recurrence, several right-hand sides per bead.
577 std::vector<MatrixXd> solve(const std::vector<MatrixXd> &b) const {
578 const size_t m = b.size();
579 std::vector<MatrixXd> y(m), x(m);
580 y[0] = b[0];
581 for (size_t k = 1; k < m; ++k) {
582 y[k] = b[k] + c_ * lu_[k - 1].solve(y[k - 1]);
583 }
584 x[m - 1] = lu_[m - 1].solve(y[m - 1]);
585 for (size_t k = m - 1; k-- > 0;) {
586 x[k] = lu_[k].solve(y[k] + c_ * x[k + 1]);
587 }
588 return x;
589 }
590
591private:
592 double c_;
593 std::vector<Eigen::PartialPivLU<ColMajorXd>> lu_;
594 double logAbsDet_ = 0.0;
595 int sign_ = 1;
596};

598// Block LU of an open block-tridiagonal chain with given diagonal blocks
599// and -c I between neighbours. Solves, and the inertia and log-determinant
600// from the Schur complements (Haynsworth: the inertia of the chain is the
601// sum over its Schur blocks).
602class HaynsworthChain {
603public:
604 HaynsworthChain(double c, const std::vector<MatrixXd> &diag, bool spectrum)
605 : c_(c) {
606 lu_.reserve(diag.size());
607 for (size_t k = 0; k < diag.size(); ++k) {
608 const long f = diag[k].rows();
609 MatrixXd d = 0.5 * (diag[k] + diag[k].transpose());
610 if (k > 0) {
611 const MatrixXd inv = lu_.back().inverse();
612 d -= c * c * 0.5 * (inv + inv.transpose());
613 }
614 if (spectrum) {
615 const ColMajorXd sym = d;
616 const Eigen::SelfAdjointEigenSolver<ColMajorXd> es(
617 sym, Eigen::EigenvaluesOnly);
618 for (long i = 0; i < f; ++i) {
619 const double lam = es.eigenvalues()(i);
620 if (lam == 0.0) {
621 throw std::runtime_error("instanton: singular chain Hessian block");
622 }
623 if (lam < 0.0) {
624 ++negative_;
625 }
626 logAbsDet_ += std::log(std::abs(lam));
627 }
628 }
629 lu_.emplace_back(ColMajorXd(d));
630 }
631 }
632 double logAbsDet() const { return logAbsDet_; }
633 long negative() const { return negative_; }
634 std::vector<VectorXd> solve(const std::vector<VectorXd> &b) const {
635 const size_t m = b.size();
636 std::vector<VectorXd> y(m), x(m);
637 y[0] = b[0];
638 for (size_t k = 1; k < m; ++k) {
639 y[k] = b[k] + c_ * lu_[k - 1].solve(y[k - 1]);
640 }
641 x[m - 1] = lu_[m - 1].solve(y[m - 1]);
642 for (size_t k = m - 1; k-- > 0;) {
643 x[k] = lu_[k].solve(y[k] + c_ * x[k + 1]);
644 }
645 return x;
646 }
647
648private:
649 double c_;
650 std::vector<Eigen::PartialPivLU<ColMajorXd>> lu_;
651 double logAbsDet_ = 0.0;
652 long negative_ = 0;
653};

669// Point at arc-length fraction f along a polyline.
670VectorXd alongPolyline(const std::vector<VectorXd> &pts,
671 const std::vector<double> &cum, double f) {
672 const double target = f * cum.back();
673 const auto it = std::upper_bound(cum.begin(), cum.end(), target);
674 const size_t k = std::clamp<size_t>(
675 static_cast<size_t>(std::distance(cum.begin(), it)), 1, pts.size() - 1);
676 const double seg = cum[k] - cum[k - 1];
677 const double t = seg > 0.0 ? (target - cum[k - 1]) / seg : 0.0;
678 return pts[k - 1] + std::clamp(t, 0.0, 1.0) * (pts[k] - pts[k - 1]);
679}

688ActionEval evaluateAction(const std::vector<VectorXd> &interior,
689 const VectorXd &start, const VectorXd &end,
690 double vStart, double vEnd, double dtau,
691 const BatchPotential &potential) {
692 ActionEval out;
693 potential(interior, out.v, out.potGrad);
694 const size_t m = interior.size();
695 if (out.v.size() != m || out.potGrad.size() != m) {
696 throw std::runtime_error("instanton: potential returned the wrong count");
697 }
698 auto bead = [&](size_t j) -> const VectorXd & {
699 return j == 0 ? start : (j == m + 1 ? end : interior[j - 1]);
700 };
701 double kinetic = 0.0;
702 for (size_t j = 0; j <= m; ++j) {
703 kinetic += (bead(j + 1) - bead(j)).squaredNorm();
704 }
705 double pot = 0.5 * (vStart + vEnd);
706 for (double vj : out.v) {
707 pot += vj;
708 }
709 out.action = 0.5 * kinetic / dtau + dtau * pot;
710 out.grad.resize(m);
711 for (size_t j = 1; j <= m; ++j) {
712 out.grad[j - 1] = (2.0 * bead(j) - bead(j - 1) - bead(j + 1)) / dtau +
713 dtau * out.potGrad[j - 1];
714 }
715 return out;
716}

719 double m = 0.0;
720 for (const auto &v : g) {
721 m = std::max(m, v.norm());

723 return m;
724}

739Instanton optimizeInstanton(const VectorXd &start, const VectorXd &end,
740 double betaHbar, std::vector<VectorXd> guess,
741 const BatchPotential &potential,
742 const InstantonOptions &options) {
743 const long P = options.beads;
744 if (P < 4 || !(betaHbar > 0.0) || start.size() != end.size()) {
745 throw std::invalid_argument("optimizeInstanton: need P >= 4, beta hbar > 0 "
746 "and ends of one dimension");
747 }
748 Instanton inst;
749 inst.betaHbar = betaHbar;
750 inst.dtau = betaHbar / static_cast<double>(P);
751 const double dtau = inst.dtau;
752
753 std::vector<double> vEnds;
754 std::vector<VectorXd> gEnds;
755 potential({start, end}, vEnds, gEnds);
756 if (vEnds.size() != 2) {
757 throw std::runtime_error("instanton: potential returned the wrong count");
758 }
759 inst.asymmetry = vEnds[1] - vEnds[0];
760
761 // Beads along the guess (or the straight line) on a tanh kink centred at
762 // beta hbar / 2 whose width follows the harmonic decay of a well.
763 if (guess.size() < 2) {
764 guess = {start, end};
765 }
766 std::vector<double> cum(guess.size(), 0.0);
767 for (size_t k = 1; k < guess.size(); ++k) {
768 cum[k] = cum[k - 1] + (guess[k] - guess[k - 1]).norm();
769 }
770 if (!(cum.back() > 0.0)) {
771 throw std::invalid_argument("optimizeInstanton: the two minima coincide");
772 }
773 const double width = betaHbar / (2.0 * options.betaHbarOmega);
774 std::vector<VectorXd> x(static_cast<size_t>(P - 1));
775 for (long j = 1; j < P; ++j) {
776 const double tau = static_cast<double>(j) * dtau - 0.5 * betaHbar;
777 const double f = 0.5 * (1.0 + std::tanh(tau / width));
778 x[static_cast<size_t>(j - 1)] = alongPolyline(guess, cum, f);
779 }
780
781 // L-BFGS with a backtracking Armijo line search.
782 ActionEval cur =
783 evaluateAction(x, start, end, vEnds[0], vEnds[1], dtau, potential);
784 std::deque<std::pair<std::vector<VectorXd>, std::vector<VectorXd>>> pairs;
785 for (long it = 0; it < options.maxIterations; ++it) {
786 inst.iterations = it;
787 if (largestBeadNorm(cur.grad) / dtau < options.forceTolerance) {
788 inst.converged = true;
789 break;
790 }
791 std::vector<VectorXd> q = cur.grad;
792 std::vector<double> alpha(pairs.size());
793 for (size_t i = pairs.size(); i-- > 0;) {
794 const double rho = 1.0 / dot(pairs[i].second, pairs[i].first);
795 alpha[i] = rho * dot(pairs[i].first, q);
796 for (size_t k = 0; k < q.size(); ++k) {
797 q[k] -= alpha[i] * pairs[i].second[k];
798 }
799 }
800 // Without history, half the inverse spring stiffness 2 / dtau.
801 double gamma = dtau / 4.0;
802 if (!pairs.empty()) {
803 gamma = dot(pairs.back().first, pairs.back().second) /
804 dot(pairs.back().second, pairs.back().second);
805 }
806 scale(q, gamma);
807 for (size_t i = 0; i < pairs.size(); ++i) {
808 const double rho = 1.0 / dot(pairs[i].second, pairs[i].first);
809 const double beta = rho * dot(pairs[i].second, q);
810 for (size_t k = 0; k < q.size(); ++k) {
811 q[k] += (alpha[i] - beta) * pairs[i].first[k];
812 }
813 }
814 // q is now the inverse-Hessian estimate times the gradient; step -q.
815 double slope = -dot(cur.grad, q);
816 if (!(slope < 0.0)) {
817 pairs.clear();
818 q = cur.grad;
819 scale(q, dtau / 4.0);
820 slope = -dot(cur.grad, q);
821 }
822 double step = 1.0;
823 ActionEval next;
824 std::vector<VectorXd> trial(x.size());
825 bool accepted = false;
826 for (int ls = 0; ls < 30; ++ls) {
827 for (size_t k = 0; k < x.size(); ++k) {
828 trial[k] = x[k] - step * q[k];
829 }
830 next = evaluateAction(trial, start, end, vEnds[0], vEnds[1], dtau,
831 potential);
832 // Near the minimum the action changes by less than its round-off;
833 // there a step that shrinks the gradient is progress too.
834 const bool armijo = next.action <= cur.action + 1e-4 * step * slope;
835 const bool flat = std::abs(next.action - cur.action) <=
836 1e-13 * std::max(1.0, std::abs(cur.action));
837 if (armijo ||
838 (flat && largestBeadNorm(next.grad) < largestBeadNorm(cur.grad))) {
839 accepted = true;
840 break;
841 }
842 step *= 0.5;
843 }
844 if (!accepted) {
845 break;
846 }
847 std::vector<VectorXd> sk(x.size()), yk(x.size());
848 for (size_t k = 0; k < x.size(); ++k) {
849 sk[k] = trial[k] - x[k];
850 yk[k] = next.grad[k] - cur.grad[k];
851 }
852 if (dot(sk, yk) > 0.0) {
853 pairs.emplace_back(std::move(sk), std::move(yk));
854 if (static_cast<long>(pairs.size()) > options.memory) {
855 pairs.pop_front();
856 }
857 }
858 x = std::move(trial);
859 cur = std::move(next);
860 }
861 if (!inst.converged &&
862 largestBeadNorm(cur.grad) / dtau < options.forceTolerance) {
863 inst.converged = true;
864 }
865
866 inst.path.reserve(static_cast<size_t>(P + 1));
867 inst.path.push_back(start);
868 inst.path.insert(inst.path.end(), x.begin(), x.end());
869 inst.path.push_back(end);
870 inst.energies.reserve(static_cast<size_t>(P + 1));
871 inst.energies.push_back(vEnds[0]);
872 inst.energies.insert(inst.energies.end(), cur.v.begin(), cur.v.end());
873 inst.energies.push_back(vEnds[1]);
874 const double sWell = betaHbar * 0.5 * (vEnds[0] + vEnds[1]);
875 inst.action = (cur.action - sWell) / kHbar;
876 double s0 = 0.0;
877 for (long j = 0; j < P; ++j) {
878 s0 += (inst.path[static_cast<size_t>(j + 1)] -
879 inst.path[static_cast<size_t>(j)])
880 .squaredNorm();
881 }
882 inst.s0 = s0 / dtau;
883 inst.symmetricEnough = std::abs(inst.asymmetry) * betaHbar / kHbar < 0.1;
884 return inst;
885}

887void instantonSplitting(Instanton &inst, const BeadHessian &hessian,
888 const MatrixXd &hessStart, const MatrixXd &hessEnd) {
889 const long P = static_cast<long>(inst.path.size()) - 1;
890 if (P < 4 || !(inst.dtau > 0.0)) {
891 throw std::invalid_argument("instantonSplitting: no optimised path");
892 }
893 const double dtau = inst.dtau;
894 const double c = 1.0 / dtau;
895 const long n = inst.path.front().size();
896 const MatrixXd spring = 2.0 * c * MatrixXd::Identity(n, n);
897
898 std::vector<MatrixXd> diag;
899 diag.reserve(static_cast<size_t>(P - 1));
900 for (long j = 1; j < P; ++j) {
901 const MatrixXd h = hessian(j, inst.path[static_cast<size_t>(j)]);
902 if (h.rows() != n || h.cols() != n) {
903 throw std::runtime_error("instantonSplitting: bead Hessian size");
904 }
905 diag.push_back(spring + dtau * 0.5 * (h + h.transpose()));
906 }
907 // The inertia comes from the Schur blocks (Haynsworth), so a path with
908 // two negative modes is not mistaken for a minimum by the determinant's
909 // sign.
910 const HaynsworthChain chain(c, diag, true);
911
912 auto wellLogDet = [&](const MatrixXd &h) {
913 const std::vector<MatrixXd> d(static_cast<size_t>(P - 1),
914 spring + dtau * 0.5 * (h + h.transpose()));
915 const HaynsworthChain well(c, d, true);
916 if (well.negative() > 0) {
917 throw std::runtime_error(
918 "instantonSplitting: a well Hessian is not positive definite");
919 }
920 return well.logAbsDet();
921 };
922 const double logDetWell = 0.5 * (wellLogDet(hessStart) + wellLogDet(hessEnd));
923
924 // The zero mode is the kink's translation in imaginary time, along the
925 // discrete velocity v; det' J = det J (v^T J^-1 v) for v its eigenvector.
926 std::vector<VectorXd> v(static_cast<size_t>(P - 1));
927 for (long j = 1; j < P; ++j) {
928 v[static_cast<size_t>(j - 1)] = inst.path[static_cast<size_t>(j + 1)] -
929 inst.path[static_cast<size_t>(j - 1)];
930 }
931 scale(v, 1.0 / std::sqrt(dot(v, v)));
932 const double vJv = dot(v, chain.solve(v));
933 // A zero mode just below zero is one negative eigenvalue the prime
934 // leaves out; any other negative eigenvalue is a second unstable mode.
935 if (chain.negative() - (vJv < 0.0 ? 1 : 0) != 0) {
936 throw std::runtime_error(
937 "instantonSplitting: the path is not a minimum of the action "
938 "(a negative mode besides the kink's translation)");
939 }
940 inst.zeroMode = 1.0 / vJv;
941 const double logDetPrime = chain.logAbsDet() + std::log(std::abs(vJv));
942
943 // Next eigenvalue: inverse iteration orthogonal to v.
944 std::vector<VectorXd> w(v.size());
945 for (size_t k = 0; k < w.size(); ++k) {
946 w[k].resize(n);
947 for (long i = 0; i < n; ++i) {
948 w[k](i) = std::sin(0.7 * static_cast<double>(k) +
949 1.3 * static_cast<double>(i) + 0.1);
950 }
951 }
952 double lambda1 = 0.0;
953 for (int it = 0; it < 40; ++it) {
954 const double proj = dot(v, w);
955 for (size_t k = 0; k < w.size(); ++k) {
956 w[k] -= proj * v[k];
957 }
958 scale(w, 1.0 / std::sqrt(dot(w, w)));
959 std::vector<VectorXd> z = chain.solve(w);
960 lambda1 = 1.0 / dot(w, z);
961 w = std::move(z);
962 }
963 inst.modeSeparation = std::abs(lambda1 / inst.zeroMode);
964
965 inst.delta0 = 2.0 * kHbar *
966 std::sqrt(inst.s0 / (2.0 * std::numbers::pi * kHbar * dtau)) *
967 std::exp(0.5 * (logDetWell - logDetPrime) - inst.action);
968}

989RingEval evaluateRing(const std::vector<VectorXd> &x, double c,
990 const BatchPotential &potential,
991 double energyShift = 0.0) {
992 RingEval out;
993 std::vector<VectorXd> gv;
994 potential(x, out.v, gv);
995 for (double &v : out.v) {
996 v -= energyShift;
997 }
998 const size_t n = x.size();
999 if (out.v.size() != n || gv.size() != n) {
1000 throw std::runtime_error(
1001 "rate instanton: potential returned the wrong count");
1002 }
1003 out.grad.resize(n);
1004 double spring = 0.0;
1005 for (size_t j = 0; j < n; ++j) {
1006 const VectorXd &prev = x[(j + n - 1) % n];
1007 const VectorXd &next = x[(j + 1) % n];
1008 out.grad[j] = gv[j] + c * (2.0 * x[j] - prev - next);
1009 spring += (next - x[j]).squaredNorm();
1010 out.u += out.v[j];
1011 }
1012 out.u += 0.5 * c * spring;
1013 return out;
1014}

1016// Lowest eigenpair of the ring Hessian by Lanczos on finite-difference
1017// products, started from `start`; full reorthogonalisation.
1018double lowestMode(const std::vector<VectorXd> &x, const RingEval &here,
1019 double c, const BatchPotential &potential,
1020 std::vector<VectorXd> &mode, long steps, double eps,
1021 bool mirror = false) {
1022 const size_t n = x.size();
1023 // The thermal instanton retraces, so the unstable mode is even. Probes
1024 // and products stay on that mirror and the potential call sees one half.
1025 const bool reflect = mirror && n % 2 == 0;
1026 auto snap = [&](std::vector<VectorXd> &q) {
1027 if (!reflect) {
1028 return;
1029 }
1030 const long m = static_cast<long>(n) / 2;
1031 for (long j = 1; j < m; ++j) {
1032 const size_t a = static_cast<size_t>(j);
1033 const size_t b = n - a;
1034 const VectorXd mid = 0.5 * (q[a] + q[b]);
1035 q[a] = mid;
1036 q[b] = mid;
1037 }
1038 };
1039 auto hv = [&](const std::vector<VectorXd> &u) {
1040 std::vector<VectorXd> xp(n);
1041 for (size_t j = 0; j < n; ++j) {
1042 xp[j] = x[j] + eps * u[j];
1043 }
1044 snap(xp);
1045 const RingEval e = evaluateRing(xp, c, potential);
1046 std::vector<VectorXd> out(n);
1047 for (size_t j = 0; j < n; ++j) {
1048 out[j] = (e.grad[j] - here.grad[j]) / eps;
1049 }
1050 snap(out);
1051 return out;
1052 };
1053 std::vector<std::vector<VectorXd>> basis;
1054 std::vector<double> alpha, beta;
1055 std::vector<VectorXd> q = mode;
1056 snap(q);
1057 scale(q, 1.0 / std::sqrt(dot(q, q)));
1058 for (long k = 0; k < steps; ++k) {
1059 basis.push_back(q);
1060 std::vector<VectorXd> w = hv(q);
1061 const double a = dot(w, q);
1062 alpha.push_back(a);
1063 for (const auto &b : basis) {
1064 const double p = dot(w, b);
1065 for (size_t j = 0; j < n; ++j) {
1066 w[j] -= p * b[j];
1067 }
1068 }
1069 snap(w);
1070 const double bnorm = std::sqrt(dot(w, w));
1071 if (!(bnorm > 1e-12) || k + 1 == steps) {
1072 break;
1073 }
1074 beta.push_back(bnorm);
1075 scale(w, 1.0 / bnorm);
1076 q = std::move(w);
1077 }
1078 const long m = static_cast<long>(alpha.size());
1079 MatrixXd t = MatrixXd::Zero(m, m);
1080 for (long i = 0; i < m; ++i) {
1081 t(i, i) = alpha[static_cast<size_t>(i)];
1082 if (i + 1 < m) {
1083 t(i, i + 1) = t(i + 1, i) = beta[static_cast<size_t>(i)];
1084 }
1085 }
1086 const Eigen::SelfAdjointEigenSolver<MatrixXd> es(t);
1087 const VectorXd y = es.eigenvectors().col(0);
1088 std::vector<VectorXd> ritz(n);
1089 for (size_t j = 0; j < n; ++j) {
1090 ritz[j] = VectorXd::Zero(x[j].size());
1091 }
1092 for (long i = 0; i < m; ++i) {
1093 for (size_t j = 0; j < n; ++j) {
1094 ritz[j] += y(i) * basis[static_cast<size_t>(i)][j];
1095 }
1096 }
1097 scale(ritz, 1.0 / std::sqrt(dot(ritz, ritz)));
1098 snap(ritz);
1099 const double rnorm = std::sqrt(dot(ritz, ritz));
1100 if (rnorm > 0.0) {
1101 scale(ritz, 1.0 / rnorm);
1102 }
1103 mode = std::move(ritz);
1104 return es.eigenvalues()(0);
1105}

1107// Lowest curvature of the ring Hessian over vectors odd under j -> N-j and
1108// orthogonal to the cycle mode x[j+1] - x[j-1], at a mirror-symmetric ring
1109// of even N. A search confined to the even sector cannot see these. The
1110// products move the ring off the mirror, so `potential` evaluates every bead.
1111// Returns +inf when no odd direction is left to probe.
1112double lowestOddMode(const std::vector<VectorXd> &x, const RingEval &here,
1113 double c, const BatchPotential &potential,
1114 std::vector<VectorXd> &mode, long steps, double eps) {
1115 const size_t n = x.size();
1116 const size_t m = n / 2;
1117 std::vector<VectorXd> cycle(n);
1118 for (size_t j = 0; j < n; ++j) {
1119 cycle[j] = x[(j + 1) % n] - x[(j + n - 1) % n];
1120 }
1121 auto odd = [&](std::vector<VectorXd> &q) {
1122 q[0].setZero();
1123 q[m].setZero();
1124 for (size_t a = 1; a < m; ++a) {
1125 const VectorXd half = 0.5 * (q[a] - q[n - a]);
1126 q[a] = half;
1127 q[n - a] = -half;
1128 }
1129 };
1130 odd(cycle);
1131 const double cnorm = std::sqrt(dot(cycle, cycle));
1132 if (cnorm > 0.0) {
1133 scale(cycle, 1.0 / cnorm);
1134 }
1135 auto project = [&](std::vector<VectorXd> &q) {
1136 odd(q);
1137 if (cnorm > 0.0) {
1138 const double p = dot(q, cycle);
1139 for (size_t j = 0; j < n; ++j) {
1140 q[j] -= p * cycle[j];
1141 }
1142 }
1143 };
1144 auto hv = [&](const std::vector<VectorXd> &u) {
1145 std::vector<VectorXd> xp(n);
1146 for (size_t j = 0; j < n; ++j) {
1147 xp[j] = x[j] + eps * u[j];
1148 }
1149 const RingEval e = evaluateRing(xp, c, potential);
1150 std::vector<VectorXd> out(n);
1151 for (size_t j = 0; j < n; ++j) {
1152 out[j] = (e.grad[j] - here.grad[j]) / eps;
1153 }
1154 project(out);
1155 return out;
1156 };
1157 // Start from the bead displacements with opposite signs on the two
1158 // halves: two copies of the turning region moving against each other.
1159 VectorXd mean = VectorXd::Zero(x[0].size());
1160 for (const auto &b : x) {
1161 mean += b;
1162 }
1163 mean /= static_cast<double>(n);
1164 std::vector<VectorXd> q(n);
1165 for (size_t j = 0; j < n; ++j) {
1166 q[j] = (j < m ? 1.0 : -1.0) * (x[j] - mean);
1167 }
1168 project(q);
1169 const double qnorm = std::sqrt(dot(q, q));
1170 if (!(qnorm > 1e-12)) {
1171 return std::numeric_limits<double>::infinity();
1172 }
1173 scale(q, 1.0 / qnorm);
1174 std::vector<std::vector<VectorXd>> basis;
1175 std::vector<double> alpha, beta;
1176 for (long k = 0; k < steps; ++k) {
1177 basis.push_back(q);
1178 std::vector<VectorXd> w = hv(q);
1179 alpha.push_back(dot(w, q));
1180 for (const auto &b : basis) {
1181 const double p = dot(w, b);
1182 for (size_t j = 0; j < n; ++j) {
1183 w[j] -= p * b[j];
1184 }
1185 }
1186 project(w);
1187 const double bnorm = std::sqrt(dot(w, w));
1188 if (!(bnorm > 1e-12) || k + 1 == steps) {
1189 break;
1190 }
1191 beta.push_back(bnorm);
1192 scale(w, 1.0 / bnorm);
1193 q = std::move(w);
1194 }
1195 const long dim = static_cast<long>(alpha.size());
1196 MatrixXd t = MatrixXd::Zero(dim, dim);
1197 for (long i = 0; i < dim; ++i) {
1198 t(i, i) = alpha[static_cast<size_t>(i)];
1199 if (i + 1 < dim) {
1200 t(i, i + 1) = t(i + 1, i) = beta[static_cast<size_t>(i)];
1201 }
1202 }
1203 const Eigen::SelfAdjointEigenSolver<MatrixXd> es(t);
1204 const VectorXd y = es.eigenvectors().col(0);
1205 mode.assign(n, VectorXd::Zero(x[0].size()));
1206 for (long i = 0; i < dim; ++i) {
1207 for (size_t j = 0; j < n; ++j) {
1208 mode[j] += y(i) * basis[static_cast<size_t>(i)][j];
1209 }
1210 }
1211 project(mode);
1212 scale(mode, 1.0 / std::sqrt(dot(mode, mode)));
1213 return es.eigenvalues()(0);
1214}

1216// Distance along +dir (sign = 1) or -dir (sign = -1) from the saddle at which
1217// V has dropped by `drop`, from a scan in steps of h; the lowest point found
1218// when V never drops that far.
1219double turningDistance(const VectorXd &saddle, const VectorXd &dir,
1220 double vSaddle, double drop, double sign, double h,
1221 long maxPoints, const BatchPotential &potential) {
1222 std::vector<VectorXd> pts;
1223 for (long k = 1; k <= maxPoints; ++k) {
1224 pts.push_back(saddle + sign * h * static_cast<double>(k) * dir);
1225 }
1226 std::vector<double> v;
1227 std::vector<VectorXd> g;
1228 potential(pts, v, g);
1229 double prevV = vSaddle, prevS = 0.0, best = 0.0, bestV = vSaddle;
1230 for (long k = 0; k < maxPoints; ++k) {
1231 const double sk = h * static_cast<double>(k + 1);
1232 const double vk = v[static_cast<size_t>(k)];
1233 if (vk <= vSaddle - drop) {
1234 const double t = (prevV - (vSaddle - drop)) / (prevV - vk);
1235 return prevS + t * (sk - prevS);
1236 }
1237 if (vk < bestV) {
1238 bestV = vk;
1239 best = sk;
1240 }
1241 if (vk > prevV + 1e-12 && k > 0) {
1242 break; // past the minimum on this side
1243 }
1244 prevV = vk;
1245 prevS = sk;
1246 }
1247 return best;
1248}

1250double sideDrop(const VectorXd &saddle, const VectorXd &dir, double vSaddle,
1251 double sign, double h, long maxPoints,
1252 const BatchPotential &potential) {
1253 std::vector<VectorXd> pts;
1254 for (long k = 1; k <= maxPoints; ++k) {
1255 pts.push_back(saddle + sign * h * static_cast<double>(k) * dir);
1256 }
1257 std::vector<double> v;
1258 std::vector<VectorXd> g;
1259 potential(pts, v, g);
1260 double lowest = vSaddle;
1261 for (long k = 0; k < maxPoints; ++k) {
1262 const double vk = v[static_cast<size_t>(k)];
1263 if (vk > lowest + 1e-12 && k > 0 && lowest < vSaddle) {
1264 return vSaddle - lowest; // found this side's minimum
1265 }
1266 lowest = std::min(lowest, vk);
1267 }
1268 return vSaddle - lowest; // still falling: the drop to the last point
1269}

1277// Lowest Ritz pairs of a symmetric ring operator, full reorthogonalisation.
1278// `steps` at the dimension is the whole spectrum.
1279std::vector<RingMode> lowestRingModes(
1280 const std::function<std::vector<VectorXd>(const std::vector<VectorXd> &)>
1281 &apply,
1282 std::vector<VectorXd> start, long steps) {
1283 const size_t n = start.size();
1284 const long f = start.empty() ? 0 : start.front().size();
1285 const double n0 = std::sqrt(dot(start, start));
1286 if (!(n0 > 0.0) || f < 1) {
1287 throw std::runtime_error("instantonRate: Lanczos was given a zero vector");
1288 }
1289 scale(start, 1.0 / n0);
1290 std::vector<std::vector<VectorXd>> basis;
1291 std::vector<double> alpha;
1292 std::vector<double> beta;
1293 std::vector<VectorXd> q = std::move(start);
1294 for (long k = 0; k < steps; ++k) {
1295 basis.push_back(q);
1296 std::vector<VectorXd> w = apply(q);
1297 alpha.push_back(dot(w, q));
1298 for (int pass = 0; pass < 2; ++pass) {
1299 for (const auto &b : basis) {
1300 const double p = dot(w, b);
1301 for (size_t j = 0; j < n; ++j) {
1302 w[j] -= p * b[j];
1303 }
1304 }
1305 }
1306 const double bnorm = std::sqrt(dot(w, w));
1307 if (!(bnorm > 1e-14) || k + 1 == steps) {
1308 break;
1309 }
1310 beta.push_back(bnorm);
1311 scale(w, 1.0 / bnorm);
1312 q = std::move(w);
1313 }
1314 const long m = static_cast<long>(alpha.size());
1315 MatrixXd tridiag = MatrixXd::Zero(m, m);
1316 for (long i = 0; i < m; ++i) {
1317 tridiag(i, i) = alpha[static_cast<size_t>(i)];
1318 if (i + 1 < m) {
1319 tridiag(i, i + 1) = tridiag(i + 1, i) = beta[static_cast<size_t>(i)];
1320 }
1321 }
1322 const Eigen::SelfAdjointEigenSolver<MatrixXd> es(tridiag);
1323 std::vector<RingMode> modes;
1324 modes.reserve(static_cast<size_t>(m));
1325 for (long i = 0; i < m; ++i) {
1326 RingMode mode;
1327 mode.theta = es.eigenvalues()(i);
1328 mode.vector.assign(n, VectorXd::Zero(f));
1329 const VectorXd y = es.eigenvectors().col(i);
1330 for (long s = 0; s < m; ++s) {
1331 for (size_t j = 0; j < n; ++j) {
1332 mode.vector[j] += y(s) * basis[static_cast<size_t>(s)][j];
1333 }
1334 }
1335 scale(mode.vector, 1.0 / std::sqrt(dot(mode.vector, mode.vector)));
1336 const std::vector<VectorXd> applied = apply(mode.vector);
1337 double residual = 0.0;
1338 for (size_t j = 0; j < n; ++j) {
1339 residual += (applied[j] - mode.theta * mode.vector[j]).squaredNorm();
1340 }
1341 mode.residual = std::sqrt(residual);
1342 modes.push_back(std::move(mode));
1343 }
1344 return modes;
1345}

1347// Factored cyclic ring Hessian: the open chain plus the corner coupling
1348// M between bead 0 and bead N-1.
1349// det(H_open + P M P^T) = det(H_open) det(M) det(M^{-1} + P^T H_open^{-1} P),
1350// and |det M| = c^{2f}. A singular ring makes the last factor zero.
1351struct CyclicFactor {
1352 BlockChain open;
1353 Eigen::PartialPivLU<ColMajorXd> cornerLu;
1354 double c;
1355 long f;
1356 long n;
1357 double logAbs;
1358 bool singular;
1359
1360 CyclicFactor(double cIn, const std::vector<MatrixXd> &diag)
1361 : open(cIn, diag),
1362 c(cIn),
1363 f(diag.front().rows()),
1364 n(static_cast<long>(diag.size())),
1365 logAbs(0.0),
1366 singular(false) {
1367 const MatrixXd eye = MatrixXd::Identity(f, f);
1368 const MatrixXd zero = MatrixXd::Zero(f, f);
1369 std::vector<MatrixXd> rhs(static_cast<size_t>(n), zero);
1370 rhs.front() = eye;
1371 const std::vector<MatrixXd> fromFirst = open.solve(rhs);
1372 rhs.front() = zero;
1373 rhs.back() = eye;
1374 const std::vector<MatrixXd> fromLast = open.solve(rhs);
1375 MatrixXd corner(2 * f, 2 * f);
1376 corner.topLeftCorner(f, f) = fromFirst.front();
1377 corner.bottomLeftCorner(f, f) = fromFirst.back();
1378 corner.topRightCorner(f, f) = fromLast.front();
1379 corner.bottomRightCorner(f, f) = fromLast.back();
1380 corner.topRightCorner(f, f) -= eye / c;
1381 corner.bottomLeftCorner(f, f) -= eye / c;
1382 cornerLu.compute(ColMajorXd(corner));
1383 logAbs = open.logAbsDet() + 2.0 * static_cast<double>(f) * std::log(c);
1384 const MatrixXd &upper = cornerLu.matrixLU();
1385 for (long i = 0; i < upper.rows(); ++i) {
1386 const double pivot = upper(i, i);
1387 if (pivot == 0.0) {
1388 singular = true;
1389 logAbs = -std::numeric_limits<double>::infinity();
1390 return;
1391 }
1392 logAbs += std::log(std::abs(pivot));
1393 }
1394 }
1395
1396 std::vector<VectorXd> solve(const std::vector<VectorXd> &rhs) const {
1397 if (singular || static_cast<long>(rhs.size()) != n) {
1398 throw std::runtime_error("cyclic ring: singular");
1399 }
1400 std::vector<VectorXd> y = open.solve(rhs);
1401 VectorXd g(2 * f);
1402 g.head(f) = y.front();
1403 g.tail(f) = y.back();
1404 const VectorXd z = cornerLu.solve(g);
1405 std::vector<VectorXd> bump(static_cast<size_t>(n), VectorXd::Zero(f));
1406 bump.front() = z.head(f);
1407 bump.back() = z.tail(f);
1408 const std::vector<VectorXd> corr = open.solve(bump);
1409 for (long j = 0; j < n; ++j) {
1410 y[static_cast<size_t>(j)] -= corr[static_cast<size_t>(j)];
1411 }
1412 return y;
1413 }
1414};

1416// J = T + G K G^T over the chain T: the closure blocks (-c I between the
1417// last bead and the first) when `closed`, and symmetric rank-one terms
1418// kappa_i u_i u_i^T. Woodbury gives J^{-1} b, the determinant lemma
1419// ln|det J| and Haynsworth the inertia, all O(N f^3), so the N f by N f
1420// matrix is never formed.
1421class WoodburyRing {
1422public:
1423 WoodburyRing(double c, const std::vector<MatrixXd> &diag, bool closed,
1424 const std::vector<std::vector<VectorXd>> &extras,
1425 const std::vector<double> &kappas, bool spectrum)
1426 : c_(c),
1427 n_(static_cast<long>(diag.size())),
1428 f_(diag.front().rows()),
1429 closed_(closed),
1430 chain_(c, diag, spectrum),
1431 extras_(extras) {
1432 const long base = closed ? 2 * f_ : 0;
1433 const long m = base + static_cast<long>(extras.size());
1434 kinv_ = MatrixXd::Zero(m, m);
1435 MatrixXd k = MatrixXd::Zero(m, m);
1436 if (closed) {
1437 kinv_.block(0, f_, f_, f_) = -MatrixXd::Identity(f_, f_) / c;
1438 kinv_.block(f_, 0, f_, f_) = -MatrixXd::Identity(f_, f_) / c;
1439 k.block(0, f_, f_, f_) = -c * MatrixXd::Identity(f_, f_);
1440 k.block(f_, 0, f_, f_) = -c * MatrixXd::Identity(f_, f_);
1441 }
1442 for (size_t i = 0; i < extras.size(); ++i) {
1443 const long r = base + static_cast<long>(i);
1444 kinv_(r, r) = 1.0 / kappas[i];
1445 k(r, r) = kappas[i];
1446 }
1447 if (m == 0) {
1448 ok_ = true;
1449 logAbsDet_ = chain_.logAbsDet();
1450 negative_ = chain_.negative();
1451 return;
1452 }
1453 gtg_ = MatrixXd::Zero(m, m);
1454 for (long col = 0; col < m; ++col) {
1455 gtg_.col(col) = pieces(chain_.solve(column(col)));
1456 }
1457 woodbury_.compute(ColMajorXd(kinv_ + gtg_));
1458 ok_ = gtg_.array().isFinite().all();
1459 if (spectrum) {
1460 const Eigen::PartialPivLU<ColMajorXd> lu(
1461 ColMajorXd(MatrixXd::Identity(m, m) + k * gtg_));
1462 double logDet = 0.0;
1463 for (long i = 0; i < m; ++i) {
1464 const double u = lu.matrixLU()(i, i);
1465 if (u == 0.0) {
1466 ok_ = false;
1467 logAbsDet_ = -std::numeric_limits<double>::infinity();
1468 return;
1469 }
1470 logDet += std::log(std::abs(u));
1471 }
1472 logAbsDet_ = chain_.logAbsDet() + logDet;
1473 // neg(J) = neg(T) + neg(S) - neg(-K^{-1}), S = -K^{-1} - G^T T^{-1} G.
1474 const ColMajorXd sMat = -kinv_ - gtg_;
1475 const ColMajorXd sSym = 0.5 * (sMat + sMat.transpose());
1476 const Eigen::SelfAdjointEigenSolver<ColMajorXd> es(
1477 sSym, Eigen::EigenvaluesOnly);
1478 const ColMajorXd kn = -kinv_;
1479 const Eigen::SelfAdjointEigenSolver<ColMajorXd> ek(
1480 kn, Eigen::EigenvaluesOnly);
1481 negative_ = chain_.negative() + (es.eigenvalues().array() < 0.0).count() -
1482 (ek.eigenvalues().array() < 0.0).count();
1483 }
1484 }
1485 bool ok() const { return ok_; }
1486 std::vector<VectorXd> solve(const std::vector<VectorXd> &b) const {
1487 std::vector<VectorXd> tb = chain_.solve(b);
1488 if (kinv_.rows() == 0) {
1489 return tb;
1490 }
1491 const VectorXd y = woodbury_.solve(pieces(tb));
1492 const std::vector<VectorXd> gy = chain_.solve(expand(y));
1493 for (size_t j = 0; j < tb.size(); ++j) {
1494 tb[j] -= gy[j];
1495 }
1496 return tb;
1497 }
1498 double logAbsDet() const { return logAbsDet_; }
1499 long negative() const { return negative_; }
1500
1501private:
1502 std::vector<VectorXd> column(long col) const {
1503 const long base = closed_ ? 2 * f_ : 0;
1504 if (col < base) {
1505 std::vector<VectorXd> g(static_cast<size_t>(n_), VectorXd::Zero(f_));
1506 g[static_cast<size_t>(col < f_ ? 0 : n_ - 1)](col % f_) = 1.0;
1507 return g;
1508 }
1509 return extras_[static_cast<size_t>(col - base)];
1510 }
1511 VectorXd pieces(const std::vector<VectorXd> &x) const {
1512 const long base = closed_ ? 2 * f_ : 0;
1513 VectorXd out(base + static_cast<long>(extras_.size()));
1514 if (closed_) {
1515 out.head(f_) = x[0];
1516 out.segment(f_, f_) = x[static_cast<size_t>(n_ - 1)];
1517 }
1518 for (size_t i = 0; i < extras_.size(); ++i) {
1519 out(base + static_cast<long>(i)) = dot(extras_[i], x);
1520 }
1521 return out;
1522 }
1523 std::vector<VectorXd> expand(const VectorXd &y) const {
1524 const long base = closed_ ? 2 * f_ : 0;
1525 std::vector<VectorXd> g(static_cast<size_t>(n_), VectorXd::Zero(f_));
1526 if (closed_) {
1527 g[0] += y.head(f_);
1528 g[static_cast<size_t>(n_ - 1)] += y.segment(f_, f_);
1529 }
1530 for (size_t i = 0; i < extras_.size(); ++i) {
1531 const double w = y(base + static_cast<long>(i));
1532 for (long j = 0; j < n_; ++j) {
1533 g[static_cast<size_t>(j)] += w * extras_[i][static_cast<size_t>(j)];
1534 }
1535 }
1536 return g;
1537 }
1538
1539 double c_;
1540 long n_, f_;
1541 bool closed_;
1542 HaynsworthChain chain_;
1543 std::vector<std::vector<VectorXd>> extras_;
1544 MatrixXd kinv_, gtg_;
1545 Eigen::PartialPivLU<ColMajorXd> woodbury_;
1546 bool ok_ = false;
1547 double logAbsDet_ = 0.0;
1548 long negative_ = 0;
1549};

1551// Diagonal blocks of the closed ring (H + 2 c I) or of the half chain from
1552// one turning point to the other (end blocks H / 2 + c I).
1553std::vector<MatrixXd> ringDiagonal(const std::vector<MatrixXd> &physical,
1554 double c, bool half) {
1555 const long beads = static_cast<long>(physical.size());
1556 const long f = physical.front().rows();
1557 std::vector<MatrixXd> diag(static_cast<size_t>(beads));
1558 for (long j = 0; j < beads; ++j) {
1559 const bool end = half && (j == 0 || j + 1 == beads);
1560 MatrixXd block = 0.5 * (physical[static_cast<size_t>(j)] +
1561 physical[static_cast<size_t>(j)].transpose());
1562 if (end) {
1563 block *= 0.5;
1564 }
1565 block += (end ? c : 2.0 * c) * MatrixXd::Identity(f, f);
1566 diag[static_cast<size_t>(j)] = block;
1567 }
1568 return diag;
1569}

1571// (diag - c neighbours) v for the closed ring or the open half chain.
1572std::vector<VectorXd> applyDiagonal(const std::vector<MatrixXd> &diag, double c,
1573 bool closed,
1574 const std::vector<VectorXd> &v) {
1575 const size_t n = v.size();
1576 std::vector<VectorXd> out(n);
1577 for (size_t j = 0; j < n; ++j) {
1578 out[j] = diag[j] * v[j];
1579 if (j > 0 || closed) {
1580 out[j] -= c * v[(j + n - 1) % n];
1581 }
1582 if (j + 1 < n || closed) {
1583 out[j] -= c * v[(j + 1) % n];
1584 }
1585 }
1586 return out;
1587}

1657// Central difference of dV/dq. One batch of 2 f displaced beads.
1658MatrixXd fdPhysicalHessian(const VectorXd &q, const BatchPotential &potential,
1659 double eps) {
1660 const long f = q.size();
1661 std::vector<VectorXd> pts;
1662 pts.reserve(static_cast<size_t>(2 * f));
1663 for (long a = 0; a < f; ++a) {
1664 VectorXd qp = q;
1665 VectorXd qm = q;
1666 qp[a] += eps;
1667 qm[a] -= eps;
1668 pts.push_back(std::move(qp));
1669 pts.push_back(std::move(qm));
1670 }
1671 std::vector<double> v;
1672 std::vector<VectorXd> g;
1673 potential(pts, v, g);
1674 if (static_cast<long>(g.size()) != 2 * f) {
1675 throw std::runtime_error(
1676 "rate instanton: the Hessian sample returned the wrong count");
1677 }
1678 MatrixXd h(f, f);
1679 for (long a = 0; a < f; ++a) {
1680 h.col(a) =
1681 (g[static_cast<size_t>(2 * a)] - g[static_cast<size_t>(2 * a + 1)]) /
1682 (2.0 * eps);
1683 }
1684 return (0.5 * (h + h.transpose())).eval();
1685}

1687// Bofill mix of Powell's symmetric Broyden and SR1. H is d2V/dq2.
1688void bofillUpdate(MatrixXd &h, const VectorXd &dq, const VectorXd &dg) {
1689 const double dq2 = dq.squaredNorm();
1690 if (!(dq2 > 1e-24)) {
1691 return;
1692 }
1693 const VectorXd r = dg - h * dq;
1694 const double rq = r.dot(dq);
1695 const double r2 = r.squaredNorm();
1696 const double phi = r2 * dq2 > 1e-30 ? (rq * rq) / (r2 * dq2) : 0.0;
1697 const double inv = 1.0 / dq2;
1698 h.noalias() += (1.0 - phi) * inv *
1699 (r * dq.transpose() + dq * r.transpose() -
1700 (rq * inv) * (dq * dq.transpose()));
1701 if (std::abs(rq) > 1e-12 * std::sqrt(std::max(0.0, r2 * dq2))) {
1702 h.noalias() += (phi / rq) * (r * r.transpose());
1703 }
1704 h = (0.5 * (h + h.transpose())).eval();
1705}

1783// Cosine between the turning points, opened by (1 - T / Tc) of the lower
1784// barrier. The far turning point sits on bead 0.
1785std::vector<VectorXd> cosineSeed(const VectorXd &saddle, const VectorXd &dir,
1786 double lambda0, double temperature,
1787 double crossover, long nBeads,
1788 const BatchPotential &potential) {

1791 potential({saddle}, v0, g0);
1792 const double vS = v0.at(0);
1793 const double h = 0.25 * std::sqrt(2.0 * kBoltzmann * crossover / -lambda0);
1794 const long pts = 200;
1795 const double dPlus = sideDrop(saddle, dir, vS, 1.0, h, pts, potential);
1796 const double dMinus = sideDrop(saddle, dir, vS, -1.0, h, pts, potential);
1797 const double dMin = std::min(dPlus, dMinus);

1799 (dMin > 0.0 ? dMin : kBoltzmann * crossover);
1800 const double sPlus =
1801 turningDistance(saddle, dir, vS, drop, 1.0, h, pts, potential);
1802 const double sMinus =
1803 turningDistance(saddle, dir, vS, drop, -1.0, h, pts, potential);
1804 std::vector<VectorXd> guess(static_cast<size_t>(nBeads));
1805 for (long j = 0; j < nBeads; ++j) {

1807 static_cast<double>(nBeads));
1808 guess[static_cast<size_t>(j)] =
1809 saddle + dir * (ct >= 0.0 ? sPlus * ct : sMinus * ct);
1810 }
1811 return guess;

1814// Index-1 Newton step through the block chain: a negative climb eigenvalue
1815// stays and a raw Newton step climbs it; a positive one is flipped, as is
1816// every other negative Ritz value off the cycle; a tiny one is parked at a
1817// spring-sized curvature; the cycle itself is held with a spring-sized
1818// curvature and its component removed from the step. Each flip is a
1819// rank-one term in the Woodbury correction, so the solve stays O(N f^3).
1820VectorXd chainIndexOneStep(const std::vector<RingMode> &ritz,
1821 const Climb &climb,
1822 const std::vector<MatrixXd> &diag, double spring,
1823 bool closed, const std::vector<VectorXd> &grad,
1824 const VectorXd &tau,
1825 const std::vector<std::vector<VectorXd>> &nullRing) {
1826 if (climb.index < 0 || ritz.empty() || grad.empty()) {
1827 return VectorXd();
1828 }
1829 const long f = grad.front().size();
1830 const long dim = static_cast<long>(grad.size()) * f;
1831 const double cut = -1e-8 * std::max(1.0, spring);
1832 const double tiny = 1e-8 * std::max(1.0, spring);
1833 const double parked = std::max(1.0, spring);
1834 std::vector<std::vector<VectorXd>> extras;
1835 std::vector<double> kappas;
1836 std::vector<VectorXd> tauRing;
1837 if (tau.size() == dim) {
1838 tauRing.assign(grad.size(), VectorXd::Zero(f));
1839 for (size_t j = 0; j < grad.size(); ++j) {
1840 tauRing[j] = tau.segment(static_cast<long>(j) * f, f);
1841 }
1842 // On a discrete ring the time shift has a small curvature of its own,
1843 // and the stationary ring is where Newton on that curvature leads; the
1844 // lift is only for a cycle too flat to solve with.
1845 double cycleCurvature = 0.0;
1846 for (const auto &m : ritz) {
1847 if (std::abs(dot(m.vector, tauRing)) > 0.5) {
1848 cycleCurvature = m.theta;
1849 break;
1850 }
1851 }
1852 if (std::abs(cycleCurvature) <= 1e-6 * std::max(1.0, spring)) {
1853 extras.push_back(tauRing);
1854 kappas.push_back(spring);
1855 }
1856 }
1857 // The rigid ring motions are lifted like the cycle.
1858 auto onNull = [&](const std::vector<VectorXd> &m) {
1859 for (const auto &r : nullRing) {
1860 if (std::abs(dot(m, r)) > 0.5) {
1861 return true;
1862 }
1863 }
1864 return false;
1865 };
1866 for (const auto &r : nullRing) {
1867 extras.push_back(r);
1868 kappas.push_back(spring);
1869 }
1870 for (size_t i = 0; i < ritz.size(); ++i) {
1871 if (!tauRing.empty() && std::abs(dot(ritz[i].vector, tauRing)) > 0.5) {
1872 continue;
1873 }
1874 if (onNull(ritz[i].vector)) {
1875 continue;
1876 }
1877 const double li = ritz[i].theta;
1878 const bool isClimb = static_cast<long>(i) == climb.index;
1879 if ((isClimb && li > 0.0) || (!isClimb && li < cut)) {
1880 extras.push_back(ritz[i].vector);
1881 kappas.push_back(-2.0 * li);
1882 } else if (!isClimb && std::abs(li) <= tiny) {
1883 extras.push_back(ritz[i].vector);
1884 kappas.push_back(parked - li);
1885 }
1886 }
1887 std::vector<VectorXd> rhs = grad;
1888 scale(rhs, -1.0);
1889 // The operator the step solves with: the ring plus every rank-one term.
1890 auto applyShifted = [&](const std::vector<VectorXd> &x) {
1891 std::vector<VectorXd> out = applyDiagonal(diag, spring, closed, x);
1892 for (size_t i = 0; i < extras.size(); ++i) {
1893 const double w = kappas[i] * dot(extras[i], x);
1894 for (size_t j = 0; j < out.size(); ++j) {
1895 out[j] += w * extras[i][j];
1896 }
1897 }
1898 return out;
1899 };
1900 const double rhsNorm = std::sqrt(dot(rhs, rhs));
1901 std::vector<VectorXd> stepRing;
1902 bool solved = false;
1903 try {
1904 if (dim <= kDenseRing) {
1905 throw std::runtime_error("small ring: dense solve");
1906 }
1907 const WoodburyRing ring(spring, diag, closed, extras, kappas, false);
1908 if (ring.ok()) {
1909 stepRing = ring.solve(rhs);
1910 // Cutting the ring open can leave a Schur pivot near zero next to
1911 // the barrier, and the Woodbury solve then loses digits; iterative
1912 // refinement against the exact operator recovers them.
1913 for (int pass = 0; pass < 4; ++pass) {
1914 std::vector<VectorXd> r = applyShifted(stepRing);
1915 for (size_t j = 0; j < r.size(); ++j) {
1916 r[j] = rhs[j] - r[j];
1917 }
1918 const double rn = std::sqrt(dot(r, r));
1919 if (!std::isfinite(rn)) {
1920 break;
1921 }
1922 if (rn <= 1e-10 * std::max(1.0, rhsNorm)) {
1923 solved = true;
1924 break;
1925 }
1926 const std::vector<VectorXd> dx = ring.solve(r);
1927 for (size_t j = 0; j < stepRing.size(); ++j) {
1928 stepRing[j] += dx[j];
1929 }
1930 }
1931 }
1932 } catch (const std::runtime_error &) {
1933 solved = false;
1934 }
1935 if (!solved && dim <= kDenseRing) {
1936 // A small ring is cheap to solve densely when the chain cannot.
1937 const long n = static_cast<long>(grad.size());
1938 ColMajorXd jt = ColMajorXd::Zero(dim, dim);
1939 for (long col = 0; col < dim; ++col) {
1940 std::vector<VectorXd> e(static_cast<size_t>(n), VectorXd::Zero(f));
1941 e[static_cast<size_t>(col / f)](col % f) = 1.0;
1942 jt.col(col) = packBeads(applyShifted(e));
1943 }
1944 const Eigen::PartialPivLU<ColMajorXd> lu(jt);
1945 const VectorXd x = lu.solve(packBeads(rhs));
1946 stepRing.assign(static_cast<size_t>(n), VectorXd::Zero(f));
1947 for (long j = 0; j < n; ++j) {
1948 stepRing[static_cast<size_t>(j)] = x.segment(j * f, f);
1949 }
1950 solved = x.array().isFinite().all();
1951 }
1952 if (!solved && stepRing.empty()) {
1953 return VectorXd();
1954 }
1955 // The cycle component stays: a gradient along it has to be stepped out.
1956 VectorXd step = packBeads(stepRing);
1957 for (const auto &r : nullRing) {
1958 const VectorXd rf = packBeads(r);
1959 if (rf.size() == step.size()) {
1960 step -= step.dot(rf) * rf;
1961 }
1962 }
1963 if (!step.array().isFinite().all()) {
1964 return VectorXd();
1965 }
1966 return step;
1967}

1969// Non-increasing isotonic regression (pool adjacent violators).
1970std::vector<double> pavaNonIncreasing(const std::vector<double> &values) {
1971 const long n = static_cast<long>(values.size());
1972 struct Block {
1973 long start;
1974 long end;
1975 double average;
1976 double weight;
1977 };
1978 std::vector<Block> blocks;
1979 blocks.reserve(static_cast<size_t>(n));
1980 for (long i = 0; i < n; ++i) {
1981 blocks.push_back(Block{i, i, -values[static_cast<size_t>(i)], 1.0});
1982 while (blocks.size() >= 2 &&
1983 blocks[blocks.size() - 2].average > blocks.back().average) {
1984 const Block b = blocks.back();
1985 blocks.pop_back();
1986 Block &a = blocks.back();
1987 const double w = a.weight + b.weight;
1988 a.average = (a.average * a.weight + b.average * b.weight) / w;
1989 a.end = b.end;
1990 a.weight = w;
1991 }
1992 }
1993 std::vector<double> out(static_cast<size_t>(n));
1994 for (const Block &b : blocks) {
1995 for (long i = b.start; i <= b.end; ++i) {
1996 out[static_cast<size_t>(i)] = -b.average;
1997 }
1998 }
1999 return out;
2000}

2003// monotone.
2004// The sense follows the two ends, so an arbitrary eigenvector sign is harmless.
2005void projectMonotonePrefix(std::vector<VectorXd> &q, long last,

2008 return;

2025 v = -v;

2040std::vector<std::vector<VectorXd>>
2041ringRigidBasis(const std::vector<VectorXd> &q,
2042 const std::vector<double> &sqrtMasses, const VectorXd &reference,
2043 const std::array<bool, 3> &rotations) {
2044 std::vector<std::vector<VectorXd>> out;
2045 const long nAtoms = static_cast<long>(sqrtMasses.size());
2046 if (nAtoms == 0 || q.empty() || reference.size() != 3 * nAtoms ||
2047 q.front().size() != 3 * nAtoms) {
2048 return out;
2049 }
2050 const long nb = static_cast<long>(q.size());
2051 // Cartesian positions and the ring's centre of mass.
2052 Eigen::Vector3d centre = Eigen::Vector3d::Zero();
2053 double total = 0.0;
2054 for (long j = 0; j < nb; ++j) {
2055 for (long k = 0; k < nAtoms; ++k) {
2056 const double sm = sqrtMasses[static_cast<size_t>(k)];
2057 const Eigen::Vector3d r =
2058 reference.segment<3>(3 * k) +
2059 q[static_cast<size_t>(j)].segment<3>(3 * k) / sm;
2060 centre += sm * sm * r;
2061 total += sm * sm;
2062 }
2063 }
2064 centre /= total;
2065 std::vector<int> kinds{0, 1, 2};
2066 for (int c = 0; c < 3; ++c) {
2067 if (rotations[static_cast<size_t>(c)]) {
2068 kinds.push_back(3 + c);
2069 }
2070 }
2071 const long dim = nb * 3 * nAtoms;
2072 MatrixXd g = MatrixXd::Zero(dim, static_cast<long>(kinds.size()));
2073 for (size_t col = 0; col < kinds.size(); ++col) {
2074 const int kind = kinds[col];
2075 for (long j = 0; j < nb; ++j) {
2076 for (long k = 0; k < nAtoms; ++k) {
2077 const double sm = sqrtMasses[static_cast<size_t>(k)];
2078 Eigen::Vector3d d = Eigen::Vector3d::Zero();
2079 if (kind < 3) {
2080 d(kind) = sm;
2081 } else {
2082 const Eigen::Vector3d r =
2083 reference.segment<3>(3 * k) +
2084 q[static_cast<size_t>(j)].segment<3>(3 * k) / sm;
2085 Eigen::Vector3d e = Eigen::Vector3d::Zero();
2086 e(kind - 3) = 1.0;
2087 d = sm * e.cross(r - centre);
2088 }
2089 g.block(j * 3 * nAtoms + 3 * k, static_cast<long>(col), 3, 1) = d;
2090 }
2091 }
2092 }
2093 const ColMajorXd gc = g;
2094 const Eigen::ColPivHouseholderQR<ColMajorXd> qr(gc);
2095 const long rank = qr.rank();
2096 const ColMajorXd basis = qr.householderQ() * ColMajorXd::Identity(dim, rank);
2097 for (long r = 0; r < rank; ++r) {
2098 std::vector<VectorXd> u(static_cast<size_t>(nb));
2099 for (long j = 0; j < nb; ++j) {
2100 u[static_cast<size_t>(j)] =
2101 basis.col(r).segment(j * 3 * nAtoms, 3 * nAtoms);
2102 }
2103 out.push_back(std::move(u));
2104 }
2105 return out;
2106}

2800// converge walks down from 0.85 Tc, and that walk replaces the first try.
2801RateInstanton optimizeRateByNewton(const VectorXd &saddle,
2802 const MatrixXd &hessSaddle, double beta,
2803 std::vector<VectorXd> guess,
2804 const BatchPotential &potential,
2805 const RateInstantonOptions &options) {
2806 const long nBeads = options.beads;
2807 if (nBeads < 4 || !(beta > 0.0) || hessSaddle.rows() != saddle.size()) {
2808 throw std::invalid_argument(
2809 "optimizeRateInstanton: need N >= 4, beta > 0 and a saddle Hessian "
2810 "of the saddle's dimension");
2811 }
2812 RateInstanton inst;
2813 inst.beta = beta;
2814 inst.betaN = beta / static_cast<double>(nBeads);
2815 inst.temperature = 1.0 / (kBoltzmann * beta);
2816 inst.crossover = crossoverTemperature(hessSaddle);
2817 if (!(inst.temperature < inst.crossover)) {
2818 throw std::invalid_argument(
2819 "optimizeRateInstanton: T is at or above the crossover temperature; "
2820 "the ring collapses onto the saddle and steepest descent needs the "
2821 "parabolic barrier correction, of which classical transition-state "
2822 "theory is only the one-bead limit");
2823 }
2824
2825 const bool cool = static_cast<long>(guess.size()) != nBeads &&
2826 inst.temperature < 0.75 * inst.crossover;
2827 // One dimension solves the requested temperature before any walk. A
2828 // higher-dimensional empty guess below 0.75 Tc keeps the walk, which
2829 // starts from a curvature copied off the saddle.
2830 if (saddle.size() == 1 || !cool) {
2831 if (static_cast<long>(guess.size()) != nBeads) {
2832 const ColMajorXd hS = 0.5 * (hessSaddle + hessSaddle.transpose());
2833 const Eigen::SelfAdjointEigenSolver<ColMajorXd> es(hS);
2834 guess = cosineSeed(saddle, es.eigenvectors().col(0), es.eigenvalues()(0),
2835 inst.temperature, inst.crossover, nBeads, potential);
2836 }
2837 const double bnh = inst.betaN * kHbar;

2839 const NewtonOut got = newtonInstanton(std::move(guess), spring, hessSaddle,
2840 options, potential);
2841 inst.beads = got.beads;
2842 inst.energies = got.energies;
2843 inst.ringPotential = got.ringPotential;
2844 inst.bN = got.bN;
2845 inst.iterations = got.iterations;
2846 inst.converged = got.converged;
2847 inst.collapsed = got.collapsed;
2848 if (inst.converged || !cool) {
2849 return inst;
2850 }
2851 }
2852 if (cool) {
2853 std::vector<double> temps;
2854 for (double t = 0.85 * inst.crossover; t > inst.temperature * 1.05;
2855 t *= 0.75) {
2856 temps.push_back(t);
2857 }
2858 temps.push_back(inst.temperature);
2859 std::vector<VectorXd> beads;
2860 RateInstanton last;
2861 long used = 0;
2862 bool targetRan = false;
2863 const ColMajorXd hS = 0.5 * (hessSaddle + hessSaddle.transpose());
2864 const Eigen::SelfAdjointEigenSolver<ColMajorXd> es(hS);
2865 for (size_t s = 0; s < temps.size(); ++s) {
2866 const long remain = options.maxIterations - used;
2867 if (remain <= 0) {
2868 break;
2869 }
2870 RateInstantonOptions opt = options;
2871 opt.maxIterations = remain;
2872 const bool target = s + 1 == temps.size();
2873 opt.checkOddSector = options.checkOddSector && target;
2874 const double betaStage = target ? beta : 1.0 / (kBoltzmann * temps[s]);
2875 std::vector<VectorXd> stageGuess = beads;
2876 if (static_cast<long>(stageGuess.size()) != nBeads) {
2877 stageGuess =
2878 cosineSeed(saddle, es.eigenvectors().col(0), es.eigenvalues()(0),
2879 temps[s], inst.crossover, nBeads, potential);
2880 }
2881 last = optimizeRateInstanton(saddle, hessSaddle, betaStage,
2882 std::move(stageGuess), potential, opt);
2883 used += last.iterations;
2884 if (last.collapsed) {
2885 break;
2886 }
2887 beads = last.beads;
2888 if (target) {
2889 targetRan = true;
2890 }
2891 }
2892 last.iterations = used;
2893 if (targetRan) {
2894 last.beta = beta;
2895 last.betaN = beta / static_cast<double>(nBeads);
2896 last.temperature = inst.temperature;
2897 last.crossover = inst.crossover;
2898 } else {
2899 last.converged = false;
2900 }
2901 return last;
2902 }
2903
2904 if (static_cast<long>(guess.size()) != nBeads) {
2905 const ColMajorXd hS = 0.5 * (hessSaddle + hessSaddle.transpose());
2906 const Eigen::SelfAdjointEigenSolver<ColMajorXd> es(hS);
2907 guess = cosineSeed(saddle, es.eigenvectors().col(0), es.eigenvalues()(0),
2908 inst.temperature, inst.crossover, nBeads, potential);
2909 }
2910 const double bnh = inst.betaN * kHbar;
2911 const double spring = 1.0 / (bnh * bnh);
2912 NewtonOut got =
2913 newtonInstanton(std::move(guess), spring, hessSaddle, options, potential);
2914 // A half ring cannot see modes odd under j -> N-j, so two copies of the
2915 // instanton are a stationary point it can accept or sit on. An unstable
2916 // odd mode there sends the search onto the whole ring from a kick along
2917 // that mode.
2918 const long nGot = static_cast<long>(got.beads.size());
2919 bool mirrored = (got.converged || got.stalledHalf) && options.halfRing &&
2920 options.checkOddSector && nGot == nBeads && nGot % 2 == 0;
2921 for (long j = 1; mirrored && j < nGot / 2; ++j) {
2922 mirrored = (got.beads[static_cast<size_t>(j)] -
2923 got.beads[static_cast<size_t>(nGot - j)])
2924 .norm() <= 1e-8;
2925 }
2926 if (mirrored) {
2927 const Eigen::SelfAdjointEigenSolver<MatrixXd> es0(
2928 0.5 * (hessSaddle + hessSaddle.transpose()));
2929 const double barrierCurvature = std::abs(es0.eigenvalues()(0));
2930 const RingEval here =
2931 evaluateRing(got.beads, spring, potential, options.energyShift);
2932 std::vector<VectorXd> oddMode;
2933 const double oddCurv =
2934 lowestOddMode(got.beads, here, spring, potential, oddMode,
2935 options.lanczosFirst, options.lanczosStep);
2936 if (oddCurv < -1e-3 * barrierCurvature &&
2937 oddMode.size() == got.beads.size()) {
2938 std::vector<VectorXd> kicked = got.beads;
2939 const double kick = std::sqrt(2.0 / (inst.betaN * -oddCurv));
2940 for (size_t k = 0; k < kicked.size(); ++k) {
2941 kicked[k] += kick * oddMode[k];
2942 }
2943 RateInstantonOptions whole = options;
2944 whole.halfRing = false;
2945 const long before = got.iterations;
2946 got = newtonInstanton(std::move(kicked), spring, hessSaddle, whole,
2947 potential);
2948 got.iterations += before;
2949 }
2950 }
2951 inst.beads = got.beads;
2952 inst.energies = got.energies;
2953 inst.ringPotential = got.ringPotential;
2954 inst.bN = got.bN;
2955 inst.iterations = got.iterations;
2956 inst.converged = got.converged;
2957 inst.collapsed = got.collapsed;
2958 return inst;
2959}

2993 }
2994 }

2998 active * saddle.size() <= options.newtonLimit) {
2999 return optimizeRateByNewton(saddle, hessSaddle, beta, std::move(guess),
3000 potential, options);
3001 }
3002 const double bnh = inst.betaN * kHbar;
3003 const double c = 1.0 / (bnh * bnh);
3004
3005 const ColMajorXd saddleCurvature =
3006 0.5 * (hessSaddle + hessSaddle.transpose());

3008 const VectorXd dir = es.eigenvectors().col(0);
3009
3010 if (static_cast<long>(guess.size()) != N) {
3011 std::vector<double> v0;
3012 std::vector<VectorXd> g0;
3013 potential({saddle}, v0, g0);
3014 const double vS = v0.at(0);
3015 // Scan in steps of a quarter of the length over which the barrier's
3016 // curvature drops V by kB T_c.
3017 const double h = 0.25 * std::sqrt(2.0 * kBoltzmann * inst.crossover /
3018 -es.eigenvalues()(0));
3019 const long pts = 200;
3020 const double dPlus = sideDrop(saddle, dir, vS, 1.0, h, pts, potential);
3021 const double dMinus = sideDrop(saddle, dir, vS, -1.0, h, pts, potential);
3022 const double dMin = std::min(dPlus, dMinus);
3023 const double drop =
3024 (1.0 - inst.temperature / inst.crossover) *
3025 (std::isfinite(dMin) ? dMin : kBoltzmann * inst.crossover);
3026 const double sPlus =
3027 turningDistance(saddle, dir, vS, drop, 1.0, h, pts, potential);
3028 const double sMinus =
3029 turningDistance(saddle, dir, vS, drop, -1.0, h, pts, potential);
3030 guess.resize(static_cast<size_t>(N));
3031 for (long j = 0; j < N; ++j) {
3032 const double ct =
3033 std::cos(2.0 * std::numbers::pi * static_cast<double>(j) /
3034 static_cast<double>(N));
3035 guess[static_cast<size_t>(j)] =
3036 saddle + dir * (ct >= 0.0 ? sPlus * ct : sMinus * ct);
3037 }
3038 }
3039
3040 // An even count whose beads already match under j -> N-j is the
3041 // out-and-back instanton. The images are assigned equal, so the
3042 // potential on one half is copied, and the step stays on the closed ring.
3043 const bool wantMirror = options.halfRing && N % 2 == 0;
3044 std::vector<VectorXd> x = std::move(guess);
3045 bool fold = false;
3046 if (wantMirror) {
3047 fold = true;
3048 const long m = N / 2;
3049 for (long j = 1; j < m; ++j) {
3050 if ((x[static_cast<size_t>(j)] - x[static_cast<size_t>(N - j)]).norm() >
3051 1e-8) {
3052 fold = false;
3053 break;
3054 }
3055 }
3056 }
3057 BatchPotential evalPot = potential;
3058 if (fold) {
3059 evalPot = [&](const std::vector<VectorXd> &q, std::vector<double> &v,
3060 std::vector<VectorXd> &g) {
3061 const long m = N / 2;
3062 bool sym = true;
3063 for (long j = 1; j < m; ++j) {
3064 if ((q[static_cast<size_t>(j)] - q[static_cast<size_t>(N - j)])
3065 .squaredNorm() != 0.0) {
3066 sym = false;
3067 break;
3068 }
3069 }
3070 if (!sym) {
3071 potential(q, v, g);
3072 return;
3073 }
3074 std::vector<VectorXd> uniq(static_cast<size_t>(m + 1));
3075 for (long j = 0; j <= m; ++j) {
3076 uniq[static_cast<size_t>(j)] = q[static_cast<size_t>(j)];
3077 }
3078 std::vector<double> vu;
3079 std::vector<VectorXd> gu;
3080 potential(uniq, vu, gu);
3081 v.assign(static_cast<size_t>(N), 0.0);
3082 g.assign(static_cast<size_t>(N), VectorXd());
3083 for (long j = 0; j <= m; ++j) {
3084 v[static_cast<size_t>(j)] = vu[static_cast<size_t>(j)];
3085 g[static_cast<size_t>(j)] = gu[static_cast<size_t>(j)];
3086 }
3087 for (long j = 1; j < m; ++j) {
3088 v[static_cast<size_t>(N - j)] = vu[static_cast<size_t>(j)];
3089 g[static_cast<size_t>(N - j)] = gu[static_cast<size_t>(j)];
3090 }
3091 };
3092 }
3093 auto symmetrize = [&](std::vector<VectorXd> &q) {
3094 if (!fold) {
3095 return;
3096 }
3097 const long m = N / 2;
3098 for (long j = 1; j < m; ++j) {
3099 const size_t a = static_cast<size_t>(j);
3100 const size_t b = static_cast<size_t>(N - j);
3101 const VectorXd mid = 0.5 * (q[a] + q[b]);
3102 q[a] = mid;
3103 q[b] = mid;
3104 }
3105 };
3106 // The mirror of an out-and-back bounce is symmetric too. A monotone
3107 // reaction coordinate on beads 0..N/2 leaves that bounce out of the step.
3108 auto projectFold = [&](std::vector<VectorXd> &q) {
3109 if (!fold) {
3110 return;
3111 }
3112 projectMonotonePrefix(q, N / 2, dir);
3113 const long m = N / 2;
3114 for (long j = 1; j < m; ++j) {
3115 q[static_cast<size_t>(N - j)] = q[static_cast<size_t>(j)];
3116 }
3117 };
3118 symmetrize(x);
3119 projectFold(x);
3120
3121 RingEval cur = evaluateRing(x, c, evalPot, options.energyShift);
3122 // The unstable mode of the ring starts as every bead moving along the
3123 // saddle's unstable direction.
3124 std::vector<VectorXd> mode(x.size(), dir);
3125 double curvature = lowestMode(x, cur, c, evalPot, mode, options.lanczosFirst,
3126 options.lanczosStep, fold);
3127 std::deque<std::pair<std::vector<VectorXd>, std::vector<VectorXd>>> pairs;
3128 auto effective = [&](const std::vector<VectorXd> &g) {
3129 const double par = dot(g, mode);
3130 std::vector<VectorXd> e = g;
3131 const double f = curvature < 0.0 ? 2.0 : 1.0;
3132 for (size_t j = 0; j < e.size(); ++j) {
3133 e[j] -= f * par * mode[j];
3134 if (!(curvature < 0.0)) {
3135 e[j] = -par * mode[j]; // climb along the mode only
3136 }
3137 }
3138 return e;
3139 };
3140 std::vector<VectorXd> geff = effective(cur.grad);
3141 long limit = options.maxIterations;
3142 for (long it = 0; it < limit; ++it) {
3143 inst.iterations = it;
3144 if (curvature < 0.0 && largestBeadNorm(cur.grad) < options.forceTolerance) {
3145 std::vector<VectorXd> oddMode;
3146 const double oddCurv =
3147 fold ? lowestOddMode(x, cur, c, potential, oddMode,
3148 options.lanczosFirst, options.lanczosStep)
3149 : 0.0;
3150 if (!(oddCurv < -1e-3 * std::abs(curvature))) {
3151 inst.converged = true;
3152 break;
3153 }
3154 // A mirror-symmetric stationary point with a second unstable mode
3155 // odd under the mirror, such as two copies of the instanton on one
3156 // ring. The rest of the search runs on the whole ring from a kick
3157 // along that mode which lowers beta_N U_N by one in the quadratic
3158 // model.
3159 fold = false;
3160 evalPot = potential;
3161 const double kick = std::sqrt(2.0 / (inst.betaN * -oddCurv));
3162 for (size_t k = 0; k < x.size(); ++k) {
3163 x[k] += kick * oddMode[k];
3164 }
3165 cur = evaluateRing(x, c, evalPot, options.energyShift);
3166 curvature = lowestMode(x, cur, c, evalPot, mode, options.lanczosFirst,
3167 options.lanczosStep, fold);
3168 pairs.clear();
3169 geff = effective(cur.grad);
3170 limit = it + options.maxIterations;
3171 }
3172 std::vector<VectorXd> trial(x.size());
3173 std::vector<VectorXd> d = geff;
3174 std::vector<double> alpha(pairs.size());
3175 for (size_t i = pairs.size(); i-- > 0;) {
3176 const double rho = 1.0 / dot(pairs[i].second, pairs[i].first);
3177 alpha[i] = rho * dot(pairs[i].first, d);
3178 for (size_t k = 0; k < d.size(); ++k) {
3179 d[k] -= alpha[i] * pairs[i].second[k];
3180 }
3181 }
3182 double gamma = 1.0 / (4.0 * c); // spring stiffness sets the first scale
3183 if (!pairs.empty()) {
3184 gamma = dot(pairs.back().first, pairs.back().second) /
3185 dot(pairs.back().second, pairs.back().second);
3186 }
3187 scale(d, gamma);
3188 for (size_t i = 0; i < pairs.size(); ++i) {
3189 const double rho = 1.0 / dot(pairs[i].second, pairs[i].first);
3190 const double b = rho * dot(pairs[i].second, d);
3191 for (size_t k = 0; k < d.size(); ++k) {
3192 d[k] += (alpha[i] - b) * pairs[i].first[k];
3193 }
3194 }
3195 if (!(dot(d, geff) > 0.0)) { // not a descent direction on geff
3196 pairs.clear();

3198 scale(d, 1.0 / (4.0 * c));
3199 }
3200 const double big = largestBeadNorm(d);
3201 if (big > options.maxStep) {
3202 scale(d, options.maxStep / big);
3203 }
3204 for (size_t k = 0; k < x.size(); ++k) {
3205 trial[k] = x[k] - d[k];
3206 }
3207 symmetrize(trial);
3208 projectFold(trial);
3209 RingEval next = evaluateRing(trial, c, evalPot, options.energyShift);
3210 const double prevCurv = curvature;
3211 const long restart = options.lanczosRestart;
3212 curvature = lowestMode(trial, next, c, evalPot, mode, restart,
3213 options.lanczosStep, fold);
3214 std::vector<VectorXd> geffNext = effective(next.grad);
3215 if ((prevCurv < 0.0) != (curvature < 0.0)) {
3216 pairs.clear();
3217 } else {
3218 std::vector<VectorXd> sk(x.size()), yk(x.size());
3219 for (size_t k = 0; k < x.size(); ++k) {
3220 sk[k] = trial[k] - x[k];
3221 yk[k] = geffNext[k] - geff[k];
3222 }
3223 if (dot(sk, yk) > 0.0) {
3224 pairs.emplace_back(std::move(sk), std::move(yk));
3225 if (static_cast<long>(pairs.size()) > options.memory) {
3226 pairs.pop_front();
3227 }
3228 }
3229 }
3230 x = std::move(trial);
3231 cur = std::move(next);
3232 geff = std::move(geffNext);
3233 }
3234 if (!inst.converged && curvature < 0.0 &&
3235 largestBeadNorm(cur.grad) < options.forceTolerance) {
3236 inst.converged = true;
3237 }
3238 inst.beads = x;
3239 inst.energies = cur.v;
3240 inst.ringPotential = cur.u;
3241 inst.bN = 0.0;
3242 for (size_t j = 0; j < x.size(); ++j) {
3243 inst.bN += (x[(j + 1) % x.size()] - x[j]).squaredNorm();
3244 }
3245 return inst;
3246}

3270void instantonRate(RateInstanton &inst, const RingBeadHessian &hessian,
3271 const MatrixXd &hessReactant, double vReactant,
3272 const MatrixXd &hessSaddle, double vSaddle, long rigidModes,
3273 long denseLimit, const RingRigidBodies &rigidBodies) {
3274 const long N = static_cast<long>(inst.beads.size());
3275 if (N < 4 || !(inst.betaN > 0.0)) {
3276 throw std::invalid_argument("instantonRate: no optimised ring");
3277 }
3278 const long f = inst.beads.front().size();
3279 if (rigidModes < 0 || rigidModes > f) {
3280 throw std::invalid_argument(
3281 "instantonRate: rigidModes exceeds the degrees of freedom");
3282 }
3283 const double bnh = inst.betaN * kHbar;
3284 const double c = 1.0 / (bnh * bnh);
3285 const MatrixXd eye = MatrixXd::Identity(f, f);
3286
3287 std::vector<MatrixXd> hBead(static_cast<size_t>(N));
3288 std::vector<MatrixXd> diag(static_cast<size_t>(N));
3289 for (long j = 0; j < N; ++j) {
3290 const MatrixXd h = hessian(j, inst.beads[static_cast<size_t>(j)]);
3291 if (h.rows() != f || h.cols() != f) {
3292 throw std::runtime_error("instantonRate: bead Hessian size");
3293 }
3294 hBead[static_cast<size_t>(j)] = 0.5 * (h + h.transpose());
3295 diag[static_cast<size_t>(j)] =
3296 hBead[static_cast<size_t>(j)] + 2.0 * c * eye;
3297 }
3298
3299 const Eigen::SelfAdjointEigenSolver<MatrixXd> er(
3300 0.5 * (hessReactant + hessReactant.transpose()));
3301 const VectorXd &lr = er.eigenvalues();
3302 const std::vector<bool> rigidR = nearestZero(lr, rigidModes);
3303 MatrixXd nullBasis(f, 0);
3304 // The rigid vectors leave the product whether or not every bead Hessian
3305 // annihilates them exactly; a finite-difference Hessian never does, and
3306 // the reactant side drops the same modes at k = 0.
3307 for (long m = 0; m < lr.size(); ++m) {
3308 if (!rigidR[static_cast<size_t>(m)]) {
3309 continue;
3310 }
3311 nullBasis.conservativeResize(f, nullBasis.cols() + 1);
3312 nullBasis.col(nullBasis.cols() - 1) = er.eigenvectors().col(m);
3313 }
3314 if (rigidModes >= f) {
3315 throw std::runtime_error("instantonRate: every direction is a rigid mode");
3316 }
3317 // det' through the block chain: the cyclic zero mode and the rigid null
3318 // vectors leave the product by the determinant lemma, the inertia comes
3319 // from the Schur complements, and the N f by N f matrix is never formed.
3320 std::vector<VectorXd> cycle(static_cast<size_t>(N));
3321 double cycleNorm = 0.0;
3322 for (long j = 0; j < N; ++j) {
