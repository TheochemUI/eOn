29// Helper: enum <-> string via magic_enum
30template <typename E> static json enum_to_json(E val) {
31 return std::string(magic_enum::enum_name(val));
32}
33
34template <typename E> static E enum_from_json(const json &j, E fallback) {
35 if (j.is_string()) {
36 auto result = magic_enum::enum_cast<E>(j.get<std::string>(),
37 magic_enum::case_insensitive);
38 return result.value_or(fallback);
39 }
40 return fallback;
41}

43// Macro for optional JSON field extraction with default
44#define JSON_OPT(j, key, target) \
45 if ((j).contains(key)) \
46 (target) = (j).at(key).get<decltype(target)>()

369static std::string lowerCopy(std::string value) {
370 for (char &ch : value) {
371 ch = static_cast<char>(std::tolower(static_cast<unsigned char>(ch)));
372 }
373 return value;
374}

376static std::vector<double> temperaturesFromJson(const json &value) {
377 if (value.is_array()) {
378 return value.get<std::vector<double>>();
379 }
380 if (!value.is_string()) {
381 throw std::invalid_argument(
382 "[Instanton] temperatures must be a list or a comma-separated "
383 "string");
384 }
385 std::vector<double> out;
386 std::stringstream ss(value.get<std::string>());
387 std::string token;
388 while (std::getline(ss, token, ',')) {
389 const size_t start = token.find_first_not_of(" \t");
390 const size_t end = token.find_last_not_of(" \t");
391 if (start == std::string::npos) {
392 continue;
393 }
394 try {
395 out.push_back(std::stod(token.substr(start, end - start + 1)));
396 } catch (const std::exception &) {
397 throw std::invalid_argument(
398 "[Instanton] temperatures must be comma-separated kelvin "
399 "values, not " +
400 token);
401 }
402 }
403 return out;
404}

406// Path-integral keys live on [Dynamics] in an ini file and under Thermostat
407// in the JSON writer. Either object may carry them. The damping time is
408// stored in femtoseconds; from_json converts it once every object is read.
409static void readPathIntegralKeys(const json &s, Parameters &p) {
410 auto &th = ParametersLoadAccess::thermostat_options(p);
411 JSON_OPT(s, "path_beads", th.path_beads);
412 if (s.contains("path_springs")) {
413 th.path_springs = lowerCopy(s.at("path_springs").get<std::string>());
414 if (th.path_springs != "trotter" && th.path_springs != "eco") {
415 throw std::invalid_argument(
416 "[Dynamics] path_springs must be trotter or eco, not " +
417 th.path_springs);
418 }
419 }
420 JSON_OPT(s, "path_eco_omega_max", th.path_eco_omega_max);
421 JSON_OPT(s, "path_gle_file", th.path_gle_file);
422 JSON_OPT(s, "path_pile_tau", th.path_pile_tau_input);
423 JSON_OPT(s, "path_pile_scale", th.path_pile_scale);
424 if (s.contains("path_seed")) {
425 const long seed = s.at("path_seed").get<long>();
426 if (seed < 0) {
427 throw std::invalid_argument("[Dynamics] path_seed must be non-negative");
428 }
429 th.path_seed = static_cast<std::uint64_t>(seed);
430 }
431}

433static void readBathKeys(const json &s, Parameters &p) {
434 auto &th = ParametersLoadAccess::thermostat_options(p);
435 if (s.contains("thermostat")) {
436 th.kind = lowerCopy(s.at("thermostat").get<std::string>());
437 }
438 if (s.contains("kind")) {
439 th.kind = lowerCopy(s.at("kind").get<std::string>());
440 }
441 JSON_OPT(s, "andersen_alpha", th.andersen_alpha);
442 JSON_OPT(s, "andersen_collision_period", th.andersen_tcol_input);
443 JSON_OPT(s, "nose_mass", th.nose_mass);
444 JSON_OPT(s, "langevin_friction", th.langevin_friction_input);
445 readPathIntegralKeys(s, p);
446}
