81 void forceBatch(long nSystems, long nAtoms, const double *const *positions,
82 const int *const *atomicNrs, double *const *forces,
83 double *energies, double *variances,
84 const double *const *boxes) override {
85 if (!fn_) {
86 throw std::runtime_error("HostSurfacePotential: NULL surface");
87 }
88 const long npos = 3 * nAtoms * nSystems;
89 const long nbox = 9 * nSystems;
90 std::vector<double> pos(static_cast<size_t>(npos), 0.0);
91 std::vector<double> frc(static_cast<size_t>(npos), 0.0);
92 std::vector<double> box(static_cast<size_t>(nbox), 0.0);
93 std::vector<int32_t> z(static_cast<size_t>(nAtoms), 0);
94 std::vector<int64_t> ids(static_cast<size_t>(nSystems), 0);
95 if (atomicNrs && atomicNrs[0]) {
96 for (long a = 0; a < nAtoms; ++a) {
97 z[static_cast<size_t>(a)] = atomicNrs[0][a];
98 }
99 }
100 for (long s = 0; s < nSystems; ++s) {
101 int64_t id = EON_RELAX_IMAGE_NONE;
102 if (positions && positions[s]) {
103 for (size_t i = 0; i < path_.size(); ++i) {
104 if (path_[i] && path_[i]->getPositions().data() == positions[s]) {
105 id = static_cast<int64_t>(i);
106 break;
107 }
108 }
109 }
110 ids[static_cast<size_t>(s)] = id;
111 if (positions && positions[s]) {
112 for (long k = 0; k < 3 * nAtoms; ++k) {
113 pos[static_cast<size_t>(s * 3 * nAtoms + k)] = positions[s][k];
114 }
115 }
116 if (boxes && boxes[s]) {
117 for (long k = 0; k < 9; ++k) {
118 box[static_cast<size_t>(s * 9 + k)] = boxes[s][k];
119 }
120 }
121 }
122 std::vector<double> ebuf(static_cast<size_t>(nSystems), 0.0);
123 std::vector<double> vbuf(static_cast<size_t>(nSystems), 0.0);
124 std::uint64_t new_epoch = epoch_;
125 eon_relax_surface_request_t req{};
126 req.version.major = EON_RELAX_ABI_MAJOR;
127 req.version.minor = EON_RELAX_ABI_MINOR;
128 req.n_images = static_cast<int64_t>(nSystems);
129 req.n_atoms = static_cast<int64_t>(nAtoms);
130 req.positions = pos.data();
131 req.atomic_nrs = z.data();
132 req.boxes = box.data();
133 req.image_ids = ids.data();
134 req.energies = ebuf.data();
135 req.forces = frc.data();
136 req.variances = vbuf.data();
137 req.epoch_out = &new_epoch;
138 last_ = fn_(user_, &req);
139 if (last_ < 0) {
140 throw std::runtime_error("HostSurfacePotential: surface failed");
141 }
142 if (last_ > 0) {
143 throw SurfaceRecoverable(last_);
144 }
145 if (new_epoch != epoch_) {
146 epoch_ = new_epoch;
147 }
148 if (count_calls_) {
149 for (long s = 0; s < nSystems; ++s) {
150 forceCallCounter++;
151 PotRegistry::get().on_force_call(ptype);
152 }
153 }
154 for (long s = 0; s < nSystems; ++s) {
155 if (energies) {
156 energies[s] = ebuf[static_cast<size_t>(s)];
157 }
158 if (variances) {
159 variances[s] = vbuf[static_cast<size_t>(s)];
160 }
161 if (forces && forces[s]) {
162 for (long k = 0; k < 3 * nAtoms; ++k) {
163 forces[s][k] = frc[static_cast<size_t>(s * 3 * nAtoms + k)];
164 }
165 }
166 }
167 }

169private:
170 eon_relax_surface_fn fn_;
171 void *user_;
172 std::uint64_t epoch_{0};
173 int last_{0};
174 bool count_calls_{true};
175 std::vector<std::shared_ptr<Matter>> path_;
176};
