31// Position / difference MIC conventions (issue #176). Legacy matches historical
32// applyPeriodicBoundary (fmod to [0,1) fractional). MinimumImage matches the
33// existing eonc::pbc::apply path used for interatomic distances (frac in
34// [-0.5,0.5)).
35enum class PbcConvention {
36 Legacy = 0,
37 MinimumImage = 1,
38};
39
40namespace pbc {
41
42// Minimum-image on a difference vector (fractional in [-0.5, 0.5)).
43inline AtomMatrix apply(const AtomMatrix &diff, const Matrix3d &cell,
44 const Matrix3d &cellInverse) {
45 // Fractional coordinates, minimum-image wrap to [-0.5, 0.5), back to Cartesian.
46 // x - floor(x + 0.5), Highway when the build has it.
47 AtomMatrix frac = diff * cellInverse;
48 wrapMinimumImage(frac.data(), static_cast<size_t>(frac.size()));
49 return frac * cell;
50}
51
52// Legacy position wrap: fractional coords into [0, 1) via fmod (historical
53// Matter).
54inline AtomMatrix applyLegacy(const AtomMatrix &coords, const Matrix3d &cell,
55 const Matrix3d &cellInverse) {
56 AtomMatrix frac = coords * cellInverse;
57 // Same values as fmod(x + 1, 1): (x + 1) - trunc(x + 1).
58 wrapLegacyUnit(frac.data(), static_cast<size_t>(frac.size()));
59 return frac * cell;
60}
61
62// Position wrap selecting convention (Legacy vs MinimumImage centering).
63inline AtomMatrix applyPositions(const AtomMatrix &coords, const Matrix3d &cell,
64 const Matrix3d &cellInverse,
65 PbcConvention convention) {
66 if (convention == PbcConvention::MinimumImage) {
67 return apply(coords, cell, cellInverse);
68 }
69 return applyLegacy(coords, cell, cellInverse);
70}
71
72inline VectorXd applyV(const VectorXd &diffVector, const Matrix3d &cell,
73 const Matrix3d &cellInverse) {
74 AtomMatrix pbcMatrix =
75 apply(AtomMatrix::Map(diffVector.data(), diffVector.size() / 3, 3), cell,
76 cellInverse);
77 return VectorXd(VectorXd::Map(pbcMatrix.data(), diffVector.size()));
78}
